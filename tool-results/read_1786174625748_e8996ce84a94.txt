   290→  1135→// ============================================================================
   291→  1136→// Bookie Odds Dampener — Heavy Favorite Adjustment
   292→  1137→// ============================================================================
   293→  1138→// Based on live results analysis: when bookie odds for one side are <=1.70,
   294→  1139→// the model systematically overestimates O2.5 probability by 10-15%.
   295→  1140→// Heavy favorites tend to win tight, controlling games (1-0, 2-0) rather
   296→  1141→// than high-scoring shootouts. This dampener accounts for that pattern.
   297→  1142→// ============================================================================
   298→  1143→
   299→  1144→/** Heavy favorite odds threshold (decimal odds) */
   300→  1145→const HEAVY_FAVORITE_ODDS = 1.70;
   301→  1146→
   302→  1147→/** Maximum dampener amount (percentage points to subtract) */
   303→  1148→const HEAVY_FAVORITE_DAMPENER = 10;
   304→  1149→
   305→  1150→/**
   306→  1151→ * Apply bookie odds dampener to goal market probabilities.
   307→  1152→ * When either team is a heavy favorite (avg odds <=1.70), reduce O2.5/O3.5/BTTS
   308→  1153→ * probabilities to account for the "park the bus" effect.
   309→  1154→ *
   310→  1155→ * Only applies when valid odds are available (non-null, positive).
   311→  1156→ *
   312→  1157→ * @param over25Prob  - Model O2.5 probability (%)
   313→  1158→ * @param over35Prob  - Model O3.5 probability (%)
   314→  1159→ * @param bttsProb    - Model BTTS probability (%)
   315→  1160→ * @param homeAvgOdds - Home team average odds (decimal)
   316→  1161→ * @param awayAvgOdds - Away team average odds (decimal)
   317→  1162→ * @returns Adjusted probabilities with dampener applied
   318→  1163→ */
   319→  1164→export function applyBookieOddsDampener(
   320→  1165→  over25Prob: number,
   321→  1166→  over35Prob: number,
   322→  1167→  bttsProb: number,
   323→  1168→  homeAvgOdds: number | null | undefined,
   324→  1169→  awayAvgOdds: number | null | undefined,
   325→  1170→  leagueAvgGoalsPerGame?: number
   326→  1171→): { over25: number; over35: number; btts: number; dampened: boolean; reason: string } {
   327→  1172→  if (!homeAvgOdds || !awayAvgOdds || homeAvgOdds <= 0 || awayAvgOdds <= 0) {
   328→  1173→    return { over25: over25Prob, over35: over35Prob, btts: bttsProb, dampened: false, reason: '' };
   329→  1174→  }
   330→  1175→
   331→  1176→  // League-adaptive: gentler dampener in high-variance leagues
   332→  1177→  const variance = leagueAvgGoalsPerGame ? classifyLeagueVariance(leagueAvgGoalsPerGame) : 'low';
   333→  1178→  const heavyThreshold = variance === 'high' ? 1.40 : variance === 'medium' ? 1.60 : HEAVY_FAVORITE_ODDS;
   334→  1179→  const maxDampener = variance === 'high' ? 5 : variance === 'medium' ? 7 : HEAVY_FAVORITE_DAMPENER;
   335→  1180→
   336→  1181→  const homeIsHeavy = homeAvgOdds <= heavyThreshold;
   337→  1182→  const awayIsHeavy = awayAvgOdds <= heavyThreshold;
   338→  1183→
   339→  1184→  if (!homeIsHeavy && !awayIsHeavy) {
   340→  1185→    return { over25: over25Prob, over35: over35Prob, btts: bttsProb, dampened: false, reason: '' };
   341→  1186→  }
   342→  1187→
   343→  1188→  const heavyOdds = Math.min(homeAvgOdds, awayAvgOdds);
   344→  1189→  // Scale dampener: heavier favorite = bigger reduction
   345→  1190→  // At 1.17 (PSG level): full dampener
   346→  1191→  // At threshold: minimal 2% dampener
   347→  1192→  const minDampener = variance === 'high' ? 2 : 3;
   348→  1193→  const scale = Math.max(0, 1 - (heavyOdds - 1.17) / (heavyThreshold - 1.17));
   349→  1194→  const dampener = Math.round(minDampener + (maxDampener - minDampener) * scale);
   350→  1195→
   351→  1196→  const leagueLabel = variance.charAt(0).toUpperCase() + variance.slice(1);
   352→  1197→  const reason = homeIsHeavy
   353→  1198→    ? `${leagueLabel}-variance: Heavy home favorite (${homeAvgOdds.toFixed(2)}): -${dampener}% dampener`
   354→  1199→    : `${leagueLabel}-variance: Heavy away favorite (${awayAvgOdds.toFixed(2)}): -${dampener}% dampener`;
   355→  1200→
   356→  1201→  return {
   357→  1202→    over25: Math.max(30, over25Prob - dampener),
   358→  1203→    over35: Math.max(20, over35Prob - dampener),
   359→  1204→    btts: Math.max(25, bttsProb - Math.round(dampener * 0.5)), // BTTS dampened less
   360→  1205→    dampened: true,
   361→  1206→    reason,
   362→  1207→  };
   363→  1208→}

   365→  1210→/**
   366→  1211→ * Compute average bookie odds for a team from match results.
   367→  1212→ * Averages odds across all matches for that team.
   368→  1213→ *
   369→  1214→ * @param matches - Match results with odds data
   370→  1215→ * @param team    - Team name
   371→  1216→ * @returns Average decimal odds, or null if insufficient data
   372→  1217→ */
   373→  1218→export function computeTeamAvgOdds(
   374→  1219→  matches: { homeTeam: string; awayTeam: string; oddsAvgHome: number | null; oddsAvgAway: number | null }[],
   375→  1220→  team: string
   376→  1221→): number | null {
   377→  1222→  const teamOdds: number[] = [];
   378→  1223→
   379→  1224→  for (const m of matches) {
   380→  1225→    const odds = m.homeTeam === team ? m.oddsAvgHome : (m.awayTeam === team ? m.oddsAvgAway : null);
   381→  1226→    if (odds && odds > 1 && odds < 20) {
   382→  1227→      teamOdds.push(odds);
   383→  1228→    }
   384→  1229→  }
   385→  1230→
   386→  1231→  if (teamOdds.length < 3) return null;
   387→  1232→
   388→  1233→  return teamOdds.reduce((s, o) => s + o, 0) / teamOdds.length;
   389→  1234→}

   392→  1237→// League Variance Classification — Used by blowout/dampener adaptivity
   393→  1238→// ============================================================================
   394→  1239→// Classifies a league by its historical goal variance so that blowout and
   395→  1240→// dampener thresholds can be calibrated to the league's natural scoring profile.
   396→  1241→// High-variance leagues (2. Bundesliga, Eredivisie) see more goal explosions
   397→  1242→// from heavy favorites, so suppression mechanisms should be gentler.
   398→  1243→
   399→  1244→export type LeagueVariance = 'high' | 'medium' | 'low';
   400→  1245→
   401→  1246→export function classifyLeagueVariance(avgGoalsPerGame: number): LeagueVariance {
   402→  1247→  if (avgGoalsPerGame >= 3.0) return 'high';
   403→  1248→  if (avgGoalsPerGame >= 2.5) return 'medium';
   404→  1249→  return 'low';
   405→  1250→}

   407→  1252→// ============================================================================
   408→  1253→// Blowout Risk Qualifier — Override Under signals in massive quality gaps
   409→  1254→// ============================================================================
   410→  1255→// Based on live results analysis: when there's a huge quality gap (favorite's
   411→  1256→// avg odds <= 1.50), the Regression "Under" signal is unreliable. Heavy favorites
   412→  1257→// tend to dominate regardless of recent form — a "cooling off" period doesn't
   413→  1258→// matter when you're 2-3 quality tiers above the opponent.
   414→  1259→//
   415→  1260→// This qualifier prevents the "Under" regression signal from canceling out
   416→  1261→// legitimate "Over" signals (xG, Z-Score) in STRONG BET / GREY RESULT / GOAL FEST.
   417→  1262→//
   418→  1263→// Example: Man City (odds 1.20) vs Luton — Regression says "Under" (City only
   419→  1264→// scored 1 goal last 3 games), but City will likely score 3-4 regardless.
   420→  1265→// Without this qualifier, the Under signal would block 2 points in STRONG BET.
   421→  1266→// ============================================================================
   422→  1267→
   423→  1268→/** Blowout threshold: favorite's average odds at or below this triggers override */
   424→  1269→const BLOWOUT_ODDS_THRESHOLD = 1.50;
   425→  1270→
   426→  1271→/**
   427→  1272→ * Apply blowout risk qualifier to regression signal.
   428→  1273→ * When either team is a massive favorite (avg odds <= 1.50), override
   429→  1274→ * Regression "Under"/"Strong Under" to "Neutral" because quality advantage
   430→  1275→ * outweighs recent form in one-sided matchups.
   431→  1276→ *
   432→  1277→ * @param signals      - Current signal set (xgSignal, regressionSignal, zScoreSignal)
   433→  1278→ * @param homeAvgOdds  - Home team average odds (decimal)
   434→  1279→ * @param awayAvgOdds  - Away team average odds (decimal)
   435→  1280→ * @returns Modified signals (regression may be overridden to Neutral)
   436→  1281→ */
   437→  1282→export function applyBlowoutQualifier(
   438→  1283→  signals: SignalInput,
   439→  1284→  homeAvgOdds: number | null | undefined,
   440→  1285→  awayAvgOdds: number | null | undefined,
   441→  1286→  leagueAvgGoalsPerGame?: number
   442→  1287→): { signals: SignalInput; blowoutOverride: boolean; reason: string } {
   443→  1288→  // Only applies when valid odds are available
   444→  1289→  if (!homeAvgOdds || !awayAvgOdds || homeAvgOdds <= 0 || awayAvgOdds <= 0) {
   445→  1290→    return { signals, blowoutOverride: false, reason: '' };
   446→  1291→  }
   447→  1292→
   448→  1293→  // League-adaptive threshold: gentler in high-variance leagues
   449→  1294→  const variance = leagueAvgGoalsPerGame ? classifyLeagueVariance(leagueAvgGoalsPerGame) : 'low';
   450→  1295→  const blowoutThreshold = variance === 'high' ? 1.25 : variance === 'medium' ? 1.40 : BLOWOUT_ODDS_THRESHOLD;
   451→  1296→
   452→  1297→  const homeIsMassiveFavorite = homeAvgOdds <= blowoutThreshold;
   453→  1298→  const awayIsMassiveFavorite = awayAvgOdds <= blowoutThreshold;
   454→  1299→
   455→  1300→  // Only trigger when one team is a massive favorite
   456→  1301→  if (!homeIsMassiveFavorite && !awayIsMassiveFavorite) {
   457→  1302→    return { signals, blowoutOverride: false, reason: '' };
   458→  1303→  }
   459→  1304→
   460→  1305→  // In high-variance leagues, only override Strong Under (keep mild signals intact)
   461→  1306→  // because hot favorites in goal-rich leagues often score freely
   462→  1307→  const shouldOverride = variance === 'high'
   463→  1308→    ? signals.regressionSignal === 'Strong Under'
   464→  1309→    : signals.regressionSignal === 'Under' || signals.regressionSignal === 'Strong Under';
   465→  1310→  if (!shouldOverride) {
   466→  1311→    return { signals, blowoutOverride: false, reason: '' };
   467→  1312→  }
   468→  1313→
   469→  1314→  const favoriteOdds = Math.min(homeAvgOdds, awayAvgOdds);
   470→  1315→  const favorite = homeIsMassiveFavorite ? 'Home' : 'Away';
   471→  1316→  const reason = `Blowout qualifier (${variance}-variance): ${favorite} favorite (${favoriteOdds.toFixed(2)} <= ${blowoutThreshold}) — Regression "${signals.regressionSignal}" overridden to Neutral`;
   472→  1317→
   473→  1318→  return {
   474→  1319→    signals: {
   475→  1320→      ...signals,
   476→  1321→      regressionSignal: 'Neutral',
   477→  1322→    },
   478→  1323→    blowoutOverride: true,
   479→  1324→    reason,
   480→  1325→  };
   481→  1326→}

   569→  1414→// While Regression asks "is this team due to bounce back?", Momentum asks
   570→  1415→// "is this team on a sustained hot streak that will continue?"
   571→  1416→// Both tracks run simultaneously; a match can qualify through either pathway.

   573→  1418→export interface MomentumTeamInput {
   574→  1419→  last5Goals: number[];    // total goals per match in last 5
   575→  1420→  seasonAvg: number;       // season avg total goals per game
   576→  1421→  xgDiff: number;          // actualGoals/matches - totalXg/matches (positive = overperforming)
   577→  1422→  sotPerGame: number | null;
   578→  1423→}
   579→  1424→
   580→  1425→export interface MomentumResult {
   581→  1426→  homeSignal: string;      // 'Strong Momentum' | 'Momentum' | 'Neutral' | 'Cooling' | 'Cold'
   582→  1427→  awaySignal: string;
   583→  1428→  matchSignal: string;     // 'MOMENTUM OVER' | 'MOMENTUM LEAN OVER' | 'MOMENTUM NEUTRAL' | 'MOMENTUM LEAN UNDER' | 'MOMENTUM UNDER'
   584→  1429→}
   585→  1430→
   586→  1431→function classifyTeamMomentum(team: MomentumTeamInput): string {
   587→  1432→  if (team.last5Goals.length < 3) return 'Neutral';
   588→  1433→
   589→  1434→  const last5Avg = team.last5Goals.reduce((s, g) => s + g, 0) / team.last5Goals.length;
   590→  1435→  const last3 = team.last5Goals.slice(0, Math.min(3, team.last5Goals.length));
   591→  1436→  const last3Avg = last3.reduce((s, g) => s + g, 0) / last3.length;
   592→  1437→
   593→  1438→  // Consistency: coefficient of variation of last 5
   594→  1439→  const mean = last5Avg;
   595→  1440→  const variance = team.last5Goals.reduce((s, g) => s + (g - mean) ** 2, 0) / team.last5Goals.length;
   596→  1441→  const stdDev = Math.sqrt(variance);
   597→  1442→  const cv = mean > 0 ? stdDev / mean : 999;
   598→  1443→
   599→  1444→  // Sustained: 5+ game window above season average by meaningful margin
   600→  1445→  const sustained = last5Avg > team.seasonAvg * 1.15;
   601→  1446→
   602→  1447→  // Accelerating: last 3 even hotter than last 5
   603→  1448→  const accelerating = last3Avg > last5Avg * 1.05;
   604→  1449→
   605→  1450→  // xG sustainability: overperforming xG with shot volume to back it
   606→  1451→  const xgSustainable = team.xgDiff > 0 && team.sotPerGame !== null && team.sotPerGame > 5.0;
   607→  1452→
   608→  1453→  // Strong Momentum: sustained + consistent + high output
   609→  1454→  if (sustained && cv < 0.5 && last5Avg >= 2.0) return 'Strong Momentum';
   610→  1455→  // Momentum: sustained with moderate variance, or recent surge with sustainable backing
   611→  1456→  if (sustained && cv < 0.65) return 'Momentum';
   612→  1457→  if (last5Avg > team.seasonAvg * 1.15 && accelerating && (xgSustainable || team.xgDiff <= 0)) return 'Momentum';
   613→  1458→  // Cold: declining output
   614→  1459→  if (last5Avg < team.seasonAvg * 0.75) {
   615→  1460→    return last3Avg < last5Avg * 0.8 ? 'Cold' : 'Cooling';
   616→  1461→  }
   617→  1462→  return 'Neutral';
   618→  1463→}

   620→  1465→export function computeMomentumSignal(home: MomentumTeamInput, away: MomentumTeamInput): MomentumResult {
   621→  1466→  const homeSignal = classifyTeamMomentum(home);
   622→  1467→  const awaySignal = classifyTeamMomentum(away);
   623→  1468→
   624→  1469→  const isOver = (s: string) => s === 'Strong Momentum' || s === 'Momentum';
   625→  1470→  const isUnder = (s: string) => s === 'Cold' || s === 'Cooling';
   626→  1471→
   627→  1472→  let matchSignal: string;
   628→  1473→  if (isOver(homeSignal) && isOver(awaySignal)) matchSignal = 'MOMENTUM OVER';
   629→  1474→  else if ((isOver(homeSignal) || isOver(awaySignal)) && !isUnder(homeSignal) && !isUnder(awaySignal)) matchSignal = 'MOMENTUM LEAN OVER';
   630→  1475→  else if (isUnder(homeSignal) && isUnder(awaySignal)) matchSignal = 'MOMENTUM UNDER';
   631→  1476→  else if (isUnder(homeSignal) || isUnder(awaySignal)) matchSignal = 'MOMENTUM LEAN UNDER';
   632→  1477→  else matchSignal = 'MOMENTUM NEUTRAL';
   633→  1478→
   634→  1479→  return { homeSignal, awaySignal, matchSignal };
   635→  1480→}

   637→  1482→// ============================================================================
   638→  1483→// BTTS Dual-Team Qualification System
   639→  1484→// ============================================================================
   640→  1485→// Evaluates each team independently to determine whether BOTH teams are likely
   641→  1486→// to score. Returns a tier that caps BTTS probability independently from the
   642→  1487→// Over goal thesis.
   643→  1488→//
   644→  1489→// Now supports TWO qualification pathways:
   645→  1490→//   1. Regression Under: both teams underperforming → due to bounce back
   646→  1491→//   2. Momentum Over: both teams on hot streaks → likely to keep scoring
   647→  1492→//
   648→  1493→// Tier mapping:
   649→  1494→//   5+ → BTTS STRONG   (cap 70%) — both teams individually qualify
   650→  1495→//   3–4 → BTTS QUALIFIED (cap 55%) — mild BTTS confidence
   651→  1496→//   1–2 → BTTS WEAK     (cap 40%) — one-sided regression likely
   652→  1497→//   ≤0  → BTTS AVOID    (cap 25%) — do not bet BTTS
   653→  1498→// ============================================================================
   654→  1499→export function computeBTTSQualification(input: BttsQualificationInput): BttsQualificationResult {
   655→  1500→  let score = 0;
   656→  1501→  const checks: { check: string; points: number; passed: boolean }[] = [];
   657→  1502→
   658→  1503→  const isRegressionUnder = (sig: string) => sig === 'Under' || sig === 'Strong Under';
   659→  1504→  const isXgOver = (diff: number) => diff <= -0.5; // underperforming xG = goals due
   660→  1505→  const isZCold = (z: number) => z >= 1.0; // overperforming Z = cold streak coming
   661→  1506→  const isZNeutralOrBetter = (z: number) => z < 1.0; // not overheated
   662→  1507→
   663→  1508→  // Momentum helpers
   664→  1509→  const isMomentum = (s: string | undefined) => s === 'Strong Momentum' || s === 'Momentum';
   665→  1510→  const homeMom = isMomentum(input.homeMomentumSignal);
   666→  1511→  const awayMom = isMomentum(input.awayMomentumSignal);
   667→  1512→  const bothMom = homeMom && awayMom;
   668→  1513→  const oneMom = homeMom || awayMom;
   669→  1514→
   670→  1515→  // Check 1: Regression Under OR Momentum Over — dual qualification pathway
   671→  1516→  const bothRegUnder = isRegressionUnder(input.homeRegressionSignal) && isRegressionUnder(input.awayRegressionSignal);
   672→  1517→  const oneRegUnder = isRegressionUnder(input.homeRegressionSignal) || isRegressionUnder(input.awayRegressionSignal);
   673→  1518→
   674→  1519→  if (bothRegUnder && bothMom) {
   675→  1520→    // Supercharged: both signals agree from both sides
   676→  1521→    score += 5;
   677→  1522→    checks.push({ check: 'Both Regression Under AND Momentum Over (supercharged)', points: 5, passed: true });
   678→  1523→  } else if (bothRegUnder || bothMom) {
   679→  1524→    score += 3;
   680→  1525→    checks.push({ check: bothRegUnder ? 'Both teams Regression Under/StrongUnder' : 'Both teams Momentum Over (hot streak)', points: 3, passed: true });
   681→  1526→  } else if ((oneRegUnder && oneMom)) {
   682→  1527→    score += 2; // Mixed but both signal types point to goals
   683→  1528→    checks.push({ check: 'One Regression Under + One Momentum Over', points: 2, passed: true });
   684→  1529→  } else if (oneRegUnder || oneMom) {
   685→  1530→    score += 1;
   686→  1531→    checks.push({ check: oneRegUnder ? 'One team Regression Under' : 'One team Momentum Over', points: 1, passed: true });
   687→  1532→  } else {
   688→  1533→    checks.push({ check: 'No Regression Under or Momentum Over', points: 3, passed: false });
   689→  1534→  }
   690→  1535→
   691→  1536→  // Check 2: BOTH teams xG Over (underperforming xG = creating chances, finishing due)
   692→  1537→  const bothXgOver = isXgOver(input.homeXgDiff) && isXgOver(input.awayXgDiff);
   693→  1538→  if (bothXgOver) {
   694→  1539→    score += 2;
   695→  1540→    checks.push({ check: 'Both teams xG Over (underperforming)', points: 2, passed: true });
   696→  1541→  } else {
   697→  1542→    checks.push({ check: 'Both teams xG Over (underperforming)', points: 2, passed: false });
   698→  1543→  }
   699→  1544→
   700→  1545→  // Check 3: BOTH teams Z-Score not overheated
   701→  1546→  const bothZOk = isZNeutralOrBetter(input.homeZScore) && isZNeutralOrBetter(input.awayZScore);
   702→  1547→  if (bothZOk) {
   703→  1548→    score += 1;
   704→  1549→    checks.push({ check: 'Both teams Z-Score Neutral/better', points: 1, passed: true });
   705→  1550→  } else {
   706→  1551→    checks.push({ check: 'Both teams Z-Score Neutral/better', points: 1, passed: false });
   707→  1552→  }
   708→  1553→
   709→  1554→  // Check 4: Competitive matchup (favorite odds 1.80–2.50)
   710→  1555→  if (input.favoriteOdds && input.favoriteOdds >= 1.80 && input.favoriteOdds <= 2.50) {
   711→  1556→    score += 1;
   712→  1557→    checks.push({ check: 'Competitive matchup (odds 1.80-2.50)', points: 1, passed: true });
   713→  1558→  } else {
   714→  1559→    checks.push({ check: 'Competitive matchup (odds 1.80-2.50)', points: 1, passed: false });
   715→  1560→  }
   716→  1561→
   717→  1562→  // Penalty 5: Massive favorite (≤1.50) — one side dominates, BTTS unlikely
   718→  1563→  if (input.favoriteOdds && input.favoriteOdds <= 1.50) {
   719→  1564→    score -= 2;
   720→  1565→    checks.push({ check: 'Massive favorite (odds ≤1.50)', points: -2, passed: true });
   721→  1566→  } else {
   722→  1567→    checks.push({ check: 'Massive favorite (odds ≤1.50)', points: -2, passed: false });
   723→  1568→  }
   724→  1569→
   725→  1570→  // Penalty 6: Either team can't finish (SOT conversion <25%)
   726→  1571→  const homeCantFinish = input.homeSotConversion !== null && input.homeSotConversion < 25;
   727→  1572→  const awayCantFinish = input.awaySotConversion !== null && input.awaySotConversion < 25;
   728→  1573→  if (homeCantFinish) score -= 1;
   729→  1574→  if (awayCantFinish) score -= 1;
   730→  1575→  checks.push({
   731→  1576→    check: `Home SOT conversion ${homeCantFinish ? '< 25%' : '≥ 25%'}${input.homeSotConversion !== null ? ` (${input.homeSotConversion.toFixed(0)}%)` : ' (N/A)'}`,
   732→  1577→    points: -1,
   733→  1578→    passed: !homeCantFinish,
   734→  1579→  });
   735→  1580→  checks.push({
   736→  1581→    check: `Away SOT conversion ${awayCantFinish ? '< 25%' : '≥ 25%'}${input.awaySotConversion !== null ? ` (${input.awaySotConversion.toFixed(0)}%)` : ' (N/A)'}`,
   737→  1582→    points: -1,
   738→  1583→    passed: !awayCantFinish,
   739→  1584→  });
   740→  1585→
   741→  1586→  // Penalty 7: Either team in a strong cold Z-Score streak
   742→  1587→  const homeCold = isZCold(input.homeZScore);
   743→  1588→  const awayCold = isZCold(input.awayZScore);
   744→  1589→  if (homeCold) score -= 2;
   745→  1590→  if (awayCold) score -= 2;
   746→  1591→  checks.push({
   747→  1592→    check: `Home Z-Score ${homeCold ? 'cold (≥1.0)' : 'ok'}`,
   748→  1593→    points: -2,
   749→  1594→    passed: !homeCold,

   844→  1689→    score += 1;
   845→  1690→    checks.push({ check: `Combined xG ${input.combinedXgTotal.toFixed(2)} in 2.5-3.0`, points: 1, passed: true });
   846→  1691→  } else {

   848→  1693→  }

   851→  1696→  const bothRegUnder = isRegressionUnder(input.homeRegressionSignal) && isRegressionUnder(input.awayRegressionSignal);
   852→  1697→  if (bothRegUnder) {
   853→  1698→    score += 2;
   854→  1699→    checks.push({ check: 'Both teams Regression Under', points: 2, passed: true });
   855→  1700→  } else {

   857→  1702→  }
   858→  1703→

   860→  1705→  const bothXgOver = isXgOver(input.homeXgSignal) && isXgOver(input.awayXgSignal);
   861→  1706→  if (bothXgOver) {
   862→  1707→    score += 2;

   883→  1728→  if (xgBalanced) {

   896→  1741→  }

   919→  1764→    points: -1,

   923→  1768→    check: `Away SOT conversion ${awayCantFinish ? '< 25%' : '≥ 25%'}${input.awaySotConversion !== null ? ` (${input.awaySotConversion.toFixed(0)}%)` : ' (N/A)'}`,

   928→  1773→  // Determine tier
   929→  1774→  let tier: GoalTier;
   930→  1775→  let recommendation: string;
   931→  1776→  if (score >= 6) {
   932→  1777→    tier = 'GOAL RICH';
   933→  1778→    recommendation = 'Full confidence on Over 2.5, consider Over 3.5';
   934→  1779→  } else if (score >= 4) {
   935→  1780→    tier = 'GOAL LIKELY';
   936→  1781→    recommendation = 'Solid Over 2.5 bet';
   937→  1782→  } else if (score >= 2) {
   938→  1783→    tier = 'GOAL BORDERLINE';
   939→  1784→    recommendation = 'Over 1.5 strong, Over 2.5 cautious';
   940→  1785→  } else if (score >= 0) {
   941→  1786→    tier = 'GOAL THIN';
   942→  1787→    recommendation = 'Over 1.5 only, skip Over 2.5';
   943→  1788→  } else {
   944→  1789→    tier = 'GOAL STALL';
   945→  1790→    recommendation = 'Likely 1-2 goals, avoid Over markets above 1.5';
   946→  1791→  }
   947→  1792→
   948→  1793→  return { tier, score, breakdown: checks, recommendation };

  1062→                        const tau = (x: number, y: number, lambda1: number, lambda2: number, rho: number): number => {
  1063→                          if (x === 0 && y === 0) return 1 - lambda1 * lambda2 * rho
  1064→                          if (x === 0 && y === 1) return 1 + lambda1 * rho

   101→// ---- Season Weighting ----
   102→const DECAY_FACTOR = 0.65;
   103→function calculateSeasonWeights(seasons: string[]): Map<string, number> {
   104→  const weights = new Map<string, number>();
   105→  if (seasons.length === 0) return weights;
   106→  const sorted = [...seasons].sort();
   107→  let weight = 1.0;
   108→  let totalWeight = 0;
   109→  for (const season of sorted) {
   110→    weights.set(season, weight);
   111→    totalWeight += weight;
   112→    weight *= DECAY_FACTOR;
   113→  }
   114→  for (const [season, w] of weights) weights.set(season, w / totalWeight);
   115→  return weights;
   116→}

   118→// ---- Team Stats ----
   119→function calculateBacktestTeamStats(
   120→  results: MatchResult[], team: string, seasonWeights?: Map<string, number>
   121→) {
   122→  let weightedResults = results;
   123→  if (seasonWeights && seasonWeights.size > 0) {
   124→    weightedResults = [];
   125→    for (const match of results) {
   126→      const weight = seasonWeights.get(match.season) ?? 1;
   127→      const copies = Math.max(1, Math.round(weight * 10));
   128→      for (let i = 0; i < copies; i++) weightedResults.push(match);
   129→    }
   130→  }
   131→
   132→  const homeGames = weightedResults.filter(m => m.homeTeam === team);
   133→  const awayGames = weightedResults.filter(m => m.awayTeam === team);
   134→  const homeScored = homeGames.reduce((sum, m) => sum + m.ftHomeGoals, 0);
   135→  const homeConceded = homeGames.reduce((sum, m) => sum + m.ftAwayGoals, 0);
   136→  const awayScored = awayGames.reduce((sum, m) => sum + m.ftAwayGoals, 0);
   137→  const awayConceded = awayGames.reduce((sum, m) => sum + m.ftHomeGoals, 0);
   138→  const homeWins = homeGames.filter(m => m.ftResult === 'H').length;
   139→  const awayWins = awayGames.filter(m => m.ftResult === 'A').length;
   140→  const draws = homeGames.filter(m => m.ftResult === 'D').length + awayGames.filter(m => m.ftResult === 'D').length;
   141→  const totalGames = homeGames.length + awayGames.length;
   142→
   143→  return {
   144→    homeGames: homeGames.length,
   145→    awayGames: awayGames.length,
   146→    totalGames,
   147→    avgHomeScored: homeGames.length > 0 ? homeScored / homeGames.length : 0,
   148→    avgHomeConceded: homeGames.length > 0 ? homeConceded / homeGames.length : 0,
   149→    avgAwayScored: awayGames.length > 0 ? awayScored / awayGames.length : 0,
   150→    avgAwayConceded: awayGames.length > 0 ? awayConceded / awayGames.length : 0,
   151→    form: totalGames > 0 ? (homeWins + awayWins + 0.5 * draws) / totalGames : 0.5,
   152→  };
   153→}

   161→function calculateLeagueAverages(results: MatchResult[]): LeagueAverages {
   162→  if (results.length === 0) return { avgHomeGoals: 1.5, avgAwayGoals: 1.2, avgTotalGoals: 2.7, homeWinRate: 0.45, drawRate: 0.25, awayWinRate: 0.30 };
   163→  const totalHomeGoals = results.reduce((sum, m) => sum + m.ftHomeGoals, 0);
   164→  const totalAwayGoals = results.reduce((sum, m) => sum + m.ftAwayGoals, 0);
   165→  return {
   166→    avgHomeGoals: totalHomeGoals / results.length,
   167→    avgAwayGoals: totalAwayGoals / results.length,
   168→    avgTotalGoals: (totalHomeGoals + totalAwayGoals) / results.length,
   169→    homeWinRate: results.filter(m => m.ftResult === 'H').length / results.length,
   170→    drawRate: results.filter(m => m.ftResult === 'D').length / results.length,
   171→    awayWinRate: results.filter(m => m.ftResult === 'A').length / results.length,
   172→  };
   173→}

   175→// ---- Generate Predictions ----
   176→function generatePrediction(
   177→  trainingData: MatchResult[], homeTeam: string, awayTeam: string,
   178→  leagueAvgs: LeagueAverages, seasonWeights?: Map<string, number>
   179→) {
   180→  const homeStats = calculateBacktestTeamStats(trainingData, homeTeam, seasonWeights);
   181→  const awayStats = calculateBacktestTeamStats(trainingData, awayTeam, seasonWeights);
   182→
   183→  const ha = calculateBidirectionalHomeAdvantage(
   184→    homeStats.homeGames > 0 ? homeStats.avgHomeScored : leagueAvgs.avgHomeGoals,
   185→    homeStats.homeGames > 0 ? homeStats.avgHomeConceded : leagueAvgs.avgAwayGoals,
   186→    homeStats.awayGames > 0 ? homeStats.avgAwayScored : leagueAvgs.avgAwayGoals,
   187→    homeStats.awayGames > 0 ? homeStats.avgAwayConceded : leagueAvgs.avgHomeGoals,
   188→    leagueAvgs.avgHomeGoals, leagueAvgs.avgAwayGoals
   189→  );
   190→
   191→  const homeAttackRatio = leagueAvgs.avgTotalGoals > 0
   192→    ? (homeStats.homeGames > 0 ? homeStats.avgHomeScored : leagueAvgs.avgHomeGoals) / (leagueAvgs.avgTotalGoals / 2) : 1;
   193→  const awayDefenseRatio = leagueAvgs.avgTotalGoals > 0
   194→    ? (awayStats.awayGames > 0 ? awayStats.avgAwayConceded : leagueAvgs.avgHomeGoals) / (leagueAvgs.avgTotalGoals / 2) : 1;
   195→  const awayAttackRatio = leagueAvgs.avgTotalGoals > 0
   196→    ? (awayStats.awayGames > 0 ? awayStats.avgAwayScored : leagueAvgs.avgAwayGoals) / (leagueAvgs.avgTotalGoals / 2) : 1;
   197→  const homeDefenseRatio = leagueAvgs.avgTotalGoals > 0
   198→    ? (homeStats.homeGames > 0 ? homeStats.avgHomeConceded : leagueAvgs.avgAwayGoals) / (leagueAvgs.avgTotalGoals / 2) : 1;
   199→
   200→  const homeXg = homeAttackRatio * awayDefenseRatio * leagueAvgs.avgHomeGoals * ha.scoringAdvantage;
   201→  const awayXg = awayAttackRatio * homeDefenseRatio * leagueAvgs.avgAwayGoals * ha.defensiveAdvantage;
   202→  const totalXg = homeXg + awayXg;
   203→
   204→  const allGoals = trainingData.map(m => m.ftHomeGoals + m.ftAwayGoals);
   205→  const dispersion = estimateDispersion(allGoals);
   206→  const useNB = isFinite(dispersion) && dispersion <= 100;
   207→
   208→  // 1X2 probs
   209→  let homeWinProb = 0, awayWinProb = 0, drawProbCalc = 0;
   210→  for (let i = 0; i <= 7; i++) {
   211→    for (let j = 0; j <= 7; j++) {
   212→      const p = goalProb(homeXg, i, dispersion) * goalProb(awayXg, j, dispersion);
   213→      if (i > j) homeWinProb += p;
   214→      else if (j > i) awayWinProb += p;
   215→      else drawProbCalc += p;
   216→    }
   217→  }
   218→
   219→  let drawProb = 1 - homeWinProb - awayWinProb;
   220→  const formAdjustment = (homeStats.form - awayStats.form) * 0.1;
   221→  const adjustedHomeWin = Math.max(0.1, Math.min(0.8, homeWinProb + formAdjustment));
   222→  const adjustedAwayWin = Math.max(0.1, Math.min(0.8, awayWinProb - formAdjustment));
   223→  drawProb = Math.max(0.15, Math.min(0.4, 1 - adjustedHomeWin - adjustedAwayWin));
   224→  const total = adjustedHomeWin + drawProb + adjustedAwayWin;
   225→  const finalHomeWin = adjustedHomeWin / total;
   226→  const finalDraw = drawProb / total;
   227→  const finalAwayWin = adjustedAwayWin / total;
   228→
   229→  // Goals markets
   230→  const p0 = goalProb(totalXg, 0, dispersion);
   231→  const p1 = goalProb(totalXg, 1, dispersion);
   232→  const p2 = goalProb(totalXg, 2, dispersion);
   233→  const over15Prob = 1 - p0 - p1;
   234→  const over25Prob = 1 - p0 - p1 - p2;
   235→
   236→  // BTTS
   237→  let bttsProbCalc = 0;
   238→  for (let i = 1; i <= 7; i++) {
   239→    for (let j = 1; j <= 7; j++) {
   240→      bttsProbCalc += goalProb(homeXg, i, dispersion) * goalProb(awayXg, j, dispersion);
   241→    }
   242→  }
   243→  let bttsProb = bttsProbCalc;
   244→  const balancedLambda = totalXg / 2;
   245→  const bttsBalanced = (1 - Math.exp(-balancedLambda)) * (1 - Math.exp(-balancedLambda));
   246→  const lambdaImbalance = Math.abs(homeXg - awayXg) / (homeXg + awayXg + 0.001);
   247→  const correctionFraction = 0.55 * lambdaImbalance;
   248→  bttsProb = bttsProb + (bttsBalanced - bttsProb) * correctionFraction;
   249→
   250→  // Most likely scoreline
   251→  let bestScore = '0-0', bestProb = 0;
   252→  for (let i = 0; i <= 5; i++) {
   253→    for (let j = 0; j <= 5; j++) {
   254→      const p = goalProb(homeXg, i, dispersion) * goalProb(awayXg, j, dispersion);
   255→      if (p > bestProb) { bestProb = p; bestScore = `${i}-${j}`; }
   256→    }
   257→  }
   258→
   259→  return {
   260→    homeWin: Math.round(finalHomeWin * 100),
   261→    draw: Math.round(finalDraw * 100),
   262→    awayWin: Math.round(finalAwayWin * 100),
   263→    over15: Math.round(Math.min(95, Math.max(40, over15Prob * 100))),
   264→    over25: Math.round(Math.min(85, Math.max(35, over25Prob * 100))),
   265→    btts: Math.round(bttsProb * 100),
   266→    totalXg: Math.round(totalXg * 100) / 100,
   267→    homeXg: Math.round(homeXg * 100) / 100,
   268→    awayXg: Math.round(awayXg * 100) / 100,
   269→    likelyScore: bestScore,
   270→    likelyScoreProb: Math.round(bestProb * 1000) / 10,
   271→    dispersion: useNB ? Math.round(dispersion * 100) / 100 : null,
   272→  };
   273→}

   278→function parseNumber(val: string | undefined): number | null {
   279→  if (!val || val.trim() === '') return null;
   280→  const n = parseFloat(val);
   281→  return isNaN(n) ? null : n;

   284→function parseCSV(csvText: string, season: string): MatchResult[] {
   285→  const lines = csvText.trim().split('\n');
   286→  if (lines.length < 2) return [];
   287→  const headerLine = lines[0].replace(/^\uFEFF/, '');
   288→  const header = headerLine.split(',');
   289→  const colIndex: Record<string, number> = {};
   290→  header.forEach((col, idx) => { colIndex[col.trim()] = idx; });
   291→
   292→  const results: MatchResult[] = [];
   293→  for (let i = 1; i < lines.length; i++) {
   294→    const line = lines[i];
   295→    if (!line.trim()) continue;
   296→    const values: string[] = [];
   297→    let currentValue = '';
   298→    let inQuotes = false;
   299→    for (const char of line) {
   300→      if (char === '"') inQuotes = !inQuotes;
   301→      else if (char === ',' && !inQuotes) { values.push(currentValue.trim()); currentValue = ''; }
   302→      else currentValue += char;
   303→    }
   304→    values.push(currentValue.trim());
   305→
   306→    const homeTeam = values[colIndex['HomeTeam']] || '';
   307→    const awayTeam = values[colIndex['AwayTeam']] || '';
   308→    if (!homeTeam || !awayTeam) continue;
   309→
   310→    results.push({
   311→      date: values[colIndex['Date']] || '',
   312→      time: values[colIndex['Time']] || '',
   313→      homeTeam, awayTeam,
   314→      ftHomeGoals: parseInt(values[colIndex['FTHG']] || '0', 10) || 0,
   315→      ftAwayGoals: parseInt(values[colIndex['FTAG']] || '0', 10) || 0,
   316→      ftResult: (values[colIndex['FTR']] || 'D') as 'H' | 'D' | 'A',
   317→      htHomeGoals: parseInt(values[colIndex['HTHG']] || '0', 10) || 0,
   318→      htAwayGoals: parseInt(values[colIndex['HTAG']] || '0', 10) || 0,
   319→      htResult: (values[colIndex['HTR']] || 'D') as 'H' | 'D' | 'A',
   320→      homeShots: parseInt(values[colIndex['HS']] || '0', 10) || 0,
   321→      awayShots: parseInt(values[colIndex['AS']] || '0', 10) || 0,
   322→      homeShotsOnTarget: parseInt(values[colIndex['HST']] || '0', 10) || 0,
   323→      awayShotsOnTarget: parseInt(values[colIndex['AST']] || '0', 10) || 0,
   324→      oddsAvgHome: parseNumber(values[colIndex['AvgH']]),
   325→      oddsAvgDraw: parseNumber(values[colIndex['AvgD']]),
   326→      oddsAvgAway: parseNumber(values[colIndex['AvgA']]),
   327→      oddsB365Home: parseNumber(values[colIndex['B365H']]),
   328→      oddsB365Draw: parseNumber(values[colIndex['B365D']]),
   329→      oddsB365Away: parseNumber(values[colIndex['B365A']]),
   330→      oddsOver25: parseNumber(values[colIndex['B365>2.5']]),
   331→      oddsUnder25: parseNumber(values[colIndex['B365<2.5']]),
   332→      oddsAvgOver25: parseNumber(values[colIndex['Avg>2.5']]),
   333→      season,
   334→    });
   335→  }
   336→  return results;
   337→}

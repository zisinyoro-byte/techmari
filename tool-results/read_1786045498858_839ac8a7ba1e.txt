    24→    24→/**
    25→    25→ * Compute league baselines from results + analytics data.
    26→    26→ * Call this once per league/season, pass the result into all filter functions.
    27→    27→ */
    28→    28→export function computeLeagueBaselines(
    29→    29→  results: { ftHomeGoals: number; ftAwayGoals: number }[],
    30→    30→  analytics: {
    31→    31→    avgGoalsPerGame: number;
    32→    32→    over25Percent: number;
    33→    33→    avgHomeGoals: number;
    34→    34→    avgAwayGoals: number;
    35→    35→    overallShotConversion: string | number;
    36→    36→  }
    37→    37→): LeagueBaselines {
    38→    38→  const total = results.length || 1;
    39→    39→  const bttsCount = results.filter(r => r.ftHomeGoals > 0 && r.ftAwayGoals > 0).length;
    40→    40→  const over35Count = results.filter(r => r.ftHomeGoals + r.ftAwayGoals > 3.5).length;
    41→    41→
    42→    42→  return {
    43→    43→    avgGoalsPerGame: analytics.avgGoalsPerGame,
    44→    44→    over25Rate: analytics.over25Percent,
    45→    45→    bttsRate: (bttsCount / total) * 100,
    46→    46→    over35Rate: (over35Count / total) * 100,
    47→    47→    avgHomeGoals: analytics.avgHomeGoals,
    48→    48→    avgAwayGoals: analytics.avgAwayGoals,
    49→    49→    shotConversion: typeof analytics.overallShotConversion === 'string'
    50→    50→      ? parseFloat(analytics.overallShotConversion)
    51→    51→      : analytics.overallShotConversion,
    52→    52→  };
    53→    53→}

   114→   114→// ============================================================================
   115→   115→// Backtest-derived threshold registry
   116→   116→// ============================================================================
   117→   117→// This is the central store. Thresholds are added here when backtests run.
   118→   118→// Leagues without entries fall back to the hybrid formula.
   119→   119→// Supports localStorage persistence for client-side survival across page reloads.
   120→   120→// ============================================================================
   121→   121→
   122→   122→const STORAGE_KEY = 'techmari_backtest_thresholds';
   123→   123→
   124→   124→const backtestRegistry = new Map<string, LeagueBacktestThresholds>();
   125→   125→
   126→   126→let loadedFromStorage = false;
   127→   127→
   128→   128→/**
   129→   129→ * Load thresholds from localStorage into the in-memory registry.
   130→   130→ * Safe to call multiple times — only loads once per session.
   131→   131→ */
   132→   132→function loadFromLocalStorage(): void {
   133→   133→  if (loadedFromStorage || typeof window === 'undefined') return;
   134→   134→  loadedFromStorage = true;
   135→   135→  try {
   136→   136→    const raw = localStorage.getItem(STORAGE_KEY);
   137→   137→    if (!raw) return;
   138→   138→    const data = JSON.parse(raw);
   139→   139→    if (data.version !== 1) return;
   140→   140→    const entries: Record<string, LeagueBacktestThresholds> = data.thresholds;
   141→   141→    for (const [key, thresholds] of Object.entries(entries)) {
   142→   142→      backtestRegistry.set(key, thresholds);
   143→   143→    }
   144→   144→    const count = Object.keys(entries).length;
   145→   145→    if (count > 0) {
   146→   146→      console.log(`[BettingFilters] Loaded ${count} persisted threshold(s) from localStorage`);
   147→   147→    }
   148→   148→  } catch {
   149→   149→    // localStorage not available or corrupt data — silent fail
   150→   150→  }
   151→   151→}
   152→   152→
   153→   153→/**
   154→   154→ * Save all in-memory thresholds to localStorage.
   155→   155→ */
   156→   156→function saveToLocalStorage(): void {
   157→   157→  if (typeof window === 'undefined') return;
   158→   158→  try {
   159→   159→    const data: Record<string, LeagueBacktestThresholds> = {};
   160→   160→    for (const [key, thresholds] of backtestRegistry.entries()) {
   161→   161→      data[key] = thresholds;
   162→   162→    }
   163→   163→    localStorage.setItem(STORAGE_KEY, JSON.stringify({
   164→   164→      version: 1,
   165→   165→      updatedAt: new Date().toISOString(),
   166→   166→      thresholds: data,
   167→   167→    }));
   168→   168→  } catch {
   169→   169→    // localStorage not available or quota exceeded — silent fail
   170→   170→  }
   171→   171→}
   172→   172→
   173→   173→/**
   174→   174→ * Register backtest-derived thresholds for a league.
   175→   175→ * Also persists to localStorage automatically.
   176→   176→ */
   177→   177→export function registerBacktestThresholds(thresholds: LeagueBacktestThresholds): void {
   178→   178→  // Ensure we've loaded from localStorage first
   179→   179→  loadFromLocalStorage();
   180→   180→  backtestRegistry.set(thresholds.leagueName.toLowerCase(), thresholds);
   181→   181→  saveToLocalStorage();
   182→   182→}

   184→   184→/**

   187→   187→export function getBacktestThresholds(leagueName: string): LeagueBacktestThresholds | undefined {
   188→   188→  loadFromLocalStorage();
   189→   189→  return backtestRegistry.get(leagueName.toLowerCase());
   190→   190→}

   195→   195→export function getAllBacktestThresholds(): LeagueBacktestThresholds[] {
   196→   196→  loadFromLocalStorage();

   203→   203→export function hasSufficientBacktest(leagueName: string, minSampleSize = 150): boolean {

   470→   470→  baselines: LeagueBaselines,

   472→   472→) {

   474→   474→  const useBacktest = bt !== undefined && bt.sampleSize >= minSampleSize;
   475→   475→
   476→   476→  // League-level thresholds (always absolute)
   477→   477→  const league: ResolvedLeagueThresholds = useBacktest
   478→   478→    ? {
   479→   479→        bttsLeagueAvgGoals: bt!.bttsLeagueAvgGoals,
   480→   480→        bttsLeagueO25Rate: bt!.bttsLeagueO25Rate,
   481→   481→        o35LeagueAvgGoals: bt!.o35LeagueAvgGoals,
   482→   482→        o35LeagueO25Rate: bt!.o35LeagueO25Rate,
   483→   483→      }
   484→   484→    : {
   485→   485→        bttsLeagueAvgGoals: BTTS_LEAGUE_THRESHOLDS.leagueAvgGoals,
   486→   486→        bttsLeagueO25Rate: BTTS_LEAGUE_THRESHOLDS.leagueO25Rate,
   487→   487→        o35LeagueAvgGoals: OVER35_LEAGUE_THRESHOLDS.leagueAvgGoals,
   488→   488→        o35LeagueO25Rate: OVER35_LEAGUE_THRESHOLDS.leagueO25Rate,
   489→   489→      };

   491→   491→  const src: 'backtest' | 'hybrid' = useBacktest ? 'backtest' : 'hybrid';
   492→   492→
   493→   493→  // BTTS match-level thresholds
   494→   494→  const btts: ResolvedThresholds = useBacktest
   495→   495→    ? { ...bt!.btts, source: 'backtest' }
   496→   496→    : {
   497→   497→        modelBttsProb: hybridThreshold(BTTS_HYBRID_THRESHOLDS.modelBttsProb.floor, baselines.bttsRate, BTTS_HYBRID_THRESHOLDS.modelBttsProb.multiplier),
   498→   498→        homeAvgGoals: hybridThreshold(BTTS_HYBRID_THRESHOLDS.homeAvgGoals.floor, baselines.avgHomeGoals, BTTS_HYBRID_THRESHOLDS.homeAvgGoals.multiplier),
   499→   499→        awayAvgGoals: hybridThreshold(BTTS_HYBRID_THRESHOLDS.awayAvgGoals.floor, baselines.avgAwayGoals, BTTS_HYBRID_THRESHOLDS.awayAvgGoals.multiplier),
   500→   500→        modelO25Prob: hybridThreshold(BTTS_HYBRID_THRESHOLDS.modelO25Prob.floor, baselines.over25Rate, BTTS_HYBRID_THRESHOLDS.modelO25Prob.multiplier),
   501→   501→        shotConversion: hybridThreshold(BTTS_HYBRID_THRESHOLDS.shotConversion.floor, baselines.shotConversion, BTTS_HYBRID_THRESHOLDS.shotConversion.multiplier),
   502→   502→        source: 'hybrid',
   503→   503→      };
   504→   504→
   505→   505→  // Over 3.5 match-level thresholds
   506→   506→  const over35: ResolvedOver35Thresholds = useBacktest
   507→   507→    ? { ...bt!.over35, source: 'backtest' }
   508→   508→    : {
   509→   509→        modelO35Prob: hybridThreshold(OVER35_HYBRID_THRESHOLDS.modelO35Prob.floor, baselines.over35Rate, OVER35_HYBRID_THRESHOLDS.modelO35Prob.multiplier),
   510→   510→        bttsProb: hybridThreshold(OVER35_HYBRID_THRESHOLDS.bttsProb.floor, baselines.bttsRate, OVER35_HYBRID_THRESHOLDS.bttsProb.multiplier),
   511→   511→        homeAvgGoals: hybridThreshold(OVER35_HYBRID_THRESHOLDS.homeAvgGoals.floor, baselines.avgHomeGoals, OVER35_HYBRID_THRESHOLDS.homeAvgGoals.multiplier),
   512→   512→        awayAvgGoals: hybridThreshold(OVER35_HYBRID_THRESHOLDS.awayAvgGoals.floor, baselines.avgAwayGoals, OVER35_HYBRID_THRESHOLDS.awayAvgGoals.multiplier),
   513→   513→        shotConversion: hybridThreshold(OVER35_HYBRID_THRESHOLDS.shotConversion.floor, baselines.shotConversion, OVER35_HYBRID_THRESHOLDS.shotConversion.multiplier),
   514→   514→        source: 'hybrid',
   515→   515→      };
   516→   516→
   517→   517→  // STRONG BET thresholds
   518→   518→  const strongBet: ResolvedStrongBetThresholds = useBacktest
   519→   519→    ? { ...bt!.strongBet, source: 'backtest' }

   524→   524→        bttsChecklistCount: STRONG_BET_HYBRID.bttsChecklistCount,
   525→   525→        source: 'hybrid',

   529→   529→  const greyResult: ResolvedGreyResultThresholds = useBacktest
   530→   530→    ? {
   531→   531→        ...bt!.greyResult,
   532→   532→        bttsProbMax: (bt!.greyResult as any).bttsProbMax ?? GREY_RESULT_CONFIG.bttsProb.max,
   533→   533→        source: 'backtest' as const,
   534→   534→      }
   535→   535→    : {
   536→   536→        bttsProb: GREY_RESULT_CONFIG.bttsProb.min,
   537→   537→        bttsProbMax: GREY_RESULT_CONFIG.bttsProb.max,
   538→   538→        o25Prob: hybridThreshold(GREY_RESULT_CONFIG.o25Prob.floor, baselines.over25Rate, GREY_RESULT_CONFIG.o25Prob.multiplier),
   539→   539→        o35Prob: hybridThreshold(GREY_RESULT_CONFIG.o35Prob.floor, baselines.over35Rate, GREY_RESULT_CONFIG.o35Prob.multiplier),
   540→   540→        bttsChecklistCount: GREY_RESULT_CONFIG.bttsChecklistCount,
   541→   541→        over35ChecklistCount: GREY_RESULT_CONFIG.over35ChecklistCount,
   542→   542→        requiredChecks: GREY_RESULT_CONFIG.requiredChecks,
   543→   543→        source: 'hybrid',
   544→   544→      };
   545→   545→
   546→   546→  // GOAL FEST thresholds (v2 — includes O3.5 and requiredChecks)
   547→   547→  const goalFest: ResolvedGoalFestThresholds = useBacktest
   548→   548→    ? {
   549→   549→        o25Prob: bt!.strongBet.o25Prob,  // reuse STRONG BET O2.5 threshold
   550→   550→        bttsProb: bt!.strongBet.bttsProb, // reuse STRONG BET BTTS threshold
   551→   551→        o35Prob: bt!.over35.modelO35Prob, // reuse Over 3.5 O3.5 threshold
   552→   552→        requiredChecks: GOAL_FEST_CONFIG.requiredChecks,
   553→   553→        source: 'backtest',
   554→   554→      }
   555→   555→    : {
   556→   556→        o25Prob: hybridThreshold(GOAL_FEST_CONFIG.o25Prob.floor, baselines.over25Rate, GOAL_FEST_CONFIG.o25Prob.multiplier),
   557→   557→        bttsProb: hybridThreshold(GOAL_FEST_CONFIG.bttsProb.floor, baselines.bttsRate, GOAL_FEST_CONFIG.bttsProb.multiplier),
   558→   558→        o35Prob: hybridThreshold(GOAL_FEST_CONFIG.o35Prob.floor, baselines.over35Rate, GOAL_FEST_CONFIG.o35Prob.multiplier),
   559→   559→        requiredChecks: GOAL_FEST_CONFIG.requiredChecks,
   560→   560→        source: 'hybrid',
   561→   561→      };
   562→   562→
   563→   563→  return { league, btts, over35, strongBet, greyResult, goalFest, source: src };
   564→   564→}

   597→   597→/**
   598→   598→ * Compute BTTS checklist score (0-9) using resolved thresholds.
   599→   599→ * Expanded from 7 to 9 checks: added O2.5 implied probability + rolling combined scoring.
   600→   600→ * These two features were the strongest predictors in 10K-match analysis.
   601→   601→ */
   602→   602→export function computeBttsChecklist(
   603→   603→  input: ChecklistInput,
   604→   604→  resolved: ReturnType<typeof resolveAllThresholds>
   605→   605→): number {
   606→   606→  const lt = resolved.league;
   607→   607→  const rt = resolved.btts;
   608→   608→  let count = 0;
   609→   609→  if (input.avgGoalsPerGame >= lt.bttsLeagueAvgGoals) count++;
   610→   610→  if (input.over25Percent >= lt.bttsLeagueO25Rate) count++;
   611→   611→  if (input.bttsProb >= rt.modelBttsProb) count++;
   612→   612→  if (input.avgHomeGoals >= rt.homeAvgGoals) count++;
   613→   613→  if (input.avgAwayGoals >= rt.awayAvgGoals) count++;
   614→   614→  if (input.o25Prob >= rt.modelO25Prob) count++;
   615→   615→  if (input.overallShotConversion >= rt.shotConversion) count++;
   616→   616→  // NEW: O2.5 implied probability (strongest single predictor — corr 0.192)
   617→   617→  if (input.o25ImpliedProb !== null && input.o25ImpliedProb >= O25_IMPLIED_THRESHOLDS.btts) count++;
   618→   618→  // NEW: Rolling combined scoring (adds ~20pp discrimination over season averages)
   619→   619→  if (input.rollingCombinedScoring >= ROLLING_SCORING_THRESHOLDS.btts) count++;
   620→   620→  return count;
   621→   621→}

   692→   692→/**
   693→   693→ * Compute STRONG BET using points-based system with resolved thresholds.
   694→   694→ * Signal checks updated based on 7,110-match backtest analysis:
   695→   695→ *   xG = Over (mild underperformance, finishing due to click)
   696→   696→ *   Regression = Under (recently hot, goals keep flowing)
   697→   697→ *   Z-Score = Neutral (no anomaly, natural attacking football)
   698→   698→ *   Signal Divergence (xG + Regression point opposite = high-variance games)
   699→   699→ * Returns { isStrongBet, points, maxPoints, breakdown }
   700→   700→ */
   701→   701→export function computeStrongBet(
   702→   702→  checklistInput: ChecklistInput,
   703→   703→  signals: SignalInput,
   704→   704→  resolved: ReturnType<typeof resolveAllThresholds>,
   705→   705→  options?: { momentumSignal?: string; leagueBttsRate?: number }
   706→   706→): {
   707→   707→  isStrongBet: boolean;
   708→   708→  points: number;
   709→   709→  maxPoints: number;
   710→   710→  breakdown: { check: string; points: number; passed: boolean }[];
   711→   711→} {
   712→   712→  const st = resolved.strongBet;
   713→   713→  const p = STRONG_BET_POINTS;
   714→   714→
   715→   715→  const bttsCount = computeBttsChecklist(checklistInput, resolved);
   716→   716→
   717→   717→  // Signal helpers — backtest-optimized conditions
   718→   718→  const isXgMild = (sig: string) => sig === 'Over' || sig === 'Under';
   719→   719→  const isRegressionUnder = (sig: string) => sig === 'Under' || sig === 'Strong Under';
   720→   720→  const isZScoreNeutral = (sig: string) => sig === 'Neutral';
   721→   721→  // Divergence: xG points one way (Over=goals due), Regression points opposite (Under=hot)
   722→   722→  const hasSignalDivergence = (xg: string, reg: string) =>
   723→   723→    (xg === 'Over' || xg === 'Strong Over') && isRegressionUnder(reg);
   724→   724→
   725→   725→  // League-adaptive BTTS weights: reduce in BTTS-poor leagues, boost Over weights
   726→   726→  const bttsRate = options?.leagueBttsRate;
   727→   727→  const adaptiveBttsProb = bttsRate !== undefined && bttsRate < 55 ? 0 : p.btts;
   728→   728→  const adaptiveBttsChecklist = bttsRate !== undefined && bttsRate < 45 ? 1 : p.bttsChecklist;
   729→   729→  const freedPts = (p.btts - adaptiveBttsProb) + (p.bttsChecklist - adaptiveBttsChecklist);
   730→   730→  const adaptiveO25 = p.o25 + (freedPts > 1 ? 1 : 0);
   731→   731→  const adaptiveO35 = p.o35 + (freedPts > 2 ? 1 : 0);
   732→   732→
   733→   733→  // Momentum alternative: Regression Under OR Momentum Over qualifies
   734→   734→  const isMomentumOver = options?.momentumSignal === 'MOMENTUM OVER' || options?.momentumSignal === 'MOMENTUM LEAN OVER';
   735→   735→  const regressionOrMomentum = isRegressionUnder(signals.regressionSignal) || isMomentumOver;
   736→   736→
   737→   737→  const checks = [
   738→   738→    { check: `O2.5 >=${st.o25Prob.toFixed(0)}%`, points: adaptiveO25, passed: checklistInput.o25Prob >= st.o25Prob },
   739→   739→    { check: `O3.5 >=${st.o35Prob.toFixed(0)}%`, points: adaptiveO35, passed: checklistInput.o35Prob >= st.o35Prob },
   740→   740→    { check: `BTTS >=${st.bttsProb.toFixed(0)}%`, points: adaptiveBttsProb, passed: adaptiveBttsProb > 0 && checklistInput.bttsProb >= st.bttsProb },
   741→   741→    { check: `BTTS Checklist >=${st.bttsChecklistCount}/9`, points: adaptiveBttsChecklist, passed: bttsCount >= st.bttsChecklistCount },
   742→   742→    { check: 'xG = Over/Under', points: p.xgSignal, passed: isXgMild(signals.xgSignal) },
   743→   743→    { check: isMomentumOver ? 'Momentum Over (alt)' : 'Regression = Under', points: p.regressionSignal, passed: regressionOrMomentum },
   744→   744→    { check: 'Z-Score = Neutral', points: p.zScoreSignal, passed: isZScoreNeutral(signals.zScoreSignal) },
   745→   745→    { check: 'Signal Divergence', points: p.signalDivergence, passed: hasSignalDivergence(signals.xgSignal, signals.regressionSignal) },
   746→   746→    // NEW: O2.5 implied probability — strongest single predictor (corr 0.192)
   747→   747→    { check: `O2.5 Implied >=${O25_IMPLIED_THRESHOLDS.strongBet}%`, points: 1, passed: checklistInput.o25ImpliedProb !== null && checklistInput.o25ImpliedProb >= O25_IMPLIED_THRESHOLDS.strongBet },
   748→   748→    // NEW: Rolling combined scoring — adds ~20pp discrimination
   749→   749→    { check: `Rolling Scoring >=${ROLLING_SCORING_THRESHOLDS.strongBet}`, points: 1, passed: checklistInput.rollingCombinedScoring >= ROLLING_SCORING_THRESHOLDS.strongBet },
   750→   750→  ];
   751→   751→
   752→   752→  const totalPoints = checks.reduce((sum, c) => sum + (c.passed ? c.points : 0), 0);
   753→   753→  const maxPoints = checks.reduce((sum, c) => sum + c.points, 0);
   754→   754→
   755→   755→  return {
   756→   756→    isStrongBet: totalPoints >= p.threshold,
   757→   757→    points: totalPoints,
   758→   758→    maxPoints,
   759→   759→    breakdown: checks,
   760→   760→  };
   761→   761→}
   762→   762→
   763→   763→/**
   764→   764→ * Compute GREY RESULT using resolved thresholds.
   765→   765→ * Signal checks updated based on 7,110-match backtest analysis:
   766→   766→ *   Regression = Neutral or Under (normal variance or recently hot)
   767→   767→ *   Z-Score = Neutral (universal goal-fest sweet spot, Strong Over nearly impossible)
   768→   768→ *   xG = Over or Under (both mild states beat extremes)
   769→   769→ * Returns { isGreyResult, score, totalChecks, breakdown }
   770→   770→ */
   771→   771→export function computeGreyResult(
   772→   772→  checklistInput: ChecklistInput,
   773→   773→  signals: SignalInput,
   774→   774→  resolved: ReturnType<typeof resolveAllThresholds>
   775→   775→): {
   776→   776→  isGreyResult: boolean;
   777→   777→  score: number;
   778→   778→  totalChecks: number;
   779→   779→  breakdown: { check: string; passed: boolean }[];
   780→   780→} {
   781→   781→  const gt = resolved.greyResult;
   782→   782→  const bttsCount = computeBttsChecklist(checklistInput, resolved);
   783→   783→  const o35Count = computeOver35Checklist(checklistInput, resolved);
   784→   784→
   785→   785→  // Backtest-optimized signal helpers
   786→   786→  // Regression: Neutral or Under = normal variance or recently hot (75% of matches)
   787→   787→  // Z-Score: Neutral or Under = team near/below their own average (not anomalously high)
   788→   788→  // xG: Over or Under = mild deviation from xG (excludes extremes)
   789→   789→  const isRegNeutralOrUnder = (sig: string) => sig === 'Neutral' || sig === 'Under' || sig === 'Strong Under';

   818→   818→/**
   819→   819→ * Compute GOAL FEST indicator v2 — backtest-optimized combo detector.
   820→   820→ * Updated from 7,230 + 7,110-match analysis across 4 leagues x 5 seasons.
   821→   821→ *
   822→   822→ * Captures two distinct goal-fest patterns:
   823→   823→ *   Pattern A (Divergence): xG=Over + Reg=Under/StrongUnder + ZS=Neutral
   824→   824→ *     → Teams underperforming xG but on hot streaks = finishing is about to explode
   825→   825→ *   Pattern B (Overperformance): xG=Under/StrongUnder + Reg=Neutral + ZS=Neutral
   826→   826→ *     → Teams scoring above xG, normal regression, no anomaly = natural high-scoring
   827→   827→ *
   828→   828→ * Both patterns share Z-Score = Neutral as the common thread.
   829→   829→ *
   830→   830→ * 6 checks, need 5 of 6 to qualify:
   831→   831→ *   1. xG = Over/Under/Strong Under (expanded — Strong Under = clinical finishing goldmine)
   832→   832→ *   2. Signal Divergence OR Regression = Under/StrongUnder (broadened combo check)
   833→   833→ *   3. Z-Score = Neutral (the universal goal-fest sweet spot)
   834→   834→ *   4. O2.5 >= threshold (model agrees on goals)
   835→   835→ *   5. BTTS >= threshold (model agrees both teams score)
   836→   836→ *   6. NEW: O3.5 >= threshold (more direct predictor for 4+ goals)
   837→   837→ *
   838→   838→ * Expected: ~300-400 matches per 7,110, ~33-35% O3.5 hit rate
   839→   839→ * Returns { isGoalFest, score, totalChecks, breakdown }
   840→   840→ */
   841→   841→export function computeGoalFest(
   842→   842→  checklistInput: ChecklistInput,
   843→   843→  signals: SignalInput,
   844→   844→  resolved: ReturnType<typeof resolveAllThresholds>
   845→   845→): {
   846→   846→  isGoalFest: boolean;
   847→   847→  score: number;
   848→   848→  totalChecks: number;
   849→   849→  breakdown: { check: string; passed: boolean }[];
   850→   850→} {
   851→   851→  const gf = resolved.goalFest;
   852→   852→
   853→   853→  // Signal conditions (v2)
   854→   854→  // Check 1: xG mild — Over, Under, or Strong Under
   855→   855→  //   Strong Under = teams massively outperforming xG = clinical finishing goldmine
   856→   856→  const isXgMild = (sig: string) => sig === 'Over' || sig === 'Under' || sig === 'Strong Under';
   857→   857→
   858→   858→  // Check 2: Signal Divergence (xG and Regression point opposite directions)
   859→   859→  //   This is the KEY selectivity filter: xG says goals due (Over/Under) but
   860→   860→  //   regression says they're already flowing (Under = hot streak).
   861→   861→  //   The divergence = high-variance game where finishing is about to explode.
   862→   862→  //   NOTE: Do NOT add `|| isRegressionBearish(reg)` — that makes the check
   863→   863→  //   trivially redundant and lets in non-divergent matches, hurting O3.5 hit rate.
   864→   864→  const isRegressionBearish = (sig: string) => sig === 'Under' || sig === 'Strong Under';
   865→   865→  const hasSignalDivergence = (xg: string, reg: string) =>
   866→   866→    (xg === 'Over' || xg === 'Under' || xg === 'Strong Under') && isRegressionBearish(reg);
   867→   867→
   868→   868→  const isZScoreNeutral = (sig: string) => sig === 'Neutral';
   869→   869→
   870→   870→  const checks = [
   871→   871→    { check: 'xG = Over/Under/StrongUnder', passed: isXgMild(signals.xgSignal) },
   872→   872→    { check: 'Signal Divergence (xG vs Reg)', passed: hasSignalDivergence(signals.xgSignal, signals.regressionSignal) },
   873→   873→    { check: 'Z-Score = Neutral', passed: isZScoreNeutral(signals.zScoreSignal) },
   874→   874→    { check: `O2.5 >=${gf.o25Prob.toFixed(0)}%`, passed: checklistInput.o25Prob >= gf.o25Prob },
   875→   875→    { check: `BTTS >=${gf.bttsProb.toFixed(0)}%`, passed: checklistInput.bttsProb >= gf.bttsProb },
   876→   876→    { check: `O3.5 >=${gf.o35Prob.toFixed(0)}%`, passed: checklistInput.o35Prob >= gf.o35Prob },
   877→   877→    // NEW: O2.5 implied probability confirms market expects goals
   878→   878→    { check: `O2.5 Implied >=${O25_IMPLIED_THRESHOLDS.goalFest}%`, passed: checklistInput.o25ImpliedProb !== null && checklistInput.o25ImpliedProb >= O25_IMPLIED_THRESHOLDS.goalFest },
   879→   879→    // NEW: Rolling combined scoring confirms recent hot form
   880→   880→    { check: `Rolling Scoring >=${ROLLING_SCORING_THRESHOLDS.goalFest}`, passed: checklistInput.rollingCombinedScoring >= ROLLING_SCORING_THRESHOLDS.goalFest },
   881→   881→  ];
   882→   882→
   883→   883→  const score = checks.filter(c => c.passed).length;
   884→   884→
   885→   885→  // Need 6 of 8 checks to qualify (was 5 of 6 — expanded with 2 new checks)
   886→   886→  return {
   887→   887→    isGoalFest: score >= 6,
   888→   888→    score,
   889→   889→    totalChecks: checks.length,
   890→   890→    breakdown: checks,
   891→   891→  };
   892→   892→}

   956→   956→/**
   957→   957→ * Derive optimal thresholds from backtest data using accuracy optimization.
   958→   958→ *
   959→   959→ * For each criterion, sweeps candidate threshold values and finds the one that
   960→   960→ * maximizes the correct prediction rate while meeting the minimum accuracy target.
   961→   961→ *
   962→   962→ * @param leagueName - League identifier for the registry
   963→   963→ * @param matches - Historical match data with predictions and outcomes
   964→   964→ * @param currentBaselines - Current league baselines (for fallback floors)
   965→   965→ * @param options - Tuning parameters
   966→   966→ * @returns Derived thresholds, or null if insufficient data
   967→   967→ */
   968→   968→export function deriveThresholdsFromBacktest(
   969→   969→  leagueName: string,
   970→   970→  matches: BacktestMatch[],
   971→   971→  currentBaselines: LeagueBaselines,
   972→   972→  options: BacktestDerivationOptions = {}
   973→   973→): LeagueBacktestThresholds | null {
   974→   974→  const {
   975→   975→    minSampleSize = 150,
   976→   976→    minAccuracy = 0.60,
   977→   977→    minGoalAccuracy = 0.55,
   978→   978→  } = options;
   979→   979→
   980→   980→  if (matches.length < minSampleSize) return null;
   981→   981→
   982→   982→  // --- Helper: compute BTTS hit rate at various thresholds ---
   983→   983→  const findOptimalProbThreshold = (
   984→   984→    getPredicted: (m: BacktestMatch) => number | undefined,
   985→   985→    getActual: (m: BacktestMatch) => boolean,
   986→   986→    searchRange: [number, number],
   987→   987→    step = 1,
   988→   988→    floor?: number
   989→   989→  ): number => {
   990→   990→    const [lo, hi] = searchRange;
   991→   991→    let bestThreshold = floor ?? lo;
   992→   992→    let bestAccuracy = 0;
   993→   993→    let bestSampleAbove = 0;
   994→   994→
   995→   995→    for (let t = lo; t <= hi; t += step) {
   996→   996→      const aboveThreshold = matches.filter(m => {
   997→   997→        const pred = getPredicted(m);
   998→   998→        return pred !== undefined && pred >= t;
   999→   999→      });
  1000→  1000→      if (aboveThreshold.length < 30) continue; // need meaningful sample above threshold
  1001→  1001→
  1002→  1002→      const correct = aboveThreshold.filter(m => getActual(m)).length;
  1003→  1003→      const accuracy = correct / aboveThreshold.length;
  1004→  1004→
  1005→  1005→      // Prefer: highest accuracy that meets minimum, with most samples
  1006→  1006→      if (accuracy >= minAccuracy && (accuracy > bestAccuracy || (accuracy === bestAccuracy && aboveThreshold.length > bestSampleAbove))) {
  1007→  1007→        bestThreshold = t;
  1008→  1008→        bestAccuracy = accuracy;
  1009→  1009→        bestSampleAbove = aboveThreshold.length;
  1010→  1010→      }
  1011→  1011→    }
  1012→  1012→
  1013→  1013→    // If no threshold met min accuracy, use the one with best accuracy above floor
  1014→  1014→    if (bestAccuracy < minAccuracy && floor !== undefined) {
  1015→  1015→      return floor;
  1016→  1016→    }
  1017→  1017→    return bestThreshold;
  1018→  1018→  };
  1019→  1019→
  1020→  1020→  const findOptimalGoalThreshold = (
  1021→  1021→    getPredicted: (m: BacktestMatch) => number | undefined,
  1022→  1022→    getActual: (m: BacktestMatch) => boolean,
  1023→  1023→    searchRange: [number, number],
  1024→  1024→    step = 0.05,

  1046→  1046→      }
  1047→  1047→    }
  1048→  1048→
  1049→  1049→    if (bestAccuracy < minGoalAccuracy && floor !== undefined) {

  1051→  1051→    }

  1233→  1233→/**
  1234→  1234→ * Derive thresholds from backtest data using a simpler approach when model
  1235→  1235→ * predictions are not available — uses league averages as the predictor.

  1237→  1237→export function deriveSimpleThresholdsFromBacktest(
  1238→  1238→  leagueName: string,
  1239→  1239→  matches: { ftHomeGoals: number; ftAwayGoals: number }[],
  1240→  1240→  currentBaselines: LeagueBaselines,
  1241→  1241→  options: BacktestDerivationOptions = {}
  1242→  1242→): LeagueBacktestThresholds | null {
  1243→  1243→  const { minSampleSize = 150 } = options;
  1244→  1244→
  1245→  1245→  if (matches.length < minSampleSize) return null;
  1246→  1246→
  1247→  1247→  // Simple derivation: find the league-average-based thresholds that best
  1248→  1248→  // separate BTTS from non-BTTS and O3.5 from non-O3.5 matches.
  1249→  1249→  // This uses the Dixon-Coles probability approach: given league averages,
  1250→  1250→  // the Poisson BTTS probability is already a strong predictor.
  1251→  1251→
  1252→  1252→  const bttsMatches = matches.filter(m => m.ftHomeGoals > 0 && m.ftAwayGoals > 0);
  1253→  1253→  const over35Matches = matches.filter(m => m.ftHomeGoals + m.ftAwayGoals > 3.5);
  1254→  1254→

  1256→  1256→  // A match is "above average" if its BTTS prob > league BTTS rate
  1257→  1257→  // Find the multiplier that gives best accuracy
  1258→  1258→  let bestBttsMult = BTTS_HYBRID_THRESHOLDS.modelBttsProb.multiplier;
  1259→  1259→  let bestBttsAcc = 0;
  1260→  1260→  for (let mult = 1.0; mult <= 1.3; mult += 0.01) {
  1261→  1261→    const threshold = Math.max(BTTS_HYBRID_THRESHOLDS.modelBttsProb.floor, currentBaselines.bttsRate * mult);
  1262→  1262→    const aboveCount = bttsMatches.filter(() => currentBaselines.bttsRate >= threshold).length;
  1263→  1263→    // All BTTS matches have BTTS prob ~= league rate, so check hit rate
  1264→  1264→    // For simple derivation, use the rate-based threshold directly
  1265→  1265→    const estimatedAccuracy = bttsMatches.length > 0
  1266→  1266→      ? bttsMatches.filter(() => currentBaselines.bttsRate >= threshold).length / Math.max(matches.length, 1)
  1267→  1267→      : 0;
  1268→  1268→    if (estimatedAccuracy > bestBttsAcc) {
  1269→  1269→      bestBttsAcc = estimatedAccuracy;
  1270→  1270→      bestBttsMult = mult;
  1271→  1271→    }
  1272→  1272→  }
  1273→  1273→
  1274→  1274→  // For simple derivation without per-match predictions, use the hybrid formula
  1275→  1275→  // with the derived multiplier, falling back to defaults.
  1276→  1276→  const bttsFloor = BTTS_HYBRID_THRESHOLDS.modelBttsProb.floor;
  1277→  1277→  const bttsRate = currentBaselines.bttsRate;

  1287→  1287→      homeAvgGoals: Math.max(BTTS_HYBRID_THRESHOLDS.homeAvgGoals.floor, currentBaselines.avgHomeGoals * BTTS_HYBRID_THRESHOLDS.homeAvgGoals.multiplier),
  1288→  1288→      awayAvgGoals: Math.max(BTTS_HYBRID_THRESHOLDS.awayAvgGoals.floor, currentBaselines.avgAwayGoals * BTTS_HYBRID_THRESHOLDS.awayAvgGoals.multiplier),
  1289→  1289→      modelO25Prob: Math.max(BTTS_HYBRID_THRESHOLDS.modelO25Prob.floor, currentBaselines.over25Rate * BTTS_HYBRID_THRESHOLDS.modelO25Prob.multiplier),

  1292→  1292→    over35: {

  1294→  1294→      bttsProb: Math.max(OVER35_HYBRID_THRESHOLDS.bttsProb.floor, currentBaselines.bttsRate * OVER35_HYBRID_THRESHOLDS.bttsProb.multiplier),
  1295→  1295→      homeAvgGoals: Math.max(OVER35_HYBRID_THRESHOLDS.homeAvgGoals.floor, currentBaselines.avgHomeGoals * OVER35_HYBRID_THRESHOLDS.homeAvgGoals.multiplier),
  1296→  1296→      awayAvgGoals: Math.max(OVER35_HYBRID_THRESHOLDS.awayAvgGoals.floor, currentBaselines.avgAwayGoals * OVER35_HYBRID_THRESHOLDS.awayAvgGoals.multiplier),
  1297→  1297→      shotConversion: Math.max(OVER35_HYBRID_THRESHOLDS.shotConversion.floor, currentBaselines.shotConversion * OVER35_HYBRID_THRESHOLDS.shotConversion.multiplier),
  1298→  1298→    },
  1299→  1299→    strongBet: {

  1302→  1302→      bttsProb: Math.max(STRONG_BET_HYBRID.bttsProb.floor, currentBaselines.bttsRate * STRONG_BET_HYBRID.bttsProb.multiplier),

  1306→  1306→      bttsProb: GREY_RESULT_CONFIG.bttsProb.min,
  1307→  1307→      bttsProbMax: GREY_RESULT_CONFIG.bttsProb.max,
  1308→  1308→      o25Prob: Math.max(GREY_RESULT_CONFIG.o25Prob.floor, currentBaselines.over25Rate * GREY_RESULT_CONFIG.o25Prob.multiplier),
  1309→  1309→      o35Prob: Math.max(GREY_RESULT_CONFIG.o35Prob.floor, currentBaselines.over35Rate * GREY_RESULT_CONFIG.o35Prob.multiplier),
  1310→  1310→      bttsChecklistCount: GREY_RESULT_CONFIG.bttsChecklistCount,

  1319→  1319→// ============================================================================
  1320→  1320→// Bookie Odds Dampener — Heavy Favorite Adjustment
  1321→  1321→// ============================================================================
  1322→  1322→// Based on live results analysis: when bookie odds for one side are <=1.70,
  1323→  1323→// the model systematically overestimates O2.5 probability by 10-15%.
  1324→  1324→// Heavy favorites tend to win tight, controlling games (1-0, 2-0) rather
  1325→  1325→// than high-scoring shootouts. This dampener accounts for that pattern.
  1326→  1326→// ============================================================================
  1327→  1327→
  1328→  1328→/** Heavy favorite odds threshold (decimal odds) */
  1329→  1329→const HEAVY_FAVORITE_ODDS = 1.70;
  1330→  1330→
  1331→  1331→/** Maximum dampener amount (percentage points to subtract) */
  1332→  1332→const HEAVY_FAVORITE_DAMPENER = 10;
  1333→  1333→
  1334→  1334→/**
  1335→  1335→ * Apply bookie odds dampener to goal market probabilities.
  1336→  1336→ * When either team is a heavy favorite (avg odds <=1.70), reduce O2.5/O3.5/BTTS
  1337→  1337→ * probabilities to account for the "park the bus" effect.
  1338→  1338→ *
  1339→  1339→ * Only applies when valid odds are available (non-null, positive).
  1340→  1340→ *
  1341→  1341→ * @param over25Prob  - Model O2.5 probability (%)
  1342→  1342→ * @param over35Prob  - Model O3.5 probability (%)
  1343→  1343→ * @param bttsProb    - Model BTTS probability (%)
  1344→  1344→ * @param homeAvgOdds - Home team average odds (decimal)
  1345→  1345→ * @param awayAvgOdds - Away team average odds (decimal)
  1346→  1346→ * @returns Adjusted probabilities with dampener applied
  1347→  1347→ */
  1348→  1348→export function applyBookieOddsDampener(
  1349→  1349→  over25Prob: number,
  1350→  1350→  over35Prob: number,
  1351→  1351→  bttsProb: number,
  1352→  1352→  homeAvgOdds: number | null | undefined,
  1353→  1353→  awayAvgOdds: number | null | undefined,
  1354→  1354→  leagueAvgGoalsPerGame?: number
  1355→  1355→): { over25: number; over35: number; btts: number; dampened: boolean; reason: string } {
  1356→  1356→  if (!homeAvgOdds || !awayAvgOdds || homeAvgOdds <= 0 || awayAvgOdds <= 0) {
  1357→  1357→    return { over25: over25Prob, over35: over35Prob, btts: bttsProb, dampened: false, reason: '' };
  1358→  1358→  }
  1359→  1359→
  1360→  1360→  // League-adaptive: gentler dampener in high-variance leagues
  1361→  1361→  const variance = leagueAvgGoalsPerGame ? classifyLeagueVariance(leagueAvgGoalsPerGame) : 'low';
  1362→  1362→  const heavyThreshold = variance === 'high' ? 1.40 : variance === 'medium' ? 1.60 : HEAVY_FAVORITE_ODDS;
  1363→  1363→  const maxDampener = variance === 'high' ? 5 : variance === 'medium' ? 7 : HEAVY_FAVORITE_DAMPENER;
  1364→  1364→
  1365→  1365→  const homeIsHeavy = homeAvgOdds <= heavyThreshold;
  1366→  1366→  const awayIsHeavy = awayAvgOdds <= heavyThreshold;
  1367→  1367→
  1368→  1368→  if (!homeIsHeavy && !awayIsHeavy) {
  1369→  1369→    return { over25: over25Prob, over35: over35Prob, btts: bttsProb, dampened: false, reason: '' };
  1370→  1370→  }
  1371→  1371→
  1372→  1372→  const heavyOdds = Math.min(homeAvgOdds, awayAvgOdds);
  1373→  1373→  // Scale dampener: heavier favorite = bigger reduction
  1374→  1374→  // At 1.17 (PSG level): full dampener
  1375→  1375→  // At threshold: minimal 2% dampener
  1376→  1376→  const minDampener = variance === 'high' ? 2 : 3;
  1377→  1377→  const scale = Math.max(0, 1 - (heavyOdds - 1.17) / (heavyThreshold - 1.17));
  1378→  1378→  const dampener = Math.round(minDampener + (maxDampener - minDampener) * scale);
  1379→  1379→
  1380→  1380→  const leagueLabel = variance.charAt(0).toUpperCase() + variance.slice(1);
  1381→  1381→  const reason = homeIsHeavy
  1382→  1382→    ? `${leagueLabel}-variance: Heavy home favorite (${homeAvgOdds.toFixed(2)}): -${dampener}% dampener`
  1383→  1383→    : `${leagueLabel}-variance: Heavy away favorite (${awayAvgOdds.toFixed(2)}): -${dampener}% dampener`;
  1384→  1384→
  1385→  1385→  return {
  1386→  1386→    over25: Math.max(30, over25Prob - dampener),
  1387→  1387→    over35: Math.max(20, over35Prob - dampener),
  1388→  1388→    btts: Math.max(25, bttsProb - Math.round(dampener * 0.5)), // BTTS dampened less
  1389→  1389→    dampened: true,
  1390→  1390→    reason,
  1391→  1391→  };
  1392→  1392→}
  1393→  1393→
  1394→  1394→/**
  1395→  1395→ * Compute average bookie odds for a team from match results.
  1396→  1396→ * Averages odds across all matches for that team.
  1397→  1397→ *
  1398→  1398→ * @param matches - Match results with odds data
  1399→  1399→ * @param team    - Team name
  1400→  1400→ * @returns Average decimal odds, or null if insufficient data
  1401→  1401→ */
  1402→  1402→export function computeTeamAvgOdds(
  1403→  1403→  matches: { homeTeam: string; awayTeam: string; oddsAvgHome: number | null; oddsAvgAway: number | null }[],
  1404→  1404→  team: string
  1405→  1405→): number | null {
  1406→  1406→  const teamOdds: number[] = [];
  1407→  1407→
  1408→  1408→  for (const m of matches) {
  1409→  1409→    const odds = m.homeTeam === team ? m.oddsAvgHome : (m.awayTeam === team ? m.oddsAvgAway : null);
  1410→  1410→    if (odds && odds > 1 && odds < 20) {
  1411→  1411→      teamOdds.push(odds);
  1412→  1412→    }
  1413→  1413→  }
  1414→  1414→
  1415→  1415→  if (teamOdds.length < 3) return null;
  1416→  1416→
  1417→  1417→  return teamOdds.reduce((s, o) => s + o, 0) / teamOdds.length;
  1418→  1418→}

  1420→  1420→// ============================================================================
  1421→  1421→// League Variance Classification — Used by blowout/dampener adaptivity
  1422→  1422→// ============================================================================
  1423→  1423→// Classifies a league by its historical goal variance so that blowout and
  1424→  1424→// dampener thresholds can be calibrated to the league's natural scoring profile.
  1425→  1425→// High-variance leagues (2. Bundesliga, Eredivisie) see more goal explosions
  1426→  1426→// from heavy favorites, so suppression mechanisms should be gentler.
  1427→  1427→
  1428→  1428→export type LeagueVariance = 'high' | 'medium' | 'low';
  1429→  1429→
  1430→  1430→export function classifyLeagueVariance(avgGoalsPerGame: number): LeagueVariance {
  1431→  1431→  if (avgGoalsPerGame >= 3.0) return 'high';
  1432→  1432→  if (avgGoalsPerGame >= 2.5) return 'medium';
  1433→  1433→  return 'low';
  1434→  1434→}
  1435→  1435→
  1436→  1436→// ============================================================================
  1437→  1437→// Blowout Risk Qualifier — Override Under signals in massive quality gaps
  1438→  1438→// ============================================================================
  1439→  1439→// Based on live results analysis: when there's a huge quality gap (favorite's
  1440→  1440→// avg odds <= 1.50), the Regression "Under" signal is unreliable. Heavy favorites
  1441→  1441→// tend to dominate regardless of recent form — a "cooling off" period doesn't
  1442→  1442→// matter when you're 2-3 quality tiers above the opponent.
  1443→  1443→//
  1444→  1444→// This qualifier prevents the "Under" regression signal from canceling out
  1445→  1445→// legitimate "Over" signals (xG, Z-Score) in STRONG BET / GREY RESULT / GOAL FEST.
  1446→  1446→//
  1447→  1447→// Example: Man City (odds 1.20) vs Luton — Regression says "Under" (City only
  1448→  1448→// scored 1 goal last 3 games), but City will likely score 3-4 regardless.
  1449→  1449→// Without this qualifier, the Under signal would block 2 points in STRONG BET.
  1450→  1450→// ============================================================================
  1451→  1451→
  1452→  1452→/** Blowout threshold: favorite's average odds at or below this triggers override */
  1453→  1453→const BLOWOUT_ODDS_THRESHOLD = 1.50;
  1454→  1454→
  1455→  1455→/**
  1456→  1456→ * Apply blowout risk qualifier to regression signal.
  1457→  1457→ * When either team is a massive favorite (avg odds <= 1.50), override
  1458→  1458→ * Regression "Under"/"Strong Under" to "Neutral" because quality advantage
  1459→  1459→ * outweighs recent form in one-sided matchups.
  1460→  1460→ *
  1461→  1461→ * @param signals      - Current signal set (xgSignal, regressionSignal, zScoreSignal)
  1462→  1462→ * @param homeAvgOdds  - Home team average odds (decimal)
  1463→  1463→ * @param awayAvgOdds  - Away team average odds (decimal)
  1464→  1464→ * @returns Modified signals (regression may be overridden to Neutral)
  1465→  1465→ */
  1466→  1466→export function applyBlowoutQualifier(
  1467→  1467→  signals: SignalInput,
  1468→  1468→  homeAvgOdds: number | null | undefined,
  1469→  1469→  awayAvgOdds: number | null | undefined,
  1470→  1470→  leagueAvgGoalsPerGame?: number
  1471→  1471→): { signals: SignalInput; blowoutOverride: boolean; reason: string } {
  1472→  1472→  // Only applies when valid odds are available
  1473→  1473→  if (!homeAvgOdds || !awayAvgOdds || homeAvgOdds <= 0 || awayAvgOdds <= 0) {
  1474→  1474→    return { signals, blowoutOverride: false, reason: '' };
  1475→  1475→  }
  1476→  1476→
  1477→  1477→  // League-adaptive threshold: gentler in high-variance leagues
  1478→  1478→  const variance = leagueAvgGoalsPerGame ? classifyLeagueVariance(leagueAvgGoalsPerGame) : 'low';
  1479→  1479→  const blowoutThreshold = variance === 'high' ? 1.25 : variance === 'medium' ? 1.40 : BLOWOUT_ODDS_THRESHOLD;
  1480→  1480→
  1481→  1481→  const homeIsMassiveFavorite = homeAvgOdds <= blowoutThreshold;
  1482→  1482→  const awayIsMassiveFavorite = awayAvgOdds <= blowoutThreshold;
  1483→  1483→
  1484→  1484→  // Only trigger when one team is a massive favorite
  1485→  1485→  if (!homeIsMassiveFavorite && !awayIsMassiveFavorite) {
  1486→  1486→    return { signals, blowoutOverride: false, reason: '' };
  1487→  1487→  }
  1488→  1488→
  1489→  1489→  // In high-variance leagues, only override Strong Under (keep mild signals intact)
  1490→  1490→  // because hot favorites in goal-rich leagues often score freely
  1491→  1491→  const shouldOverride = variance === 'high'
  1492→  1492→    ? signals.regressionSignal === 'Strong Under'
  1493→  1493→    : signals.regressionSignal === 'Under' || signals.regressionSignal === 'Strong Under';
  1494→  1494→  if (!shouldOverride) {
  1495→  1495→    return { signals, blowoutOverride: false, reason: '' };
  1496→  1496→  }
  1497→  1497→
  1498→  1498→  const favoriteOdds = Math.min(homeAvgOdds, awayAvgOdds);
  1499→  1499→  const favorite = homeIsMassiveFavorite ? 'Home' : 'Away';
  1500→  1500→  const reason = `Blowout qualifier (${variance}-variance): ${favorite} favorite (${favoriteOdds.toFixed(2)} <= ${blowoutThreshold}) — Regression "${signals.regressionSignal}" overridden to Neutral`;
  1501→  1501→
  1502→  1502→  return {
  1503→  1503→    signals: {
  1504→  1504→      ...signals,
  1505→  1505→      regressionSignal: 'Neutral',
  1506→  1506→    },
  1507→  1507→    blowoutOverride: true,
  1508→  1508→    reason,
  1509→  1509→  };
  1510→  1510→}

  1512→  1512→// ============================================================================
  1513→  1513→// BTTS Dual-Team Qualification System
  1514→  1514→// ============================================================================
  1515→  1515→// BTTS requires BOTH teams to score — the match-level signal combo (Regression
  1516→  1516→// Under + Z-Score Neutral + xG Over) correctly identifies "goals are coming"
  1517→  1517→// but tends to surface ONE-SIDED regressions (one team due for a breakout).

  1590→  1590→  if (zScore >= 1.5) return 'Strong Under';
  1591→  1591→  if (zScore >= 1.0) return 'Under';

  1602→  1602→export interface MomentumTeamInput {
  1603→  1603→  last5Goals: number[];    // total goals per match in last 5
  1604→  1604→  seasonAvg: number;       // season avg total goals per game
  1605→  1605→  xgDiff: number;          // actualGoals/matches - totalXg/matches (positive = overperforming)

  1610→  1610→  homeSignal: string;      // 'Strong Momentum' | 'Momentum' | 'Neutral' | 'Cooling' | 'Cold'
  1611→  1611→  awaySignal: string;
  1612→  1612→  matchSignal: string;     // 'MOMENTUM OVER' | 'MOMENTUM LEAN OVER' | 'MOMENTUM NEUTRAL' | 'MOMENTUM LEAN UNDER' | 'MOMENTUM UNDER'

  1614→  1614→
  1615→  1615→function classifyTeamMomentum(team: MomentumTeamInput): string {
  1616→  1616→  if (team.last5Goals.length < 3) return 'Neutral';
  1617→  1617→
  1618→  1618→  const last5Avg = team.last5Goals.reduce((s, g) => s + g, 0) / team.last5Goals.length;
  1619→  1619→  const last3 = team.last5Goals.slice(0, Math.min(3, team.last5Goals.length));
  1620→  1620→  const last3Avg = last3.reduce((s, g) => s + g, 0) / last3.length;
  1621→  1621→
  1622→  1622→  // Consistency: coefficient of variation of last 5
  1623→  1623→  const mean = last5Avg;
  1624→  1624→  const variance = team.last5Goals.reduce((s, g) => s + (g - mean) ** 2, 0) / team.last5Goals.length;
  1625→  1625→  const stdDev = Math.sqrt(variance);
  1626→  1626→  const cv = mean > 0 ? stdDev / mean : 999;
  1627→  1627→
  1628→  1628→  // Sustained: 5+ game window above season average by meaningful margin
  1629→  1629→  const sustained = last5Avg > team.seasonAvg * 1.15;
  1630→  1630→
  1631→  1631→  // Accelerating: last 3 even hotter than last 5
  1632→  1632→  const accelerating = last3Avg > last5Avg * 1.05;
  1633→  1633→
  1634→  1634→  // xG sustainability: overperforming xG with shot volume to back it
  1635→  1635→  const xgSustainable = team.xgDiff > 0 && team.sotPerGame !== null && team.sotPerGame > 5.0;
  1636→  1636→
  1637→  1637→  // Strong Momentum: sustained + consistent + high output
  1638→  1638→  if (sustained && cv < 0.5 && last5Avg >= 2.0) return 'Strong Momentum';
  1639→  1639→  // Momentum: sustained with moderate variance, or recent surge with sustainable backing
  1640→  1640→  if (sustained && cv < 0.65) return 'Momentum';
  1641→  1641→  if (last5Avg > team.seasonAvg * 1.15 && accelerating && (xgSustainable || team.xgDiff <= 0)) return 'Momentum';
  1642→  1642→  // Cold: declining output
  1643→  1643→  if (last5Avg < team.seasonAvg * 0.75) {
  1644→  1644→    return last3Avg < last5Avg * 0.8 ? 'Cold' : 'Cooling';
  1645→  1645→  }
  1646→  1646→  return 'Neutral';
  1647→  1647→}

  1649→  1649→export function computeMomentumSignal(home: MomentumTeamInput, away: MomentumTeamInput): MomentumResult {
  1650→  1650→  const homeSignal = classifyTeamMomentum(home);
  1651→  1651→  const awaySignal = classifyTeamMomentum(away);
  1652→  1652→
  1653→  1653→  const isOver = (s: string) => s === 'Strong Momentum' || s === 'Momentum';
  1654→  1654→  const isUnder = (s: string) => s === 'Cold' || s === 'Cooling';
  1655→  1655→
  1656→  1656→  let matchSignal: string;
  1657→  1657→  if (isOver(homeSignal) && isOver(awaySignal)) matchSignal = 'MOMENTUM OVER';
  1658→  1658→  else if ((isOver(homeSignal) || isOver(awaySignal)) && !isUnder(homeSignal) && !isUnder(awaySignal)) matchSignal = 'MOMENTUM LEAN OVER';
  1659→  1659→  else if (isUnder(homeSignal) && isUnder(awaySignal)) matchSignal = 'MOMENTUM UNDER';
  1660→  1660→  else if (isUnder(homeSignal) || isUnder(awaySignal)) matchSignal = 'MOMENTUM LEAN UNDER';
  1661→  1661→  else matchSignal = 'MOMENTUM NEUTRAL';
  1662→  1662→
  1663→  1663→  return { homeSignal, awaySignal, matchSignal };
  1664→  1664→}

  1666→  1666→// ============================================================================
  1667→  1667→// BTTS Dual-Team Qualification System
  1668→  1668→// ============================================================================
  1669→  1669→// Evaluates each team independently to determine whether BOTH teams are likely
  1670→  1670→// to score. Returns a tier that caps BTTS probability independently from the
  1671→  1671→// Over goal thesis.
  1672→  1672→//
  1673→  1673→// Now supports TWO qualification pathways:
  1674→  1674→//   1. Regression Under: both teams underperforming → due to bounce back
  1675→  1675→//   2. Momentum Over: both teams on hot streaks → likely to keep scoring
  1676→  1676→//
  1677→  1677→// Tier mapping:
  1678→  1678→//   5+ → BTTS STRONG   (cap 70%) — both teams individually qualify
  1679→  1679→//   3–4 → BTTS QUALIFIED (cap 55%) — mild BTTS confidence
  1680→  1680→//   1–2 → BTTS WEAK     (cap 40%) — one-sided regression likely
  1681→  1681→//   ≤0  → BTTS AVOID    (cap 25%) — do not bet BTTS
  1682→  1682→// ============================================================================
  1683→  1683→export function computeBTTSQualification(input: BttsQualificationInput): BttsQualificationResult {
  1684→  1684→  let score = 0;
  1685→  1685→  const checks: { check: string; points: number; passed: boolean }[] = [];
  1686→  1686→
  1687→  1687→  const isRegressionUnder = (sig: string) => sig === 'Under' || sig === 'Strong Under';
  1688→  1688→  const isXgOver = (diff: number) => diff <= -0.5; // underperforming xG = goals due
  1689→  1689→  const isZCold = (z: number) => z >= 1.0; // overperforming Z = cold streak coming
  1690→  1690→  const isZNeutralOrBetter = (z: number) => z < 1.0; // not overheated
  1691→  1691→
  1692→  1692→  // Momentum helpers
  1693→  1693→  const isMomentum = (s: string | undefined) => s === 'Strong Momentum' || s === 'Momentum';
  1694→  1694→  const homeMom = isMomentum(input.homeMomentumSignal);
  1695→  1695→  const awayMom = isMomentum(input.awayMomentumSignal);
  1696→  1696→  const bothMom = homeMom && awayMom;
  1697→  1697→  const oneMom = homeMom || awayMom;
  1698→  1698→
  1699→  1699→  // Check 1: Regression Under OR Momentum Over — dual qualification pathway
  1700→  1700→  const bothRegUnder = isRegressionUnder(input.homeRegressionSignal) && isRegressionUnder(input.awayRegressionSignal);
  1701→  1701→  const oneRegUnder = isRegressionUnder(input.homeRegressionSignal) || isRegressionUnder(input.awayRegressionSignal);
  1702→  1702→
  1703→  1703→  if (bothRegUnder && bothMom) {
  1704→  1704→    // Supercharged: both signals agree from both sides
  1705→  1705→    score += 5;
  1706→  1706→    checks.push({ check: 'Both Regression Under AND Momentum Over (supercharged)', points: 5, passed: true });
  1707→  1707→  } else if (bothRegUnder || bothMom) {
  1708→  1708→    score += 3;
  1709→  1709→    checks.push({ check: bothRegUnder ? 'Both teams Regression Under/StrongUnder' : 'Both teams Momentum Over (hot streak)', points: 3, passed: true });
  1710→  1710→  } else if ((oneRegUnder && oneMom)) {
  1711→  1711→    score += 2; // Mixed but both signal types point to goals
  1712→  1712→    checks.push({ check: 'One Regression Under + One Momentum Over', points: 2, passed: true });
  1713→  1713→  } else if (oneRegUnder || oneMom) {
  1714→  1714→    score += 1;
  1715→  1715→    checks.push({ check: oneRegUnder ? 'One team Regression Under' : 'One team Momentum Over', points: 1, passed: true });
  1716→  1716→  } else {
  1717→  1717→    checks.push({ check: 'No Regression Under or Momentum Over', points: 3, passed: false });
  1718→  1718→  }
  1719→  1719→
  1720→  1720→  // Check 2: BOTH teams xG Over (underperforming xG = creating chances, finishing due)
  1721→  1721→  const bothXgOver = isXgOver(input.homeXgDiff) && isXgOver(input.awayXgDiff);
  1722→  1722→  if (bothXgOver) {
  1723→  1723→    score += 2;
  1724→  1724→    checks.push({ check: 'Both teams xG Over (underperforming)', points: 2, passed: true });
  1725→  1725→  } else {
  1726→  1726→    checks.push({ check: 'Both teams xG Over (underperforming)', points: 2, passed: false });
  1727→  1727→  }
  1728→  1728→
  1729→  1729→  // Check 3: BOTH teams Z-Score not overheated
  1730→  1730→  const bothZOk = isZNeutralOrBetter(input.homeZScore) && isZNeutralOrBetter(input.awayZScore);
  1731→  1731→  if (bothZOk) {
  1732→  1732→    score += 1;
  1733→  1733→    checks.push({ check: 'Both teams Z-Score Neutral/better', points: 1, passed: true });
  1734→  1734→  } else {
  1735→  1735→    checks.push({ check: 'Both teams Z-Score Neutral/better', points: 1, passed: false });
  1736→  1736→  }
  1737→  1737→
  1738→  1738→  // Check 4: Competitive matchup (favorite odds 1.80–2.50)
  1739→  1739→  if (input.favoriteOdds && input.favoriteOdds >= 1.80 && input.favoriteOdds <= 2.50) {
  1740→  1740→    score += 1;
  1741→  1741→    checks.push({ check: 'Competitive matchup (odds 1.80-2.50)', points: 1, passed: true });
  1742→  1742→  } else {
  1743→  1743→    checks.push({ check: 'Competitive matchup (odds 1.80-2.50)', points: 1, passed: false });
  1744→  1744→  }
  1745→  1745→
  1746→  1746→  // Penalty 5: Massive favorite (≤1.50) — one side dominates, BTTS unlikely
  1747→  1747→  if (input.favoriteOdds && input.favoriteOdds <= 1.50) {
  1748→  1748→    score -= 2;
  1749→  1749→    checks.push({ check: 'Massive favorite (odds ≤1.50)', points: -2, passed: true });
  1750→  1750→  } else {
  1751→  1751→    checks.push({ check: 'Massive favorite (odds ≤1.50)', points: -2, passed: false });
  1752→  1752→  }
  1753→  1753→
  1754→  1754→  // Penalty 6: Either team can't finish (SOT conversion <25%)
  1755→  1755→  const homeCantFinish = input.homeSotConversion !== null && input.homeSotConversion < 25;
  1756→  1756→  const awayCantFinish = input.awaySotConversion !== null && input.awaySotConversion < 25;
  1757→  1757→  if (homeCantFinish) score -= 1;
  1758→  1758→  if (awayCantFinish) score -= 1;
  1759→  1759→  checks.push({
  1760→  1760→    check: `Home SOT conversion ${homeCantFinish ? '< 25%' : '≥ 25%'}${input.homeSotConversion !== null ? ` (${input.homeSotConversion.toFixed(0)}%)` : ' (N/A)'}`,
  1761→  1761→    points: -1,
  1762→  1762→    passed: !homeCantFinish,
  1763→  1763→  });
  1764→  1764→  checks.push({
  1765→  1765→    check: `Away SOT conversion ${awayCantFinish ? '< 25%' : '≥ 25%'}${input.awaySotConversion !== null ? ` (${input.awaySotConversion.toFixed(0)}%)` : ' (N/A)'}`,
  1766→  1766→    points: -1,
  1767→  1767→    passed: !awayCantFinish,
  1768→  1768→  });
  1769→  1769→
  1770→  1770→  // Penalty 7: Either team in a strong cold Z-Score streak
  1771→  1771→  const homeCold = isZCold(input.homeZScore);
  1772→  1772→  const awayCold = isZCold(input.awayZScore);
  1773→  1773→  if (homeCold) score -= 2;
  1774→  1774→  if (awayCold) score -= 2;
  1775→  1775→  checks.push({
  1776→  1776→    check: `Home Z-Score ${homeCold ? 'cold (≥1.0)' : 'ok'}`,
  1777→  1777→    points: -2,
  1778→  1778→    passed: !homeCold,
  1779→  1779→  });
  1780→  1780→  checks.push({
  1781→  1781→    check: `Away Z-Score ${awayCold ? 'cold (≥1.0)' : 'ok'}`,
  1782→  1782→    points: -2,
  1783→  1783→    passed: !awayCold,
  1784→  1784→  });
  1785→  1785→
  1786→  1786→  // Determine tier
  1787→  1787→  let tier: BttsQualificationTier;
  1788→  1788→  let bttsCap: number;
  1789→  1789→  if (score >= 5) {
  1790→  1790→    tier = 'BTTS STRONG';
  1791→  1791→    bttsCap = 70;
  1792→  1792→  } else if (score >= 3) {
  1793→  1793→    tier = 'BTTS QUALIFIED';
  1794→  1794→    bttsCap = 55;
  1795→  1795→  } else if (score >= 1) {
  1796→  1796→    tier = 'BTTS WEAK';
  1797→  1797→    bttsCap = 40;
  1798→  1798→  } else {
  1799→  1799→    tier = 'BTTS AVOID';
  1800→  1800→    bttsCap = 25;
  1801→  1801→  }
  1802→  1802→
  1803→  1803→  return { tier, score, breakdown: checks, bttsCap };
  1804→  1804→}

  1856→  1856→ *   4–5 → GOAL LIKELY    — Solid Over 2.5 bet
  1857→  1857→ *   2–3 → GOAL BORDERLINE — Over 1.5 strong, Over 2.5 cautious
  1858→  1858→ *   0–1 → GOAL THIN      — Over 1.5 only, skip Over 2.5

  1861→  1861→export function computeThirdGoalQualifier(input: ThirdGoalQualifierInput): ThirdGoalQualifierResult {
  1862→  1862→  let score = 0;
  1863→  1863→  const checks: { check: string; points: number; passed: boolean }[] = [];
  1864→  1864→
  1865→  1865→  const isRegressionUnder = (sig: string) => sig === 'Under' || sig === 'Strong Under';
  1866→  1866→  const isXgOver = (sig: string) => sig === 'Over' || sig === 'Strong Over';
  1867→  1867→
  1868→  1868→  // Check 1: Combined xG > 3.0 (very high chance creation)
  1869→  1869→  if (input.combinedXgTotal > 3.0) {
  1870→  1870→    score += 3;
  1871→  1871→    checks.push({ check: `Combined xG ${input.combinedXgTotal.toFixed(2)} > 3.0`, points: 3, passed: true });
  1872→  1872→  } else if (input.combinedXgTotal >= 2.5) {
  1873→  1873→    score += 1;
  1874→  1874→    checks.push({ check: `Combined xG ${input.combinedXgTotal.toFixed(2)} in 2.5-3.0`, points: 1, passed: true });
  1875→  1875→  } else {
  1876→  1876→    checks.push({ check: `Combined xG ${input.combinedXgTotal.toFixed(2)} < 2.5`, points: 3, passed: false });
  1877→  1877→  }

     1→// ============================================================================
     2→// Shared Betting Filter Configuration — Single Source of Truth
     3→// ============================================================================
     4→// Lean indicator system — only checks with proven lift survive.
     5→// Removed noise: league-level constants (zero match discrimination),
     6→// season-average goals (stale vs rolling), shot conversion (unverified),
     7→// xG/Z-Score/Regression mild states (fire 60-90% = no selectivity).
     8→//
     9→// Surviving signals:
    10→//   BTTS/O2.5/O3.5 model probabilities (direct target market)
    11→//   O2.5 implied probability from bookmaker odds (strongest single predictor)
    12→//   Rolling combined scoring (recent form, independent of model)
    13→//   Draw probability inverted (structural, independent for BTTS-BH)
    14→//   Signal divergence (xG Over + Regression Under = specific pattern)
    15→//   BTTS probability band (selective constraint for Grey Result)
    16→// =============================================================================
    17→
    18→// ============================================================================
    19→// League-level baselines — computed from actual results data
    20→// ============================================================================
    21→export interface LeagueBaselines {
    22→  avgGoalsPerGame: number;   // from analytics
    23→  over25Rate: number;        // from analytics (%)
    24→  bttsRate: number;          // computed from results (%)
    25→  over35Rate: number;        // computed from results (%)
    26→  avgHomeGoals: number;      // from analytics
    27→  avgAwayGoals: number;      // from analytics
    28→  shotConversion: number;    // from analytics (%)
    29→}
    30→
    31→/**
    32→ * Compute league baselines from results + analytics data.
    33→ * Call this once per league/season, pass the result into all filter functions.
    34→ */
    35→export function computeLeagueBaselines(
    36→  results: { ftHomeGoals: number; ftAwayGoals: number }[],
    37→  analytics: {
    38→    avgGoalsPerGame: number;
    39→    over25Percent: number;
    40→    avgHomeGoals: number;
    41→    avgAwayGoals: number;
    42→    overallShotConversion: string | number;
    43→  }
    44→): LeagueBaselines {
    45→  const total = results.length || 1;
    46→  const bttsCount = results.filter(r => r.ftHomeGoals > 0 && r.ftAwayGoals > 0).length;
    47→  const over35Count = results.filter(r => r.ftHomeGoals + r.ftAwayGoals > 3.5).length;
    48→
    49→  return {
    50→    avgGoalsPerGame: analytics.avgGoalsPerGame,
    51→    over25Rate: analytics.over25Percent,
    52→    bttsRate: (bttsCount / total) * 100,
    53→    over35Rate: (over35Count / total) * 100,
    54→    avgHomeGoals: analytics.avgHomeGoals,
    55→    avgAwayGoals: analytics.avgAwayGoals,
    56→    shotConversion: typeof analytics.overallShotConversion === 'string'
    57→      ? parseFloat(analytics.overallShotConversion)
    58→      : analytics.overallShotConversion,
    59→  };
    60→}

   110→// ============================================================================
   111→// Backtest-derived threshold registry
   112→// ============================================================================
   113→// This is the central store. Thresholds are added here when backtests run.
   114→// Leagues without entries fall back to the hybrid formula.
   115→// Supports localStorage persistence for client-side survival across page reloads.
   116→// ============================================================================
   117→
   118→const STORAGE_KEY = 'techmari_backtest_thresholds';
   119→
   120→const backtestRegistry = new Map<string, LeagueBacktestThresholds>();
   121→
   122→let loadedFromStorage = false;
   123→
   124→/**
   125→ * Load thresholds from localStorage into the in-memory registry.
   126→ * Safe to call multiple times — only loads once per session.
   127→ */
   128→function loadFromLocalStorage(): void {
   129→  if (loadedFromStorage || typeof window === 'undefined') return;
   130→  loadedFromStorage = true;
   131→  try {
   132→    const raw = localStorage.getItem(STORAGE_KEY);
   133→    if (!raw) return;
   134→    const data = JSON.parse(raw);
   135→    if (data.version !== 1) return;
   136→    const entries: Record<string, LeagueBacktestThresholds> = data.thresholds;
   137→    for (const [key, thresholds] of Object.entries(entries)) {
   138→      backtestRegistry.set(key, thresholds);
   139→    }
   140→    const count = Object.keys(entries).length;
   141→    if (count > 0) {
   142→      console.log(`[BettingFilters] Loaded ${count} persisted threshold(s) from localStorage`);
   143→    }
   144→  } catch {
   145→    // localStorage not available or corrupt data — silent fail
   146→  }
   147→}
   148→
   149→/**
   150→ * Save all in-memory thresholds to localStorage.
   151→ */
   152→function saveToLocalStorage(): void {
   153→  if (typeof window === 'undefined') return;
   154→  try {
   155→    const data: Record<string, LeagueBacktestThresholds> = {};
   156→    for (const [key, thresholds] of backtestRegistry.entries()) {
   157→      data[key] = thresholds;
   158→    }
   159→    localStorage.setItem(STORAGE_KEY, JSON.stringify({
   160→      version: 1,
   161→      updatedAt: new Date().toISOString(),
   162→      thresholds: data,
   163→    }));
   164→  } catch {
   165→    // localStorage not available or quota exceeded — silent fail
   166→  }
   167→}
   168→
   169→/**
   170→ * Register backtest-derived thresholds for a league.
   171→ * Also persists to localStorage automatically.
   172→ */
   173→export function registerBacktestThresholds(thresholds: LeagueBacktestThresholds): void {
   174→  // Ensure we've loaded from localStorage first
   175→  loadFromLocalStorage();
   176→  backtestRegistry.set(thresholds.leagueName.toLowerCase(), thresholds);
   177→  saveToLocalStorage();
   178→}
   179→
   180→/**
   181→ * Get backtest-derived thresholds for a league, if available.
   182→ */
   183→export function getBacktestThresholds(leagueName: string): LeagueBacktestThresholds | undefined {
   184→  loadFromLocalStorage();
   185→  return backtestRegistry.get(leagueName.toLowerCase());
   186→}
   187→
   188→/**
   189→ * Get all registered backtest thresholds (for display/export).
   190→ */
   191→export function getAllBacktestThresholds(): LeagueBacktestThresholds[] {
   192→  loadFromLocalStorage();
   193→  return Array.from(backtestRegistry.values());
   194→}

   196→/**
   197→ * Check if a league has backtest-derived thresholds with sufficient sample size.
   198→ */
   199→export function hasSufficientBacktest(leagueName: string, minSampleSize = 150): boolean {
   200→  loadFromLocalStorage();
   201→  const t = backtestRegistry.get(leagueName.toLowerCase());
   202→  return t !== undefined && t.sampleSize >= minSampleSize;
   203→}

   364→// ============================================================================
   365→// Hybrid threshold helper (fallback formula)

   367→function hybridThreshold(floor: number, baseline: number, multiplier: number): number {
   368→  return Math.max(floor, baseline * multiplier);

   410→/**
   411→ * Resolve all thresholds for a given league.
   412→ * Uses backtest-derived if available (sample >= minSampleSize), otherwise hybrid.
   413→ */
   414→export function resolveAllThresholds(
   415→  leagueName: string | undefined,
   416→  baselines: LeagueBaselines,
   417→  minSampleSize = 150
   418→) {
   419→  const bt = leagueName ? getBacktestThresholds(leagueName) : undefined;
   420→  const useBacktest = bt !== undefined && bt.sampleSize >= minSampleSize;
   421→
   422→  const src: 'backtest' | 'hybrid' = useBacktest ? 'backtest' : 'hybrid';
   423→
   424→  // BTTS match-level thresholds (lean 2-check)
   425→  const btts: ResolvedThresholds = useBacktest
   426→    ? { modelBttsProb: bt!.btts.modelBttsProb, modelO25Prob: bt!.btts.modelO25Prob, source: 'backtest' as const }
   427→    : {
   428→        modelBttsProb: hybridThreshold(BTTS_HYBRID_THRESHOLDS.modelBttsProb.floor, baselines.bttsRate, BTTS_HYBRID_THRESHOLDS.modelBttsProb.multiplier),
   429→        modelO25Prob: hybridThreshold(BTTS_HYBRID_THRESHOLDS.modelO25Prob.floor, baselines.over25Rate, BTTS_HYBRID_THRESHOLDS.modelO25Prob.multiplier),
   430→        source: 'hybrid' as const,
   431→      };
   432→
   433→  // Over 3.5 match-level thresholds (lean 2-check)
   434→  const over35: ResolvedOver35Thresholds = useBacktest
   435→    ? { modelO35Prob: bt!.over35.modelO35Prob, bttsProb: bt!.over35.bttsProb, source: 'backtest' as const }
   436→    : {
   437→        modelO35Prob: hybridThreshold(OVER35_HYBRID_THRESHOLDS.modelO35Prob.floor, baselines.over35Rate, OVER35_HYBRID_THRESHOLDS.modelO35Prob.multiplier),
   438→        bttsProb: hybridThreshold(OVER35_HYBRID_THRESHOLDS.bttsProb.floor, baselines.bttsRate, OVER35_HYBRID_THRESHOLDS.bttsProb.multiplier),
   439→        source: 'hybrid' as const,
   440→      };
   441→
   442→  // STRONG BET thresholds
   443→  const strongBet: ResolvedStrongBetThresholds = useBacktest
   444→    ? { ...bt!.strongBet, source: 'backtest' }
   445→    : {
   446→        o25Prob: hybridThreshold(STRONG_BET_HYBRID.o25Prob.floor, baselines.over25Rate, STRONG_BET_HYBRID.o25Prob.multiplier),
   447→        o35Prob: hybridThreshold(STRONG_BET_HYBRID.o35Prob.floor, baselines.over35Rate, STRONG_BET_HYBRID.o35Prob.multiplier),
   448→        bttsProb: hybridThreshold(STRONG_BET_HYBRID.bttsProb.floor, baselines.bttsRate, STRONG_BET_HYBRID.bttsProb.multiplier),
   449→        bttsChecklistCount: STRONG_BET_HYBRID.bttsChecklistCount,
   450→        source: 'hybrid' as const,
   451→      };
   452→
   453→  // GREY RESULT thresholds
   454→  const greyResult: ResolvedGreyResultThresholds = useBacktest
   455→    ? {
   456→        ...bt!.greyResult,
   457→        bttsProbMax: (bt!.greyResult as any).bttsProbMax ?? GREY_RESULT_CONFIG.bttsProb.max,
   458→        source: 'backtest' as const,
   459→      }
   460→    : {
   461→        bttsProb: GREY_RESULT_CONFIG.bttsProb.min,
   462→        bttsProbMax: GREY_RESULT_CONFIG.bttsProb.max,
   463→        o25Prob: hybridThreshold(GREY_RESULT_CONFIG.o25Prob.floor, baselines.over25Rate, GREY_RESULT_CONFIG.o25Prob.multiplier),
   464→        o35Prob: hybridThreshold(GREY_RESULT_CONFIG.o35Prob.floor, baselines.over35Rate, GREY_RESULT_CONFIG.o35Prob.multiplier),
   465→        bttsChecklistCount: GREY_RESULT_CONFIG.bttsChecklistCount,
   466→        over35ChecklistCount: GREY_RESULT_CONFIG.over35ChecklistCount,
   467→        requiredChecks: GREY_RESULT_CONFIG.requiredChecks,
   468→        source: 'hybrid' as const,
   469→      };
   470→
   471→  // GOAL FEST thresholds (lean 6-check)
   472→  const goalFest: ResolvedGoalFestThresholds = useBacktest
   473→    ? {
   474→        o25Prob: bt!.strongBet.o25Prob,
   475→        bttsProb: bt!.strongBet.bttsProb,
   476→        o35Prob: bt!.over35.modelO35Prob,
   477→        requiredChecks: GOAL_FEST_CONFIG.requiredChecks,
   478→        source: 'backtest' as const,
   479→      }
   480→    : {
   481→        o25Prob: hybridThreshold(GOAL_FEST_CONFIG.o25Prob.floor, baselines.over25Rate, GOAL_FEST_CONFIG.o25Prob.multiplier),
   482→        bttsProb: hybridThreshold(GOAL_FEST_CONFIG.bttsProb.floor, baselines.bttsRate, GOAL_FEST_CONFIG.bttsProb.multiplier),
   483→        o35Prob: hybridThreshold(GOAL_FEST_CONFIG.o35Prob.floor, baselines.over35Rate, GOAL_FEST_CONFIG.o35Prob.multiplier),
   484→        requiredChecks: GOAL_FEST_CONFIG.requiredChecks,
   485→        source: 'hybrid' as const,
   486→      };
   487→
   488→  return { btts, over35, strongBet, greyResult, goalFest, source: src };
   489→}

   588→/**
   589→ * Compute STRONG BET using lean points-based system.
   590→ * Only proven predictors: model probs, market implied, rolling form,
   591→ * checklist composites, and signal divergence.
   592→ */
   593→export function computeStrongBet(
   594→  checklistInput: ChecklistInput,
   595→  signals: SignalInput,
   596→  resolved: ReturnType<typeof resolveAllThresholds>,
   597→  options?: { momentumSignal?: string; leagueBttsRate?: number }
   598→): {
   599→  isStrongBet: boolean;
   600→  points: number;
   601→  maxPoints: number;
   602→  breakdown: { check: string; points: number; passed: boolean }[];
   603→} {
   604→  const st = resolved.strongBet;
   605→  const p = STRONG_BET_POINTS;
   606→
   607→  const bttsCount = computeBttsChecklist(checklistInput, resolved);
   608→
   609→  // Signal divergence: xG Over + Regression Under = high-variance game
   610→  const isRegressionUnder = (sig: string) => sig === 'Under' || sig === 'Strong Under';
   611→  const hasSignalDivergence = (xg: string, reg: string) =>
   612→    (xg === 'Over' || xg === 'Strong Over') && isRegressionUnder(reg);
   613→
   614→  const checks = [
   615→    { check: 'O2.5 >=' + st.o25Prob.toFixed(0) + '%', points: p.o25, passed: checklistInput.o25Prob >= st.o25Prob },
   616→    { check: 'O3.5 >=' + st.o35Prob.toFixed(0) + '%', points: p.o35, passed: checklistInput.o35Prob >= st.o35Prob },
   617→    { check: 'BTTS Checklist >=' + st.bttsChecklistCount + '/4', points: p.bttsChecklist, passed: bttsCount >= st.bttsChecklistCount },
   618→    { check: 'Signal Divergence (xG vs Reg)', points: p.signalDivergence, passed: hasSignalDivergence(signals.xgSignal, signals.regressionSignal) },
   619→    { check: 'O2.5 Implied >=' + O25_IMPLIED_THRESHOLDS.strongBet + '%', points: p.o25Implied, passed: checklistInput.o25ImpliedProb !== null && checklistInput.o25ImpliedProb >= O25_IMPLIED_THRESHOLDS.strongBet },
   620→    { check: 'Rolling Scoring >=' + ROLLING_SCORING_THRESHOLDS.strongBet, points: p.rollingScoring, passed: checklistInput.rollingCombinedScoring >= ROLLING_SCORING_THRESHOLDS.strongBet },
   621→  ];
   622→
   623→  const totalPoints = checks.reduce((sum, c) => sum + (c.passed ? c.points : 0), 0);
   624→  const maxPoints = checks.reduce((sum, c) => sum + c.points, 0);
   625→
   626→  return {
   627→    isStrongBet: totalPoints >= p.threshold,
   628→    points: totalPoints,
   629→    maxPoints,
   630→    breakdown: checks,
   631→  };
   632→}

   639→export function computeGreyResult(
   640→  checklistInput: ChecklistInput,

   651→  const o35Count = computeOver35Checklist(checklistInput, resolved);
   652→
   653→  const checks = [
   654→    { check: 'BTTS Checklist >=' + gt.bttsChecklistCount + '/4', passed: bttsCount >= gt.bttsChecklistCount },

   656→    { check: 'O2.5 >=' + gt.o25Prob.toFixed(0) + '%', passed: checklistInput.o25Prob >= gt.o25Prob },

   746→// ============================================================================
   747→// Backtest threshold derivation utility
   748→// ============================================================================
   749→
   750→interface BacktestMatch {
   751→  ftHomeGoals: number;
   752→  ftAwayGoals: number;
   753→  // Model predictions (raw or calibrated)
   754→  predictedBtts?: number;
   755→  predictedO25?: number;
   756→  predictedO35?: number;
   757→  // Match context
   758→  avgHomeGoals?: number;
   759→  avgAwayGoals?: number;
   760→  shotConversion?: number;
   761→}
   762→
   763→interface BacktestDerivationOptions {
   764→  /** Minimum sample size to trust derived thresholds */
   765→  minSampleSize?: number;
   766→  /** For probability thresholds: target minimum precision (e.g. 0.65 means at least 65% accuracy) */
   767→  minAccuracy?: number;
   768→  /** For numeric thresholds (goals): target minimum precision */
   769→  minGoalAccuracy?: number;
   770→}
   771→
   772→/**
   773→ * Derive optimal thresholds from backtest data using accuracy optimization.
   774→ *
   775→ * For each criterion, sweeps candidate threshold values and finds the one that
   776→ * maximizes the correct prediction rate while meeting the minimum accuracy target.
   777→ *
   778→ * @param leagueName - League identifier for the registry
   779→ * @param matches - Historical match data with predictions and outcomes
   780→ * @param currentBaselines - Current league baselines (for fallback floors)
   781→ * @param options - Tuning parameters
   782→ * @returns Derived thresholds, or null if insufficient data
   783→ */
   784→export function deriveThresholdsFromBacktest(
   785→  leagueName: string,
   786→  matches: BacktestMatch[],
   787→  currentBaselines: LeagueBaselines,
   788→  options: BacktestDerivationOptions = {}
   789→): LeagueBacktestThresholds | null {
   790→  const {
   791→    minSampleSize = 150,
   792→    minAccuracy = 0.60,
   793→    minGoalAccuracy = 0.55,
   794→  } = options;
   795→
   796→  if (matches.length < minSampleSize) return null;
   797→
   798→  // --- Helper: compute BTTS hit rate at various thresholds ---
   799→  const findOptimalProbThreshold = (
   800→    getPredicted: (m: BacktestMatch) => number | undefined,
   801→    getActual: (m: BacktestMatch) => boolean,
   802→    searchRange: [number, number],
   803→    step = 1,
   804→    floor?: number
   805→  ): number => {
   806→    const [lo, hi] = searchRange;
   807→    let bestThreshold = floor ?? lo;
   808→    let bestAccuracy = 0;
   809→    let bestSampleAbove = 0;
   810→
   811→    for (let t = lo; t <= hi; t += step) {
   812→      const aboveThreshold = matches.filter(m => {
   813→        const pred = getPredicted(m);
   814→        return pred !== undefined && pred >= t;
   815→      });
   816→      if (aboveThreshold.length < 30) continue; // need meaningful sample above threshold
   817→
   818→      const correct = aboveThreshold.filter(m => getActual(m)).length;
   819→      const accuracy = correct / aboveThreshold.length;
   820→
   821→      // Prefer: highest accuracy that meets minimum, with most samples
   822→      if (accuracy >= minAccuracy && (accuracy > bestAccuracy || (accuracy === bestAccuracy && aboveThreshold.length > bestSampleAbove))) {
   823→        bestThreshold = t;
   824→        bestAccuracy = accuracy;
   825→        bestSampleAbove = aboveThreshold.length;
   826→      }
   827→    }
   828→
   829→    // If no threshold met min accuracy, use the one with best accuracy above floor
   830→    if (bestAccuracy < minAccuracy && floor !== undefined) {
   831→      return floor;
   832→    }
   833→    return bestThreshold;
   834→  };
   835→
   836→  const findOptimalGoalThreshold = (
   837→    getPredicted: (m: BacktestMatch) => number | undefined,
   838→    getActual: (m: BacktestMatch) => boolean,
   839→    searchRange: [number, number],
   840→    step = 0.05,
   841→    floor?: number
   842→  ): number => {
   843→    const [lo, hi] = searchRange;
   844→    let bestThreshold = floor ?? lo;
   845→    let bestAccuracy = 0;
   846→    let bestSampleAbove = 0;
   847→
   848→    for (let t = lo; t <= hi; t += step) {
   849→      const aboveThreshold = matches.filter(m => {
   850→        const pred = getPredicted(m);
   851→        return pred !== undefined && pred >= t;
   852→      });
   853→      if (aboveThreshold.length < 30) continue;
   854→
   855→      const correct = aboveThreshold.filter(m => getActual(m)).length;
   856→      const accuracy = correct / aboveThreshold.length;
   857→
   858→      if (accuracy >= minGoalAccuracy && (accuracy > bestAccuracy || (accuracy === bestAccuracy && aboveThreshold.length > bestSampleAbove))) {
   859→        bestThreshold = Math.round(t * 100) / 100;
   860→        bestAccuracy = accuracy;
   861→        bestSampleAbove = aboveThreshold.length;
   862→      }
   863→    }
   864→
   865→    if (bestAccuracy < minGoalAccuracy && floor !== undefined) {
   866→      return floor;
   867→    }
   868→    return bestThreshold;
   869→  };
   870→
   871→  // --- Derive BTTS thresholds (lean 2-check) ---
   872→  const bttsModelProb = findOptimalProbThreshold(
   873→    m => m.predictedBtts,
   874→    m => m.ftHomeGoals > 0 && m.ftAwayGoals > 0,
   875→    [40, 80],
   876→    1,
   877→    BTTS_HYBRID_THRESHOLDS.modelBttsProb.floor
   878→  );
   879→
   880→  const bttsO25Prob = findOptimalProbThreshold(
   881→    m => m.predictedO25,
   882→    m => m.ftHomeGoals > 0 && m.ftAwayGoals > 0,
   883→    [45, 85],
   884→    1,
   885→    BTTS_HYBRID_THRESHOLDS.modelO25Prob.floor
   886→  );
   887→
   888→  // --- Derive Over 3.5 thresholds (lean 2-check) ---
   889→  const o35ModelProb = findOptimalProbThreshold(
   890→    m => m.predictedO35,
   891→    m => (m.ftHomeGoals + m.ftAwayGoals) > 3.5,
   892→    [20, 65],
   893→    1,
   894→    OVER35_HYBRID_THRESHOLDS.modelO35Prob.floor
   895→  );
   896→
   897→  const o35BttsProb = findOptimalProbThreshold(
   898→    m => m.predictedBtts,
   899→    m => (m.ftHomeGoals + m.ftAwayGoals) > 3.5,
   900→    [35, 80],
   901→    1,
   902→    OVER35_HYBRID_THRESHOLDS.bttsProb.floor
   903→  );
   904→
   905→  // --- STRONG BET thresholds (derive O2.5, O3.5, BTTS prob thresholds) ---
   906→  const sbO25Prob = findOptimalProbThreshold(
   907→    m => m.predictedO25,
   908→    m => m.ftHomeGoals + m.ftAwayGoals >= 3,
   909→    [50, 85],
   910→    1,
   911→    STRONG_BET_HYBRID.o25Prob.floor
   912→  );
   913→
   914→  const sbO35Prob = findOptimalProbThreshold(
   915→    m => m.predictedO35,
   916→    m => m.ftHomeGoals + m.ftAwayGoals >= 3,
   917→    [25, 60],
   918→    1,
   919→    STRONG_BET_HYBRID.o35Prob.floor
   920→  );
   921→
   922→  const sbBttsProb = findOptimalProbThreshold(
   923→    m => m.predictedBtts,
   924→    m => m.ftHomeGoals + m.ftAwayGoals >= 3,
   925→    [40, 80],
   926→    1,
   927→    STRONG_BET_HYBRID.bttsProb.floor
   928→  );
   929→
   930→  // --- GREY RESULT thresholds ---
   931→  const grBttsProb = findOptimalProbThreshold(
   932→    m => m.predictedBtts,
   933→    m => m.ftHomeGoals >= 1 && m.ftAwayGoals >= 1 && m.ftHomeGoals + m.ftAwayGoals >= 4,
   934→    [40, 80],
   935→    1,
   936→    GREY_RESULT_CONFIG.bttsProb.min
   937→  );
   938→
   939→  const grO25Prob = findOptimalProbThreshold(
   940→    m => m.predictedO25,
   941→    m => m.ftHomeGoals >= 1 && m.ftAwayGoals >= 1 && m.ftHomeGoals + m.ftAwayGoals >= 4,
   942→    [50, 85],
   943→    1,
   944→    GREY_RESULT_CONFIG.o25Prob.floor
   945→  );
   946→
   947→  const grO35Prob = findOptimalProbThreshold(
   948→    m => m.predictedO35,
   949→    m => m.ftHomeGoals >= 1 && m.ftAwayGoals >= 1 && m.ftHomeGoals + m.ftAwayGoals >= 4,
   950→    [20, 60],
   951→    1,
   952→    GREY_RESULT_CONFIG.o35Prob.floor
   953→  );
   954→
   955→  return {
   956→    leagueName,
   957→    btts: {
   958→      modelBttsProb: bttsModelProb,
   959→      modelO25Prob: bttsO25Prob,
   960→    },
   961→    over35: {
   962→      modelO35Prob: o35ModelProb,
   963→      bttsProb: o35BttsProb,
   964→    },
   965→    strongBet: {
   966→      o25Prob: sbO25Prob,
   967→      o35Prob: sbO35Prob,
   968→      bttsProb: sbBttsProb,
   969→      bttsChecklistCount: STRONG_BET_HYBRID.bttsChecklistCount,
   970→    },
   971→    greyResult: {
   972→      bttsProb: grBttsProb,
   973→      bttsProbMax: GREY_RESULT_CONFIG.bttsProb.max,
   974→      o25Prob: grO25Prob,
   975→      o35Prob: grO35Prob,
   976→      bttsChecklistCount: GREY_RESULT_CONFIG.bttsChecklistCount,
   977→      over35ChecklistCount: GREY_RESULT_CONFIG.over35ChecklistCount,
   978→      requiredChecks: GREY_RESULT_CONFIG.requiredChecks,
   979→    },
   980→    sampleSize: matches.length,
   981→    derivedAt: new Date().toISOString(),
   982→  };
   983→}

   985→/**
   986→ * Derive thresholds from backtest data using a simpler approach when model
   987→ * predictions are not available — uses league averages as the predictor.
   988→ *
   989→ * Lean version: only derives the 2-field BTTS + 2-field O3.5 + StrongBet +
   990→ * GreyResult thresholds that match the current interface. All league-level
   991→ * constants (avg goals, O2.5 rate) and season-average goals/shot conversion
   992→ * checks were removed as noise.
   993→ */
   994→export function deriveSimpleThresholdsFromBacktest(
   995→  leagueName: string,
   996→  matches: { ftHomeGoals: number; ftAwayGoals: number }[],
   997→  currentBaselines: LeagueBaselines,
   998→  options: BacktestDerivationOptions = {}
   999→): LeagueBacktestThresholds | null {
  1000→  const { minSampleSize = 150 } = options;
  1001→
  1002→  if (matches.length < minSampleSize) return null;
  1003→
  1004→  // For simple derivation without per-match predictions, use the hybrid formula
  1005→  // with default multipliers, matching the lean 2-field interfaces.
  1006→  return {
  1007→    leagueName,
  1008→    btts: {
  1009→      modelBttsProb: Math.max(BTTS_HYBRID_THRESHOLDS.modelBttsProb.floor, currentBaselines.bttsRate * BTTS_HYBRID_THRESHOLDS.modelBttsProb.multiplier),
  1010→      modelO25Prob: Math.max(BTTS_HYBRID_THRESHOLDS.modelO25Prob.floor, currentBaselines.over25Rate * BTTS_HYBRID_THRESHOLDS.modelO25Prob.multiplier),
  1011→    },
  1012→    over35: {
  1013→      modelO35Prob: Math.max(OVER35_HYBRID_THRESHOLDS.modelO35Prob.floor, currentBaselines.over35Rate * OVER35_HYBRID_THRESHOLDS.modelO35Prob.multiplier),
  1014→      bttsProb: Math.max(OVER35_HYBRID_THRESHOLDS.bttsProb.floor, currentBaselines.bttsRate * OVER35_HYBRID_THRESHOLDS.bttsProb.multiplier),
  1015→    },
  1016→    strongBet: {
  1017→      o25Prob: Math.max(STRONG_BET_HYBRID.o25Prob.floor, currentBaselines.over25Rate * STRONG_BET_HYBRID.o25Prob.multiplier),
  1018→      o35Prob: Math.max(STRONG_BET_HYBRID.o35Prob.floor, currentBaselines.over35Rate * STRONG_BET_HYBRID.o35Prob.multiplier),
  1019→      bttsProb: Math.max(STRONG_BET_HYBRID.bttsProb.floor, currentBaselines.bttsRate * STRONG_BET_HYBRID.bttsProb.multiplier),
  1020→      bttsChecklistCount: STRONG_BET_HYBRID.bttsChecklistCount,
  1021→    },
  1022→    greyResult: {
  1023→      bttsProb: GREY_RESULT_CONFIG.bttsProb.min,
  1024→      bttsProbMax: GREY_RESULT_CONFIG.bttsProb.max,
  1025→      o25Prob: Math.max(GREY_RESULT_CONFIG.o25Prob.floor, currentBaselines.over25Rate * GREY_RESULT_CONFIG.o25Prob.multiplier),
  1026→      o35Prob: Math.max(GREY_RESULT_CONFIG.o35Prob.floor, currentBaselines.over35Rate * GREY_RESULT_CONFIG.o35Prob.multiplier),
  1027→      bttsChecklistCount: GREY_RESULT_CONFIG.bttsChecklistCount,
  1028→      over35ChecklistCount: GREY_RESULT_CONFIG.over35ChecklistCount,
  1029→      requiredChecks: GREY_RESULT_CONFIG.requiredChecks,
  1030→    },
  1031→    sampleSize: matches.length,
  1032→    derivedAt: new Date().toISOString(),
  1033→  };
  1034→}
  1035→
  1036→// ============================================================================
  1037→// Bookie Odds Dampener — Heavy Favorite Adjustment
  1038→// ============================================================================
  1039→// Based on live results analysis: when bookie odds for one side are <=1.70,
  1040→// the model systematically overestimates O2.5 probability by 10-15%.
  1041→// Heavy favorites tend to win tight, controlling games (1-0, 2-0) rather
  1042→// than high-scoring shootouts. This dampener accounts for that pattern.
  1043→// ============================================================================
  1044→
  1045→/** Heavy favorite odds threshold (decimal odds) */
  1046→const HEAVY_FAVORITE_ODDS = 1.70;
  1047→
  1048→/** Maximum dampener amount (percentage points to subtract) */
  1049→const HEAVY_FAVORITE_DAMPENER = 10;
  1050→
  1051→/**
  1052→ * Apply bookie odds dampener to goal market probabilities.
  1053→ * When either team is a heavy favorite (avg odds <=1.70), reduce O2.5/O3.5/BTTS
  1054→ * probabilities to account for the "park the bus" effect.
  1055→ *
  1056→ * Only applies when valid odds are available (non-null, positive).
  1057→ *
  1058→ * @param over25Prob  - Model O2.5 probability (%)
  1059→ * @param over35Prob  - Model O3.5 probability (%)
  1060→ * @param bttsProb    - Model BTTS probability (%)
  1061→ * @param homeAvgOdds - Home team average odds (decimal)
  1062→ * @param awayAvgOdds - Away team average odds (decimal)
  1063→ * @returns Adjusted probabilities with dampener applied
  1064→ */
  1065→export function applyBookieOddsDampener(
  1066→  over25Prob: number,
  1067→  over35Prob: number,
  1068→  bttsProb: number,
  1069→  homeAvgOdds: number | null | undefined,
  1070→  awayAvgOdds: number | null | undefined,
  1071→  leagueAvgGoalsPerGame?: number
  1072→): { over25: number; over35: number; btts: number; dampened: boolean; reason: string } {
  1073→  if (!homeAvgOdds || !awayAvgOdds || homeAvgOdds <= 0 || awayAvgOdds <= 0) {
  1074→    return { over25: over25Prob, over35: over35Prob, btts: bttsProb, dampened: false, reason: '' };
  1075→  }
  1076→
  1077→  // League-adaptive: gentler dampener in high-variance leagues
  1078→  const variance = leagueAvgGoalsPerGame ? classifyLeagueVariance(leagueAvgGoalsPerGame) : 'low';
  1079→  const heavyThreshold = variance === 'high' ? 1.40 : variance === 'medium' ? 1.60 : HEAVY_FAVORITE_ODDS;
  1080→  const maxDampener = variance === 'high' ? 5 : variance === 'medium' ? 7 : HEAVY_FAVORITE_DAMPENER;
  1081→
  1082→  const homeIsHeavy = homeAvgOdds <= heavyThreshold;
  1083→  const awayIsHeavy = awayAvgOdds <= heavyThreshold;
  1084→
  1085→  if (!homeIsHeavy && !awayIsHeavy) {
  1086→    return { over25: over25Prob, over35: over35Prob, btts: bttsProb, dampened: false, reason: '' };
  1087→  }
  1088→
  1089→  const heavyOdds = Math.min(homeAvgOdds, awayAvgOdds);
  1090→  // Scale dampener: heavier favorite = bigger reduction
  1091→  // At 1.17 (PSG level): full dampener
  1092→  // At threshold: minimal 2% dampener
  1093→  const minDampener = variance === 'high' ? 2 : 3;
  1094→  const scale = Math.max(0, 1 - (heavyOdds - 1.17) / (heavyThreshold - 1.17));
  1095→  const dampener = Math.round(minDampener + (maxDampener - minDampener) * scale);
  1096→
  1097→  const leagueLabel = variance.charAt(0).toUpperCase() + variance.slice(1);
  1098→  const reason = homeIsHeavy
  1099→    ? `${leagueLabel}-variance: Heavy home favorite (${homeAvgOdds.toFixed(2)}): -${dampener}% dampener`
  1100→    : `${leagueLabel}-variance: Heavy away favorite (${awayAvgOdds.toFixed(2)}): -${dampener}% dampener`;
  1101→
  1102→  return {
  1103→    over25: Math.max(30, over25Prob - dampener),
  1104→    over35: Math.max(20, over35Prob - dampener),
  1105→    btts: Math.max(25, bttsProb - Math.round(dampener * 0.5)), // BTTS dampened less
  1106→    dampened: true,
  1107→    reason,
  1108→  };
  1109→}

  1111→/**
  1112→ * Compute average bookie odds for a team from match results.
  1113→ * Averages odds across all matches for that team.
  1114→ *
  1115→ * @param matches - Match results with odds data
  1116→ * @param team    - Team name
  1117→ * @returns Average decimal odds, or null if insufficient data
  1118→ */
  1119→export function computeTeamAvgOdds(
  1120→  matches: { homeTeam: string; awayTeam: string; oddsAvgHome: number | null; oddsAvgAway: number | null }[],
  1121→  team: string
  1122→): number | null {
  1123→  const teamOdds: number[] = [];
  1124→
  1125→  for (const m of matches) {
  1126→    const odds = m.homeTeam === team ? m.oddsAvgHome : (m.awayTeam === team ? m.oddsAvgAway : null);
  1127→    if (odds && odds > 1 && odds < 20) {
  1128→      teamOdds.push(odds);
  1129→    }
  1130→  }
  1131→
  1132→  if (teamOdds.length < 3) return null;
  1133→
  1134→  return teamOdds.reduce((s, o) => s + o, 0) / teamOdds.length;
  1135→}
  1136→
  1137→// ============================================================================
  1138→// League Variance Classification — Used by blowout/dampener adaptivity
  1139→// ============================================================================
  1140→// Classifies a league by its historical goal variance so that blowout and
  1141→// dampener thresholds can be calibrated to the league's natural scoring profile.
  1142→// High-variance leagues (2. Bundesliga, Eredivisie) see more goal explosions
  1143→// from heavy favorites, so suppression mechanisms should be gentler.
  1144→
  1145→export type LeagueVariance = 'high' | 'medium' | 'low';
  1146→
  1147→export function classifyLeagueVariance(avgGoalsPerGame: number): LeagueVariance {
  1148→  if (avgGoalsPerGame >= 3.0) return 'high';
  1149→  if (avgGoalsPerGame >= 2.5) return 'medium';
  1150→  return 'low';
  1151→}
  1152→
  1153→// ============================================================================
  1154→// Blowout Risk Qualifier — Override Under signals in massive quality gaps
  1155→// ============================================================================
  1156→// Based on live results analysis: when there's a huge quality gap (favorite's
  1157→// avg odds <= 1.50), the Regression "Under" signal is unreliable. Heavy favorites
  1158→// tend to dominate regardless of recent form — a "cooling off" period doesn't
  1159→// matter when you're 2-3 quality tiers above the opponent.
  1160→//
  1161→// This qualifier prevents the "Under" regression signal from canceling out
  1162→// legitimate "Over" signals (xG, Z-Score) in STRONG BET / GREY RESULT / GOAL FEST.
  1163→//
  1164→// Example: Man City (odds 1.20) vs Luton — Regression says "Under" (City only
  1165→// scored 1 goal last 3 games), but City will likely score 3-4 regardless.
  1166→// Without this qualifier, the Under signal would block 2 points in STRONG BET.
  1167→// ============================================================================
  1168→
  1169→/** Blowout threshold: favorite's average odds at or below this triggers override */
  1170→const BLOWOUT_ODDS_THRESHOLD = 1.50;
  1171→
  1172→/**
  1173→ * Apply blowout risk qualifier to regression signal.
  1174→ * When either team is a massive favorite (avg odds <= 1.50), override
  1175→ * Regression "Under"/"Strong Under" to "Neutral" because quality advantage
  1176→ * outweighs recent form in one-sided matchups.
  1177→ *
  1178→ * @param signals      - Current signal set (xgSignal, regressionSignal, zScoreSignal)
  1179→ * @param homeAvgOdds  - Home team average odds (decimal)
  1180→ * @param awayAvgOdds  - Away team average odds (decimal)
  1181→ * @returns Modified signals (regression may be overridden to Neutral)
  1182→ */
  1183→export function applyBlowoutQualifier(
  1184→  signals: SignalInput,
  1185→  homeAvgOdds: number | null | undefined,
  1186→  awayAvgOdds: number | null | undefined,
  1187→  leagueAvgGoalsPerGame?: number
  1188→): { signals: SignalInput; blowoutOverride: boolean; reason: string } {
  1189→  // Only applies when valid odds are available
  1190→  if (!homeAvgOdds || !awayAvgOdds || homeAvgOdds <= 0 || awayAvgOdds <= 0) {
  1191→    return { signals, blowoutOverride: false, reason: '' };
  1192→  }
  1193→
  1194→  // League-adaptive threshold: gentler in high-variance leagues
  1195→  const variance = leagueAvgGoalsPerGame ? classifyLeagueVariance(leagueAvgGoalsPerGame) : 'low';
  1196→  const blowoutThreshold = variance === 'high' ? 1.25 : variance === 'medium' ? 1.40 : BLOWOUT_ODDS_THRESHOLD;
  1197→
  1198→  const homeIsMassiveFavorite = homeAvgOdds <= blowoutThreshold;
  1199→  const awayIsMassiveFavorite = awayAvgOdds <= blowoutThreshold;
  1200→
  1201→  // Only trigger when one team is a massive favorite
  1202→  if (!homeIsMassiveFavorite && !awayIsMassiveFavorite) {
  1203→    return { signals, blowoutOverride: false, reason: '' };
  1204→  }
  1205→
  1206→  // In high-variance leagues, only override Strong Under (keep mild signals intact)
  1207→  // because hot favorites in goal-rich leagues often score freely
  1208→  const shouldOverride = variance === 'high'
  1209→    ? signals.regressionSignal === 'Strong Under'
  1210→    : signals.regressionSignal === 'Under' || signals.regressionSignal === 'Strong Under';
  1211→  if (!shouldOverride) {
  1212→    return { signals, blowoutOverride: false, reason: '' };
  1213→  }
  1214→
  1215→  const favoriteOdds = Math.min(homeAvgOdds, awayAvgOdds);
  1216→  const favorite = homeIsMassiveFavorite ? 'Home' : 'Away';
  1217→  const reason = `Blowout qualifier (${variance}-variance): ${favorite} favorite (${favoriteOdds.toFixed(2)} <= ${blowoutThreshold}) — Regression "${signals.regressionSignal}" overridden to Neutral`;
  1218→
  1219→  return {
  1220→    signals: {
  1221→      ...signals,
  1222→      regressionSignal: 'Neutral',
  1223→    },
  1224→    blowoutOverride: true,
  1225→    reason,
  1226→  };
  1227→}

  1234→// but tends to surface ONE-SIDED regressions (one team due for a breakout).
  1235→// This system evaluates each team independently and produces a BTTS-specific

  1282→  if (combinedSignal <= -0.3) return 'Over';
  1283→  if (combinedSignal >= 0.8) return 'Strong Under';

  1294→  if (xgDiff <= -0.5) return 'Over';
  1295→  if (xgDiff >= 1.0) return 'Strong Under';

  1301→ * Classify a per-team Z-Score into a signal label.

  1305→  if (zScore <= -1.5) return 'Strong Over';
  1306→  if (zScore <= -1.0) return 'Over';
  1307→  if (zScore >= 1.5) return 'Strong Under';
  1308→  if (zScore >= 1.0) return 'Under';

  1312→// ============================================================================
  1313→// Momentum Signal — Parallel track to Regression (form continuation)
  1314→// ============================================================================
  1315→// While Regression asks "is this team due to bounce back?", Momentum asks
  1316→// "is this team on a sustained hot streak that will continue?"
  1317→// Both tracks run simultaneously; a match can qualify through either pathway.
  1318→
  1319→export interface MomentumTeamInput {
  1320→  last5Goals: number[];    // total goals per match in last 5
  1321→  seasonAvg: number;       // season avg total goals per game
  1322→  xgDiff: number;          // actualGoals/matches - totalXg/matches (positive = overperforming)
  1323→  sotPerGame: number | null;
  1324→}
  1325→
  1326→export interface MomentumResult {
  1327→  homeSignal: string;      // 'Strong Momentum' | 'Momentum' | 'Neutral' | 'Cooling' | 'Cold'
  1328→  awaySignal: string;
  1329→  matchSignal: string;     // 'MOMENTUM OVER' | 'MOMENTUM LEAN OVER' | 'MOMENTUM NEUTRAL' | 'MOMENTUM LEAN UNDER' | 'MOMENTUM UNDER'
  1330→}
  1331→
  1332→function classifyTeamMomentum(team: MomentumTeamInput): string {
  1333→  if (team.last5Goals.length < 3) return 'Neutral';
  1334→
  1335→  const last5Avg = team.last5Goals.reduce((s, g) => s + g, 0) / team.last5Goals.length;
  1336→  const last3 = team.last5Goals.slice(0, Math.min(3, team.last5Goals.length));
  1337→  const last3Avg = last3.reduce((s, g) => s + g, 0) / last3.length;
  1338→
  1339→  // Consistency: coefficient of variation of last 5
  1340→  const mean = last5Avg;
  1341→  const variance = team.last5Goals.reduce((s, g) => s + (g - mean) ** 2, 0) / team.last5Goals.length;
  1342→  const stdDev = Math.sqrt(variance);
  1343→  const cv = mean > 0 ? stdDev / mean : 999;
  1344→
  1345→  // Sustained: 5+ game window above season average by meaningful margin
  1346→  const sustained = last5Avg > team.seasonAvg * 1.15;
  1347→
  1348→  // Accelerating: last 3 even hotter than last 5
  1349→  const accelerating = last3Avg > last5Avg * 1.05;
  1350→
  1351→  // xG sustainability: overperforming xG with shot volume to back it
  1352→  const xgSustainable = team.xgDiff > 0 && team.sotPerGame !== null && team.sotPerGame > 5.0;
  1353→
  1354→  // Strong Momentum: sustained + consistent + high output
  1355→  if (sustained && cv < 0.5 && last5Avg >= 2.0) return 'Strong Momentum';
  1356→  // Momentum: sustained with moderate variance, or recent surge with sustainable backing
  1357→  if (sustained && cv < 0.65) return 'Momentum';
  1358→  if (last5Avg > team.seasonAvg * 1.15 && accelerating && (xgSustainable || team.xgDiff <= 0)) return 'Momentum';
  1359→  // Cold: declining output
  1360→  if (last5Avg < team.seasonAvg * 0.75) {
  1361→    return last3Avg < last5Avg * 0.8 ? 'Cold' : 'Cooling';
  1362→  }
  1363→  return 'Neutral';
  1364→}

  1366→export function computeMomentumSignal(home: MomentumTeamInput, away: MomentumTeamInput): MomentumResult {
  1367→  const homeSignal = classifyTeamMomentum(home);
  1368→  const awaySignal = classifyTeamMomentum(away);
  1369→
  1370→  const isOver = (s: string) => s === 'Strong Momentum' || s === 'Momentum';
  1371→  const isUnder = (s: string) => s === 'Cold' || s === 'Cooling';
  1372→
  1373→  let matchSignal: string;
  1374→  if (isOver(homeSignal) && isOver(awaySignal)) matchSignal = 'MOMENTUM OVER';
  1375→  else if ((isOver(homeSignal) || isOver(awaySignal)) && !isUnder(homeSignal) && !isUnder(awaySignal)) matchSignal = 'MOMENTUM LEAN OVER';
  1376→  else if (isUnder(homeSignal) && isUnder(awaySignal)) matchSignal = 'MOMENTUM UNDER';
  1377→  else if (isUnder(homeSignal) || isUnder(awaySignal)) matchSignal = 'MOMENTUM LEAN UNDER';
  1378→  else matchSignal = 'MOMENTUM NEUTRAL';
  1379→
  1380→  return { homeSignal, awaySignal, matchSignal };
  1381→}

  1383→// ============================================================================
  1384→// BTTS Dual-Team Qualification System
  1385→// ============================================================================
  1386→// Evaluates each team independently to determine whether BOTH teams are likely
  1387→// to score. Returns a tier that caps BTTS probability independently from the
  1388→// Over goal thesis.
  1389→//
  1390→// Now supports TWO qualification pathways:
  1391→//   1. Regression Under: both teams underperforming → due to bounce back
  1392→//   2. Momentum Over: both teams on hot streaks → likely to keep scoring
  1393→//
  1394→// Tier mapping:
  1395→//   5+ → BTTS STRONG   (cap 70%) — both teams individually qualify
  1396→//   3–4 → BTTS QUALIFIED (cap 55%) — mild BTTS confidence
  1397→//   1–2 → BTTS WEAK     (cap 40%) — one-sided regression likely
  1398→//   ≤0  → BTTS AVOID    (cap 25%) — do not bet BTTS
  1399→// ============================================================================
  1400→export function computeBTTSQualification(input: BttsQualificationInput): BttsQualificationResult {
  1401→  let score = 0;
  1402→  const checks: { check: string; points: number; passed: boolean }[] = [];
  1403→
  1404→  const isRegressionUnder = (sig: string) => sig === 'Under' || sig === 'Strong Under';
  1405→  const isXgOver = (diff: number) => diff <= -0.5; // underperforming xG = goals due
  1406→  const isZCold = (z: number) => z >= 1.0; // overperforming Z = cold streak coming
  1407→  const isZNeutralOrBetter = (z: number) => z < 1.0; // not overheated
  1408→
  1409→  // Momentum helpers
  1410→  const isMomentum = (s: string | undefined) => s === 'Strong Momentum' || s === 'Momentum';
  1411→  const homeMom = isMomentum(input.homeMomentumSignal);
  1412→  const awayMom = isMomentum(input.awayMomentumSignal);
  1413→  const bothMom = homeMom && awayMom;
  1414→  const oneMom = homeMom || awayMom;
  1415→
  1416→  // Check 1: Regression Under OR Momentum Over — dual qualification pathway
  1417→  const bothRegUnder = isRegressionUnder(input.homeRegressionSignal) && isRegressionUnder(input.awayRegressionSignal);
  1418→  const oneRegUnder = isRegressionUnder(input.homeRegressionSignal) || isRegressionUnder(input.awayRegressionSignal);
  1419→
  1420→  if (bothRegUnder && bothMom) {
  1421→    // Supercharged: both signals agree from both sides
  1422→    score += 5;
  1423→    checks.push({ check: 'Both Regression Under AND Momentum Over (supercharged)', points: 5, passed: true });
  1424→  } else if (bothRegUnder || bothMom) {
  1425→    score += 3;
  1426→    checks.push({ check: bothRegUnder ? 'Both teams Regression Under/StrongUnder' : 'Both teams Momentum Over (hot streak)', points: 3, passed: true });
  1427→  } else if ((oneRegUnder && oneMom)) {
  1428→    score += 2; // Mixed but both signal types point to goals
  1429→    checks.push({ check: 'One Regression Under + One Momentum Over', points: 2, passed: true });
  1430→  } else if (oneRegUnder || oneMom) {
  1431→    score += 1;
  1432→    checks.push({ check: oneRegUnder ? 'One team Regression Under' : 'One team Momentum Over', points: 1, passed: true });
  1433→  } else {
  1434→    checks.push({ check: 'No Regression Under or Momentum Over', points: 3, passed: false });
  1435→  }
  1436→
  1437→  // Check 2: BOTH teams xG Over (underperforming xG = creating chances, finishing due)
  1438→  const bothXgOver = isXgOver(input.homeXgDiff) && isXgOver(input.awayXgDiff);
  1439→  if (bothXgOver) {
  1440→    score += 2;
  1441→    checks.push({ check: 'Both teams xG Over (underperforming)', points: 2, passed: true });
  1442→  } else {
  1443→    checks.push({ check: 'Both teams xG Over (underperforming)', points: 2, passed: false });
  1444→  }
  1445→
  1446→  // Check 3: BOTH teams Z-Score not overheated
  1447→  const bothZOk = isZNeutralOrBetter(input.homeZScore) && isZNeutralOrBetter(input.awayZScore);
  1448→  if (bothZOk) {
  1449→    score += 1;
  1450→    checks.push({ check: 'Both teams Z-Score Neutral/better', points: 1, passed: true });
  1451→  } else {
  1452→    checks.push({ check: 'Both teams Z-Score Neutral/better', points: 1, passed: false });
  1453→  }
  1454→
  1455→  // Check 4: Competitive matchup (favorite odds 1.80–2.50)
  1456→  if (input.favoriteOdds && input.favoriteOdds >= 1.80 && input.favoriteOdds <= 2.50) {
  1457→    score += 1;
  1458→    checks.push({ check: 'Competitive matchup (odds 1.80-2.50)', points: 1, passed: true });
  1459→  } else {
  1460→    checks.push({ check: 'Competitive matchup (odds 1.80-2.50)', points: 1, passed: false });
  1461→  }
  1462→
  1463→  // Penalty 5: Massive favorite (≤1.50) — one side dominates, BTTS unlikely
  1464→  if (input.favoriteOdds && input.favoriteOdds <= 1.50) {
  1465→    score -= 2;
  1466→    checks.push({ check: 'Massive favorite (odds ≤1.50)', points: -2, passed: true });
  1467→  } else {
  1468→    checks.push({ check: 'Massive favorite (odds ≤1.50)', points: -2, passed: false });
  1469→  }
  1470→
  1471→  // Penalty 6: Either team can't finish (SOT conversion <25%)
  1472→  const homeCantFinish = input.homeSotConversion !== null && input.homeSotConversion < 25;
  1473→  const awayCantFinish = input.awaySotConversion !== null && input.awaySotConversion < 25;
  1474→  if (homeCantFinish) score -= 1;
  1475→  if (awayCantFinish) score -= 1;
  1476→  checks.push({
  1477→    check: `Home SOT conversion ${homeCantFinish ? '< 25%' : '≥ 25%'}${input.homeSotConversion !== null ? ` (${input.homeSotConversion.toFixed(0)}%)` : ' (N/A)'}`,
  1478→    points: -1,
  1479→    passed: !homeCantFinish,
  1480→  });
  1481→  checks.push({
  1482→    check: `Away SOT conversion ${awayCantFinish ? '< 25%' : '≥ 25%'}${input.awaySotConversion !== null ? ` (${input.awaySotConversion.toFixed(0)}%)` : ' (N/A)'}`,
  1483→    points: -1,
  1484→    passed: !awayCantFinish,
  1485→  });
  1486→
  1487→  // Penalty 7: Either team in a strong cold Z-Score streak
  1488→  const homeCold = isZCold(input.homeZScore);
  1489→  const awayCold = isZCold(input.awayZScore);
  1490→  if (homeCold) score -= 2;
  1491→  if (awayCold) score -= 2;
  1492→  checks.push({
  1493→    check: `Home Z-Score ${homeCold ? 'cold (≥1.0)' : 'ok'}`,
  1494→    points: -2,
  1495→    passed: !homeCold,
  1496→  });
  1497→  checks.push({
  1498→    check: `Away Z-Score ${awayCold ? 'cold (≥1.0)' : 'ok'}`,
  1499→    points: -2,
  1500→    passed: !awayCold,
  1501→  });
  1502→
  1503→  // Determine tier
  1504→  let tier: BttsQualificationTier;
  1505→  let bttsCap: number;
  1506→  if (score >= 5) {
  1507→    tier = 'BTTS STRONG';
  1508→    bttsCap = 70;
  1509→  } else if (score >= 3) {
  1510→    tier = 'BTTS QUALIFIED';
  1511→    bttsCap = 55;
  1512→  } else if (score >= 1) {
  1513→    tier = 'BTTS WEAK';
  1514→    bttsCap = 40;
  1515→  } else {
  1516→    tier = 'BTTS AVOID';
  1517→    bttsCap = 25;
  1518→  }
  1519→
  1520→  return { tier, score, breakdown: checks, bttsCap };
  1521→}

  1567→/**
  1568→ * Compute Third Goal qualifier — distinguishes "2-goal games" from "3+ goal games"
  1569→ * within pre-filtered selections.
  1570→ *
  1571→ * Tier mapping:
  1572→ *   6+  → GOAL RICH      — Full confidence on Over 2.5, consider Over 3.5
  1573→ *   4–5 → GOAL LIKELY    — Solid Over 2.5 bet
  1574→ *   2–3 → GOAL BORDERLINE — Over 1.5 strong, Over 2.5 cautious
  1575→ *   0–1 → GOAL THIN      — Over 1.5 only, skip Over 2.5
  1576→ *   <0  → GOAL STALL     — Likely 1-2 goals, avoid Over markets above 1.5
  1577→ */
  1578→export function computeThirdGoalQualifier(input: ThirdGoalQualifierInput): ThirdGoalQualifierResult {
  1579→  let score = 0;
  1580→  const checks: { check: string; points: number; passed: boolean }[] = [];
  1581→
  1582→  const isRegressionUnder = (sig: string) => sig === 'Under' || sig === 'Strong Under';
  1583→  const isXgOver = (sig: string) => sig === 'Over' || sig === 'Strong Over';
  1584→
  1585→  // Check 1: Combined xG > 3.0 (very high chance creation)
  1586→  if (input.combinedXgTotal > 3.0) {
  1587→    score += 3;
  1588→    checks.push({ check: `Combined xG ${input.combinedXgTotal.toFixed(2)} > 3.0`, points: 3, passed: true });
  1589→  } else if (input.combinedXgTotal >= 2.5) {
  1590→    score += 1;
  1591→    checks.push({ check: `Combined xG ${input.combinedXgTotal.toFixed(2)} in 2.5-3.0`, points: 1, passed: true });
  1592→  } else {
  1593→    checks.push({ check: `Combined xG ${input.combinedXgTotal.toFixed(2)} < 2.5`, points: 3, passed: false });
  1594→  }
  1595→
  1596→  // Check 2: BOTH teams Regression Under (double regression)
  1597→  const bothRegUnder = isRegressionUnder(input.homeRegressionSignal) && isRegressionUnder(input.awayRegressionSignal);
  1598→  if (bothRegUnder) {
  1599→    score += 2;
  1600→    checks.push({ check: 'Both teams Regression Under', points: 2, passed: true });
  1601→  } else {
  1602→    checks.push({ check: 'Both teams Regression Under', points: 2, passed: false });
  1603→  }
  1604→
  1605→  // Check 3: BOTH teams xG Over (both creating chances)
  1606→  const bothXgOver = isXgOver(input.homeXgSignal) && isXgOver(input.awayXgSignal);
  1607→  if (bothXgOver) {
  1608→    score += 2;
  1609→    checks.push({ check: 'Both teams xG Over', points: 2, passed: true });
  1610→  } else {
  1611→    checks.push({ check: 'Both teams xG Over', points: 2, passed: false });
  1612→  }
  1613→
  1614→  // Check 4: BOTH teams average SOT > 5.5 (sustained pressure)
  1615→  const homeHighSot = input.homeAvgSot !== null && input.homeAvgSot > 5.5;
  1616→  const awayHighSot = input.awayAvgSot !== null && input.awayAvgSot > 5.5;
  1617→  if (homeHighSot && awayHighSot) {
  1618→    score += 2;
  1619→    checks.push({ check: `Both teams SOT > 5.5 (${input.homeAvgSot?.toFixed(1)} / ${input.awayAvgSot?.toFixed(1)})`, points: 2, passed: true });
  1620→  } else if (homeHighSot || awayHighSot) {
  1621→    checks.push({ check: `One team SOT > 5.5 (${input.homeAvgSot?.toFixed(1) ?? 'N/A'} / ${input.awayAvgSot?.toFixed(1) ?? 'N/A'})`, points: 2, passed: false });
  1622→  } else {
  1623→    checks.push({ check: `Neither team SOT > 5.5 (${input.homeAvgSot?.toFixed(1) ?? 'N/A'} / ${input.awayAvgSot?.toFixed(1) ?? 'N/A'})`, points: 2, passed: false });
  1624→  }
  1625→
  1626→  // Check 5: Balanced xG (within 0.8 of each other) — competitive game
  1627→  // We check if xG signals are similar (not one-sided)
  1628→  const xgBalanced = isXgOver(input.homeXgSignal) === isXgOver(input.awayXgSignal);
  1629→  if (xgBalanced) {
  1630→    score += 1;
  1631→    checks.push({ check: 'xG signals balanced (both teams similar)', points: 1, passed: true });
  1632→  } else {
  1633→    checks.push({ check: 'xG signals lopsided (one-sided)', points: 1, passed: false });
  1634→  }
  1635→
  1636→  // Check 6: Favorite odds 1.80–2.50 (open competitive game)
  1637→  if (input.favoriteOdds && input.favoriteOdds >= 1.80 && input.favoriteOdds <= 2.50) {
  1638→    score += 1;
  1639→    checks.push({ check: 'Competitive matchup (odds 1.80-2.50)', points: 1, passed: true });
  1640→  } else {
  1641→    checks.push({ check: 'Not a competitive matchup', points: 1, passed: false });
  1642→  }
  1643→
  1644→  // Penalty 7: Lopsided xG (one team xG Over, other not)
  1645→  if (!xgBalanced) {
  1646→    score -= 1;
  1647→    // Already counted in check 5 as non-passed, no duplicate
  1648→  }
  1649→
  1650→  // Penalty 8: Massive favorite (≤1.50) — might be controlled 2-0
  1651→  if (input.favoriteOdds && input.favoriteOdds <= 1.50) {
  1652→    score -= 1;
  1653→    checks.push({ check: 'Massive favorite (odds ≤1.50) — controlled game risk', points: -1, passed: true });
  1654→  } else {
  1655→    checks.push({ check: 'Massive favorite risk', points: -1, passed: false });
  1656→  }
  1657→
  1658→  // Penalty 9: Either team can't finish (SOT conversion <25%)
  1659→  const homeCantFinish = input.homeSotConversion !== null && input.homeSotConversion < 25;
  1660→  const awayCantFinish = input.awaySotConversion !== null && input.awaySotConversion < 25;
  1661→  if (homeCantFinish) score -= 1;
  1662→  if (awayCantFinish) score -= 1;
  1663→  checks.push({
  1664→    check: `Home SOT conversion ${homeCantFinish ? '< 25%' : '≥ 25%'}${input.homeSotConversion !== null ? ` (${input.homeSotConversion.toFixed(0)}%)` : ' (N/A)'}`,
  1665→    points: -1,
  1666→    passed: !homeCantFinish,
  1667→  });
  1668→  checks.push({
  1669→    check: `Away SOT conversion ${awayCantFinish ? '< 25%' : '≥ 25%'}${input.awaySotConversion !== null ? ` (${input.awaySotConversion.toFixed(0)}%)` : ' (N/A)'}`,
  1670→    points: -1,
  1671→    passed: !awayCantFinish,
  1672→  });
  1673→
  1674→  // Determine tier
  1675→  let tier: GoalTier;
  1676→  let recommendation: string;
  1677→  if (score >= 6) {
  1678→    tier = 'GOAL RICH';
  1679→    recommendation = 'Full confidence on Over 2.5, consider Over 3.5';
  1680→  } else if (score >= 4) {
  1681→    tier = 'GOAL LIKELY';
  1682→    recommendation = 'Solid Over 2.5 bet';
  1683→  } else if (score >= 2) {
  1684→    tier = 'GOAL BORDERLINE';
  1685→    recommendation = 'Over 1.5 strong, Over 2.5 cautious';
  1686→  } else if (score >= 0) {
  1687→    tier = 'GOAL THIN';
  1688→    recommendation = 'Over 1.5 only, skip Over 2.5';
  1689→  } else {
  1690→    tier = 'GOAL STALL';
  1691→    recommendation = 'Likely 1-2 goals, avoid Over markets above 1.5';
  1692→  }
  1693→
  1694→  return { tier, score, breakdown: checks, recommendation };
  1695→}

  1729→  let score = 0;
  1730→  const checks: { check: string; points: number; passed: boolean }[] = [];
  1731→
  1732→  // Positive checks
  1733→  const favOddsLow = input.favoriteOdds <= 1.40;

  1785→  return { tier, score, breakdown: checks, over25Rec, bttsRec };
  1786→}

  1832→/**
  1833→ * Compute BTTS Both Halves indicator.
  1834→ *
  1835→ * Calibrated on 24,057 matches across 7 European leagues (10 seasons each).
  1836→ * 1,249 BTTS-BH games (5.19% base rate).
  1837→ *
  1838→ * 3-check system — only signals with proven lift survive:
  1839→ *   - O2.5 implied >= 65%: 1.38x lift (strongest single predictor)
  1840→ *   - Draw prob < 25%:      1.21x lift (low draw = open, balanced play)
  1841→ *   - Rolling scoring >= 3.0: 1.10x lift (recent form, independent)
  1842→ *
  1843→ * Tier system:
  1844→ *   STRONG (3/3):    All signals aligned — rare, highest confidence
  1845→ *   QUALIFIED (2/3): Core checks pass — actionable signal
  1846→ *   BORDERLINE (1/3): One indicator only — watch list
  1847→ *   UNLIKELY (0/3):  Insufficient evidence
  1848→ */
  1849→export function computeBTTSBothHalves(input: BTTSBothHalvesInput): BTTSBothHalvesResult {
  1850→  const cfg = BTTS_BOTH_HALVES_CONFIG;

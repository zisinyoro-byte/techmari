     1→'use client'
     2→
     3→import { useMemo } from 'react'
     4→import { Card, CardContent, CardDescription, CardHeader, CardTitle } from '@/components/ui/card'
     5→import { Input } from '@/components/ui/input'
     6→import { Badge } from '@/components/ui/badge'

    10→import { ChartContainer, ChartTooltip, ChartTooltipContent } from '@/components/ui/chart'

    13→import type { MatchResult, Analytics as AnalyticsType } from '@/lib/types'
    14→import { COLORS, PIE_COLORS, SEASON_NAMES, EUROPEAN_SEASONS } from '@/lib/constants'
    15→import { parseDateSafe } from '@/lib/utils'
    16→
    17→interface OverviewTabProps {
    18→  results: MatchResult[]
    19→  analytics: AnalyticsType | null
    20→  searchTerm: string
    21→  sortConfig: { key: string; direction: 'asc' | 'desc' }
    22→  loading: boolean
    23→  error: string | null
    24→  selectedLeagueName: string
    25→  selectedSeasonName: string
    26→  isAllSeasons: boolean
    27→  setSearchTerm: (term: string) => void
    28→  handleSort: (key: string) => void
    29→  displayLimit: number
    30→  setDisplayLimit: (limit: number | ((prev: number) => number)) => void
    31→}
    32→
    33→export function OverviewTab({
    34→  results,
    35→  analytics,
    36→  searchTerm,
    37→  sortConfig,
    38→  loading,
    39→  error,
    40→  selectedLeagueName,
    41→  selectedSeasonName,
    42→  isAllSeasons,
    43→  setSearchTerm,
    44→  handleSort,
    45→  displayLimit,
    46→  setDisplayLimit,
    47→}: OverviewTabProps) {
    48→  // Filter and sort results
    49→  const filteredResults = useMemo(() => {
    50→    let filtered = results
    51→
    52→    if (searchTerm) {
    53→      const term = searchTerm.toLowerCase()
    54→      filtered = filtered.filter(
    55→        match =>
    56→          match.homeTeam.toLowerCase().includes(term) ||
    57→          match.awayTeam.toLowerCase().includes(term)
    58→      )
    59→    }
    60→
    61→    return [...filtered].sort((a, b) => {
    62→      let aVal: string | number = ''
    63→      let bVal: string | number = ''
    64→
    65→      switch (sortConfig.key) {
    66→        case 'date':
    67→          aVal = parseDateSafe(a.date).getTime()
    68→          bVal = parseDateSafe(b.date).getTime()
    69→          break
    70→        case 'homeTeam':
    71→          aVal = a.homeTeam
    72→          bVal = b.homeTeam
    73→          break
    74→        case 'awayTeam':
    75→          aVal = a.awayTeam
    76→          bVal = b.awayTeam
    77→          break
    78→        case 'ftGoals':
    79→          aVal = a.ftHomeGoals + a.ftAwayGoals
    80→          bVal = b.ftHomeGoals + b.ftAwayGoals
    81→          break
    82→        default:
    83→          return 0
    84→      }
    85→
    86→      if (aVal < bVal) return sortConfig.direction === 'asc' ? -1 : 1
    87→      if (aVal > bVal) return sortConfig.direction === 'asc' ? 1 : -1
    88→      return 0
    89→    })
    90→  }, [results, searchTerm, sortConfig])
    91→
    92→  // Calculate team form based on last 5 matches
    93→  const teamForm = useMemo(() => {
    94→    if (results.length === 0) return new Map<string, { form: ('W' | 'D' | 'L')[]; inForm: boolean; points: number }>()
    95→
    96→    const teamMatches = new Map<string, Array<{ date: string; isHome: boolean; goalsFor: number; goalsAgainst: number; result: 'W' | 'D' | 'L' }>>()
    97→
    98→    const sortedResults = [...results].sort((a, b) => {
    99→      return parseDateSafe(b.date).getTime() - parseDateSafe(a.date).getTime()
   100→    })
   101→
   102→    for (const match of sortedResults) {
   103→      const homeMatches = teamMatches.get(match.homeTeam) || []
   104→      const homeResult = match.ftResult === 'H' ? 'W' : match.ftResult === 'D' ? 'D' : 'L'
   105→      homeMatches.push({
   106→        date: match.date,
   107→        isHome: true,
   108→        goalsFor: match.ftHomeGoals,
   109→        goalsAgainst: match.ftAwayGoals,
   110→        result: homeResult
   111→      })
   112→      teamMatches.set(match.homeTeam, homeMatches)
   113→
   114→      const awayMatches = teamMatches.get(match.awayTeam) || []
   115→      const awayResult = match.ftResult === 'A' ? 'W' : match.ftResult === 'D' ? 'D' : 'L'
   116→      awayMatches.push({
   117→        date: match.date,
   118→        isHome: false,
   119→        goalsFor: match.ftAwayGoals,
   120→        goalsAgainst: match.ftHomeGoals,
   121→        result: awayResult
   122→      })
   123→      teamMatches.set(match.awayTeam, awayMatches)
   124→    }
   125→
   126→    const formMap = new Map<string, { form: ('W' | 'D' | 'L')[]; inForm: boolean; points: number }>()
   127→    teamMatches.forEach((matches, team) => {
   128→      const last5 = matches.slice(0, 5)
   129→      const form = last5.map(m => m.result)
   130→      const points = last5.reduce((acc, m) => acc + (m.result === 'W' ? 3 : m.result === 'D' ? 1 : 0), 0)
   131→      const inForm = points >= 7
   132→      formMap.set(team, { form, inForm, points })
   133→    })
   134→
   135→    return formMap
   136→  }, [results])
   137→
   138→  // Calculate last H2H results for each unique team pair
   139→  const lastH2HResults = useMemo(() => {
   140→    if (results.length === 0) return { score00: [], score10: [], score20: [] }
   141→
   142→    const h2hMap = new Map<string, { homeTeam: string; awayTeam: string; homeGoals: number; awayGoals: number; date: string; season?: string }>()
   143→
   144→    const sortedResults = [...results].sort((a, b) => {
   145→      return parseDateSafe(b.date).getTime() - parseDateSafe(a.date).getTime()
   146→    })
   147→
   148→    for (const match of sortedResults) {
   149→      const pairKey = [match.homeTeam, match.awayTeam].sort().join(' vs ')
   150→
   151→      if (!h2hMap.has(pairKey)) {
   152→        h2hMap.set(pairKey, {
   153→          homeTeam: match.homeTeam,
   154→          awayTeam: match.awayTeam,
   155→          homeGoals: match.ftHomeGoals,
   156→          awayGoals: match.ftAwayGoals,
   157→          date: match.date,
   158→          season: match.season
   159→        })
   160→      }
   161→    }
   162→
   163→    const score00: { teams: string; score: string; date: string; season?: string }[] = []
   164→    const score10: { teams: string; score: string; date: string; winner: string; season?: string }[] = []
   165→    const score20: { teams: string; score: string; date: string; winner: string; season?: string }[] = []
   166→
   167→    h2hMap.forEach((match, pairKey) => {
   168→      const { homeTeam, awayTeam, homeGoals, awayGoals, date, season } = match
   169→      const totalGoals = homeGoals + awayGoals
   170→
   171→      if (homeGoals === 0 && awayGoals === 0) {
   172→        score00.push({ teams: pairKey, score: '0-0', date, season })
   173→      } else if (totalGoals === 1) {
   174→        const winner = homeGoals === 1 ? homeTeam : awayTeam
   175→        score10.push({ teams: pairKey, score: `${homeGoals}-${awayGoals}`, date, winner, season })
   176→      } else if (totalGoals === 2 && (homeGoals === 0 || awayGoals === 0)) {
   177→        const winner = homeGoals === 2 ? homeTeam : awayTeam
   178→        score20.push({ teams: pairKey, score: `${homeGoals}-${awayGoals}`, date, winner, season })
   179→      }
   180→    })
   181→
   182→    const sortByDate = (a: { date: string }, b: { date: string }) => {
   183→      return parseDateSafe(b.date).getTime() - parseDateSafe(a.date).getTime()
   184→    }
   185→
   186→    return {
   187→      score00: score00.sort(sortByDate),
   188→      score10: score10.sort(sortByDate),
   189→      score20: score20.sort(sortByDate)
   190→    }
   191→  }, [results])
   192→
   193→  // Prepare chart data
   194→  const resultDistributionData = analytics ? [
   195→    { name: 'Home Wins', value: analytics.resultDistribution.homeWins },
   196→    { name: 'Draws', value: analytics.resultDistribution.draws },
   197→    { name: 'Away Wins', value: analytics.resultDistribution.awayWins },
   198→  ] : []
   199→
   200→  const htFtTransitionsData = analytics ? [
   201→    {
   202→      name: 'HT Home Lead',
   203→      'FT Home Win': analytics.htToftTransitions.htHomeLeads.ftHomeWin,
   204→      'FT Draw': analytics.htToftTransitions.htHomeLeads.ftDraw,
   205→      'FT Away Win': analytics.htToftTransitions.htHomeLeads.ftAwayWin,
   206→    },
   207→    {
   208→      name: 'HT Draw',
   209→      'FT Home Win': analytics.htToftTransitions.htDraw.ftHomeWin,
   210→      'FT Draw': analytics.htToftTransitions.htDraw.ftDraw,
   211→      'FT Away Win': analytics.htToftTransitions.htDraw.ftAwayWin,
   212→    },
   213→    {
   214→      name: 'HT Away Lead',
   215→      'FT Home Win': analytics.htToftTransitions.htAwayLeads.ftHomeWin,
   216→      'FT Draw': analytics.htToftTransitions.htAwayLeads.ftDraw,
   217→      'FT Away Win': analytics.htToftTransitions.htAwayLeads.ftAwayWin,
   218→    },
   219→  ] : []
   220→
   221→  const overUnderData = analytics ? [
   222→    { name: 'Over 2.5', value: analytics.over25Count, fill: COLORS.over },
   223→    { name: 'Under 2.5', value: analytics.under25Count, fill: COLORS.under },
   224→  ] : []
   225→
   226→  return (
   227→    <div className="space-y-6">
   228→      {error && (
   229→        <Card className="border-red-200 bg-red-50">
   230→          <CardContent className="py-4 text-red-600">{error}</CardContent>
   231→        </Card>
   232→      )}
   233→
   234→      {/* Search */}
   235→      <Card className="shadow-md">
   236→        <CardContent className="py-4">
   237→          <div className="relative max-w-md">
   238→            <Search className="absolute left-3 top-1/2 transform -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   239→            <Input
   240→              placeholder="Search by team name..."
   241→              value={searchTerm}
   242→              onChange={(e) => setSearchTerm(e.target.value)}
   243→              className="pl-10"
   244→            />
   245→          </div>
   246→        </CardContent>
   247→      </Card>
   248→
   249→      {/* Analytics Cards */}
   250→      {loading ? (
   251→        <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
   252→          {[1, 2, 3, 4].map(i => (
   253→            <Card key={i}><CardContent className="py-6"><Skeleton className="h-16 w-full" /></CardContent></Card>
   254→          ))}
   255→        </div>
   256→      ) : !analytics ? (
   257→        <Card className="border-amber-200 bg-amber-50">
   258→          <CardContent className="py-8 text-center">
   259→            <AlertTriangle className="w-8 h-8 mx-auto text-amber-500 mb-3" />
   260→            <p className="text-amber-700 font-medium">No data available for this selection</p>
   261→            <p className="text-sm text-muted-foreground mt-2">Try selecting a different season or league</p>
   262→          </CardContent>
   263→        </Card>
   264→      ) : analytics.totalMatches === 0 ? (
   265→        <Card className="border-blue-200 bg-blue-50">
   266→          <CardContent className="py-8 text-center">
   267→            <Calendar className="w-8 h-8 mx-auto text-blue-500 mb-3" />
   268→            <p className="text-blue-700 font-medium">Season hasn't started yet</p>
   269→            <p className="text-sm text-muted-foreground mt-2">No matches have been played in this season</p>
   270→          </CardContent>
   271→        </Card>
   272→      ) : analytics && (
   273→        <>
   274→          <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
   275→            <Card className="shadow-md hover:shadow-lg transition-shadow">
   276→              <CardContent className="py-6">
   277→                <div className="flex items-center gap-2 text-muted-foreground">
   278→                  <Trophy className="w-4 h-4" />
   279→                  <span className="text-sm font-medium">Total Matches</span>
   280→                </div>
   281→                <p className="text-3xl font-bold mt-2">{analytics.totalMatches}</p>
   282→                <p className="text-xs text-muted-foreground mt-1">{selectedLeagueName} {selectedSeasonName}{isAllSeasons && analytics.seasonsCount && ` (${analytics.seasonsCount} seasons)`}</p>
   283→              </CardContent>
   284→            </Card>
   285→
   286→            <Card className="shadow-md hover:shadow-lg transition-shadow">
   287→              <CardContent className="py-6">
   288→                <div className="flex items-center gap-2 text-muted-foreground">
   289→                  <Goal className="w-4 h-4" />
   290→                  <span className="text-sm font-medium">Avg Goals/Game</span>
   291→                </div>
   292→                <p className="text-3xl font-bold mt-2">{analytics.avgGoalsPerGame}</p>
   293→                <p className="text-xs text-muted-foreground mt-1">Home: {analytics.avgHomeGoals} | Away: {analytics.avgAwayGoals}</p>
   294→              </CardContent>
   295→            </Card>
   296→
   297→            <Card className="shadow-md hover:shadow-lg transition-shadow">
   298→              <CardContent className="py-6">
   299→                <div className="flex items-center gap-2 text-muted-foreground">
   300→                  <TrendingUp className="w-4 h-4" />
   301→                  <span className="text-sm font-medium">HT-FT Correlation</span>
   302→                </div>
   303→                <p className="text-3xl font-bold mt-2 text-green-600">{analytics.htftCorrelationPercent}%</p>
   304→                <p className="text-xs text-muted-foreground mt-1">HT result = FT result</p>
   305→              </CardContent>
   306→            </Card>
   307→
   308→            <Card className="shadow-md hover:shadow-lg transition-shadow">
   309→              <CardContent className="py-6">
   310→                <span className="text-sm font-medium text-muted-foreground">Result Distribution</span>
   311→                <div className="flex gap-2 mt-3 text-sm">
   312→                  <Badge style={{ backgroundColor: COLORS.homeWin }} className="text-white">H: {analytics.homeWinPercent}%</Badge>
   313→                  <Badge style={{ backgroundColor: COLORS.draw }} className="text-white">D: {analytics.drawPercent}%</Badge>
   314→                  <Badge style={{ backgroundColor: COLORS.awayWin }} className="text-white">A: {analytics.awayWinPercent}%</Badge>
   315→                </div>
   316→              </CardContent>
   317→            </Card>
   318→          </div>
   319→
   320→          {/* Over/Under & Odds Analysis Row */}
   321→          <div className="grid md:grid-cols-2 gap-6">
   322→            {/* Over/Under 2.5 Analysis */}
   323→            <Card className="shadow-md border-2 border-emerald-200 bg-gradient-to-r from-emerald-50 to-teal-50 dark:from-emerald-900/20 dark:to-teal-900/20">
   324→              <CardHeader>
   325→                <CardTitle className="text-lg flex items-center gap-2">
   326→                  <Goal className="w-5 h-5 text-emerald-600" />
   327→                  Over/Under 2.5 Goals
   328→                </CardTitle>
   329→                <CardDescription>Total goals per match distribution</CardDescription>
   330→              </CardHeader>
   331→              <CardContent>
   332→                <div className="grid grid-cols-2 gap-4">
   333→                  <div className="text-center p-4 rounded-lg bg-emerald-100/50 dark:bg-emerald-800/30">
   334→                    <p className="text-sm text-muted-foreground">Over 2.5 Goals</p>
   335→                    <p className="text-3xl font-bold text-emerald-600">{analytics.over25Percent}%</p>
   336→                    <p className="text-xs text-muted-foreground">{analytics.over25Count} matches</p>
   337→                  </div>
   338→                  <div className="text-center p-4 rounded-lg bg-red-100/50 dark:bg-red-800/30">
   339→                    <p className="text-sm text-muted-foreground">Under 2.5 Goals</p>
   340→                    <p className="text-3xl font-bold text-red-600">{analytics.under25Percent}%</p>
   341→                    <p className="text-xs text-muted-foreground">{analytics.under25Count} matches</p>
   342→                  </div>
   343→                </div>
   344→                <div className="mt-4 h-4 bg-gray-200 rounded-full overflow-hidden">
   345→                  <div
   346→                    className="h-full bg-emerald-500 transition-all"
   347→                    style={{ width: `${analytics.over25Percent}%` }}
   348→                  />
   349→                </div>
   350→              </CardContent>
   351→            </Card>
   352→
   353→            {/* Odds Analysis */}
   354→            <Card className="shadow-md border-2 border-amber-200 bg-gradient-to-r from-amber-50 to-yellow-50 dark:from-amber-900/20 dark:to-yellow-900/20">
   355→              <CardHeader>
   356→                <CardTitle className="text-lg flex items-center gap-2">
   357→                  <DollarSign className="w-5 h-5 text-amber-600" />
   358→                  Odds Analysis
   359→                </CardTitle>
   360→                <CardDescription>Value betting insights ({analytics.oddsAnalysis.matchesWithOdds} matches with odds)</CardDescription>
   361→              </CardHeader>
   362→              <CardContent>
   363→                <div className="grid grid-cols-3 gap-3">
   364→                  <div className="text-center p-3 rounded-lg bg-white/50 dark:bg-gray-800/50">
   365→                    <p className="text-xs text-muted-foreground">Favorite Wins</p>
   366→                    <p className="text-2xl font-bold text-amber-600">{analytics.oddsAnalysis.favoriteWinPercent}%</p>
   367→                  </div>
   368→                  <div className="text-center p-3 rounded-lg bg-white/50 dark:bg-gray-800/50">
   369→                    <p className="text-xs text-muted-foreground">Underdog Wins</p>
   370→                    <p className="text-2xl font-bold text-purple-600">{analytics.oddsAnalysis.underdogWinPercent}%</p>
   371→                  </div>
   372→                  <div className="text-center p-3 rounded-lg bg-white/50 dark:bg-gray-800/50">
   373→                    <p className="text-xs text-muted-foreground">Draws</p>
   374→                    <p className="text-2xl font-bold text-gray-600">{analytics.oddsAnalysis.drawsPercent}%</p>
   375→                  </div>
   376→                </div>
   377→                <div className="mt-4 grid grid-cols-3 gap-2 text-xs">
   378→                  <div className="text-center">
   379→                    <p className="text-muted-foreground">Avg Odds</p>
   380→                    <p className="font-semibold">H: {analytics.oddsAnalysis.avgHomeOdds}</p>
   381→                  </div>
   382→                  <div className="text-center">
   383→                    <p className="text-muted-foreground">Avg Odds</p>
   384→                    <p className="font-semibold">D: {analytics.oddsAnalysis.avgDrawOdds}</p>
   385→                  </div>
   386→                  <div className="text-center">
   387→                    <p className="text-muted-foreground">Avg Odds</p>
   388→                    <p className="font-semibold">A: {analytics.oddsAnalysis.avgAwayOdds}</p>
   389→                  </div>
   390→                </div>
   391→              </CardContent>
   392→            </Card>
   393→          </div>
   394→
   395→          {/* Implied vs Actual Probabilities */}
   396→          {analytics.oddsAnalysis.matchesWithOdds > 0 && (
   397→            <Card className="shadow-md">
   398→              <CardHeader>
   399→                <CardTitle className="text-lg flex items-center gap-2">
   400→                  <BarChart3 className="w-5 h-5 text-blue-600" />
   401→                  Implied vs Actual Probabilities
   402→                </CardTitle>
   403→                <CardDescription>Compare bookmaker implied odds with actual outcomes</CardDescription>
   404→              </CardHeader>
   405→              <CardContent>
   406→                <div className="grid grid-cols-3 gap-6">
   407→                  <div className="space-y-2">
   408→                    <p className="text-sm font-medium text-center">Home Win</p>
   409→                    <div className="flex gap-2">
   410→                      <div className="flex-1 text-center p-3 rounded bg-blue-50 dark:bg-blue-900/30">
   411→                        <p className="text-xs text-muted-foreground">Implied</p>
   412→                        <p className="text-xl font-bold text-blue-600">{analytics.oddsAnalysis.homeWinImpliedProb}%</p>
   413→                      </div>
   414→                      <div className="flex-1 text-center p-3 rounded bg-green-50 dark:bg-green-900/30">
   415→                        <p className="text-xs text-muted-foreground">Actual</p>
   416→                        <p className="text-xl font-bold text-green-600">{analytics.oddsAnalysis.homeWinActualProb}%</p>
   417→                      </div>
   418→                    </div>
   419→                  </div>
   420→                  <div className="space-y-2">
   421→                    <p className="text-sm font-medium text-center">Draw</p>
   422→                    <div className="flex gap-2">
   423→                      <div className="flex-1 text-center p-3 rounded bg-blue-50 dark:bg-blue-900/30">
   424→                        <p className="text-xs text-muted-foreground">Implied</p>
   425→                        <p className="text-xl font-bold text-blue-600">{analytics.oddsAnalysis.drawImpliedProb}%</p>
   426→                      </div>
   427→                      <div className="flex-1 text-center p-3 rounded bg-green-50 dark:bg-green-900/30">
   428→                        <p className="text-xs text-muted-foreground">Actual</p>
   429→                        <p className="text-xl font-bold text-green-600">{analytics.oddsAnalysis.drawActualProb}%</p>
   430→                      </div>
   431→                    </div>
   432→                  </div>
   433→                  <div className="space-y-2">
   434→                    <p className="text-sm font-medium text-center">Away Win</p>
   435→                    <div className="flex gap-2">
   436→                      <div className="flex-1 text-center p-3 rounded bg-blue-50 dark:bg-blue-900/30">
   437→                        <p className="text-xs text-muted-foreground">Implied</p>
   438→                        <p className="text-xl font-bold text-blue-600">{analytics.oddsAnalysis.awayWinImpliedProb}%</p>
   439→                      </div>
   440→                      <div className="flex-1 text-center p-3 rounded bg-green-50 dark:bg-green-900/30">
   441→                        <p className="text-xs text-muted-foreground">Actual</p>

   444→                    </div>

   450→
   451→          {/* Match Statistics */}
   452→          <Card className="shadow-md">
   453→            <CardHeader>
   454→              <CardTitle className="text-lg flex items-center gap-2">
   455→                <BarChart3 className="w-5 h-5 text-indigo-600" />
   456→                Average Match Statistics
   457→              </CardTitle>
   458→              <CardDescription>Per-match averages across all games</CardDescription>
   459→            </CardHeader>
   460→            <CardContent>
   461→              <div className="grid grid-cols-2 md:grid-cols-4 gap-4">
   462→                <div className="text-center p-4 rounded-lg bg-gray-50 dark:bg-gray-800/50">

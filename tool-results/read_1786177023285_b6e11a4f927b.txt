    52→// --- Helper: compute O3.5 probability from total xG ---
    53→function computeO35Prob(totalXg: number): number {
    54→  const p0 = Math.exp(-totalXg);
    55→  const p1 = totalXg * p0;
    56→  const p2 = (totalXg * totalXg / 2) * p0;
    57→  const p3 = (totalXg ** 3 / 6) * p0;
    58→  return Math.round((1 - p0 - p1 - p2 - p3) * 100);
    59→}

    61→// --- Helper: compute rolling 5-game combined scoring ---
    62→function getRollingCombinedScoring(results: MatchResult[], team: string, isHome: boolean, beforeDate: string): number {
    63→  const teamMatches = results
    64→    .filter(r => r.date < beforeDate && (isHome ? r.homeTeam === team : r.awayTeam === team))
    65→    .sort((a, b) => b.date.localeCompare(a.date))
    66→    .slice(0, 5);
    67→  
    68→  if (teamMatches.length === 0) return 0;
    69→  return teamMatches.reduce((sum, m) => sum + (isHome ? m.ftHomeGoals : m.ftAwayGoals), 0) / teamMatches.length;
    70→}

    72→// --- Helper: compute xG signal (same logic as PredictionsTab) ---
    73→function computeXGSignal(results: MatchResult[], homeTeam: string, awayTeam: string): string {
    74→  const teamXgStats = new Map<string, { matches: number; totalXg: number; actualGoals: number }>();
    75→  
    76→  results.forEach(r => {
    77→    const homeShotsOff = r.homeShots - r.homeShotsOnTarget;
    78→    const awayShotsOff = r.awayShots - r.awayShotsOnTarget;
    79→    const homeXg = (r.homeShotsOnTarget * 0.30) + (homeShotsOff * 0.08);
    80→    const awayXg = (r.awayShotsOnTarget * 0.30) + (awayShotsOff * 0.08);
    81→    
    82→    for (const [team, xg, goals] of [[r.homeTeam, homeXg, r.ftHomeGoals], [r.awayTeam, awayXg, r.ftAwayGoals]] as const) {
    83→      const s = teamXgStats.get(team) || { matches: 0, totalXg: 0, actualGoals: 0 };
    84→      s.matches++; s.totalXg += xg; s.actualGoals += goals;
    85→      teamXgStats.set(team, s);
    86→    }
    87→  });
    88→  
    89→  const hd = teamXgStats.get(homeTeam);
    90→  const ad = teamXgStats.get(awayTeam);
    91→  if (!hd || !ad || hd.matches < 5 || ad.matches < 5) return 'Neutral';
    92→  
    93→  const homeDiff = (hd.actualGoals / hd.matches) - (hd.totalXg / hd.matches);
    94→  const awayDiff = (ad.actualGoals / ad.matches) - (ad.totalXg / ad.matches);
    95→  const total = homeDiff + awayDiff;
    96→  
    97→  if (total <= -1.0) return 'Strong Over';
    98→  if (total <= -0.5) return 'Over';
    99→  if (total >= 1.0) return 'Strong Under';
   100→  if (total >= 0.5) return 'Under';
   101→  return 'Neutral';
   102→}

   104→// --- Helper: compute regression signal (same logic as PredictionsTab) ---
   105→function computeRegressionSignal(results: MatchResult[], homeTeam: string, awayTeam: string): string {
   106→  const getTeamReg = (team: string): number => {
   107→    const matches = results.filter(r => r.homeTeam === team || r.awayTeam === team)
   108→      .sort((a, b) => b.date.localeCompare(a.date));
   109→    if (matches.length < 5) return 0;
   110→    const last3 = matches.slice(0, 3);
   111→    const seasonAvg = matches.reduce((s, m) => {
   112→      const g = m.homeTeam === team ? m.ftHomeGoals : m.ftAwayGoals;
   113→      return s + g;
   114→    }, 0) / matches.length;
   115→    const last3Avg = last3.reduce((s, m) => {
   116→      const g = m.homeTeam === team ? m.ftHomeGoals : m.ftAwayGoals;
   117→      return s + g;
   118→    }, 0) / last3.length;
   119→    return last3Avg - seasonAvg;
   120→  };
   121→  
   122→  const homeReg = getTeamReg(homeTeam);
   123→  const awayReg = getTeamReg(awayTeam);
   124→  const totalReg = homeReg + awayReg;
   125→  
   126→  if (totalReg <= -0.5) return 'Under';
   127→  if (totalReg <= -0.3) return 'Strong Under';
   128→  if (totalReg >= 0.5) return 'Over';
   129→  if (totalReg >= 0.3) return 'Strong Over';
   130→  return 'Neutral';
   131→}

   133→// --- Helper: compute Z-Score signal (same logic as PredictionsTab) ---
   134→function computeZScoreSignal(results: MatchResult[], homeTeam: string, awayTeam: string): string {
   135→  const zTeamStats = new Map<string, { matches: number; goals: number[]; totalGoals: number; mean: number; stdDev: number; last3Avg: number }>();
   136→  
   137→  results.forEach(r => {
   138→    for (const [team] of [[r.homeTeam], [r.awayTeam]] as const) {
   139→      const totalGoals = r.ftHomeGoals + r.ftAwayGoals;
   140→      const s = zTeamStats.get(team) || { matches: 0, goals: [] as number[], totalGoals: 0, mean: 0, stdDev: 0, last3Avg: 0 };
   141→      s.matches++; s.goals.push(totalGoals); s.totalGoals += totalGoals;
   142→      zTeamStats.set(team, s);
   143→    }
   144→  });
   145→  
   146→  zTeamStats.forEach(stats => {
   147→    stats.mean = stats.totalGoals / stats.matches;
   148→    const last3 = stats.goals.slice(0, 3);
   149→    stats.last3Avg = last3.length > 0 ? last3.reduce((a, b) => a + b, 0) / last3.length : 0;
   150→    const sqDiffs = stats.goals.map(g => (g - stats.mean) ** 2);
   151→    stats.stdDev = Math.sqrt(sqDiffs.reduce((a, b) => a + b, 0) / stats.matches);
   152→  });
   153→  
   154→  const hz = zTeamStats.get(homeTeam);
   155→  const az = zTeamStats.get(awayTeam);
   156→  if (!hz || !az || hz.matches < 3 || az.matches < 3) return 'Neutral';
   157→  
   158→  const homeZ = hz.stdDev > 0 ? (hz.last3Avg - hz.mean) / hz.stdDev : 0;
   159→  const awayZ = az.stdDev > 0 ? (az.last3Avg - az.mean) / az.stdDev : 0;
   160→  
   161→  let score = 0;
   162→  if (homeZ <= -1.5) score += 2; else if (homeZ <= -1.0) score += 1;
   163→  if (awayZ <= -1.5) score += 2; else if (awayZ <= -1.0) score += 1;
   164→  if (homeZ >= 1.5) score -= 2; else if (homeZ >= 1.0) score -= 1;
   165→  if (awayZ >= 1.5) score -= 2; else if (awayZ >= 1.0) score -= 1;
   166→  
   167→  if (score >= 4) return 'Strong Over';
   168→  if (score >= 2.5) return 'Over';
   169→  if (score <= -3) return 'Strong Under';
   170→  if (score <= -1.5) return 'Under';
   171→  return 'Neutral';
   172→}

   174→// --- Helper: draw probability from odds ---
   175→function drawProbFromOdds(oddsAvgHome: number | null, oddsAvgDraw: number | null, oddsAvgAway: number | null): number {
   176→  if (!oddsAvgHome || !oddsAvgDraw || !oddsAvgAway) return 50;
   177→  const impliedHome = 1 / oddsAvgHome;
   178→  const impliedDraw = 1 / oddsAvgDraw;
   179→  const impliedAway = 1 / oddsAvgAway;
   180→  const total = impliedHome + impliedDraw + impliedAway;
   181→  return Math.round((impliedDraw / total) * 100);
   182→}

   210→async function runLeagueBacktest(leagueCode: string, testSeason: string): Promise<MatchRecord[]> {
   211→  const training = EUROPEAN_SEASONS.filter(s => s < testSeason).slice(-5);
   212→  
   213→  // Fetch data
   214→  const trainingResults = await Promise.all(training.map(s => fetchSeasonData(leagueCode, s)));
   215→  const trainingData = trainingResults.flat();
   216→  const testData = await fetchSeasonData(leagueCode, testSeason);
   217→  
   218→  if (testData.length === 0) return [];
   219→  
   220→  // Combine for signal computation (signals need full history)
   221→  const allData = [...trainingData, ...testData].sort((a, b) => a.date.localeCompare(b.date));
   222→  
   223→  const seasonWeights = calculateSeasonWeights(training);
   224→  const leagueAvgs = calculateLeagueAverages(trainingData);
   225→  
   226→  // Compute league baselines for threshold resolution
   227→  const analyticsForBaseline = {
   228→    avgGoalsPerGame: leagueAvgs.avgTotalGoals,
   229→    over25Percent: leagueAvgs.over25Rate * 100,
   230→    avgHomeGoals: leagueAvgs.avgHomeGoals,
   231→    avgAwayGoals: leagueAvgs.avgAwayGoals,
   232→    overallShotConversion: '10',
   233→  };
   234→  const baselines = computeLeagueBaselines(trainingData, analyticsForBaseline);
   235→  const resolved = resolveAllThresholds(leagueCode, baselines);
   236→  
   237→  const records: MatchRecord[] = [];
   238→  
   239→  for (const match of testData) {
   240→    // Skip if either team lacks training history
   241→    const homeInTraining = trainingData.some(m => m.homeTeam === match.homeTeam || m.awayTeam === match.homeTeam);
   242→    const awayInTraining = trainingData.some(m => m.homeTeam === match.awayTeam || m.awayTeam === match.awayTeam);
   243→    if (!homeInTraining || !awayInTraining) continue;
   244→    
   245→    // Generate predictions
   246→    const pred = generateBacktestPredictions(trainingData, match.homeTeam, match.awayTeam, leagueAvgs, seasonWeights);
   247→    const o35Prob = computeO35Prob(pred.totalXg);
   248→    
   249→    // Compute detector inputs
   250→    const homeRollingScored = getRollingCombinedScoring(allData, match.homeTeam, true, match.date);
   251→    const awayRollingScored = getRollingCombinedScoring(allData, match.awayTeam, false, match.date);
   252→    const rollingCombined = homeRollingScored + awayRollingScored;
   253→    const o25ImpliedProb = match.oddsAvgOver25 ? (1 / match.oddsAvgOver25) * 100 : null;
   254→    
   255→    // Compute signals (using all data before this match for Z-Score/regression)
   256→    const beforeThisMatch = allData.filter(r => r.date < match.date);
   257→    const xgSignal = computeXGSignal(beforeThisMatch, match.homeTeam, match.awayTeam);
   258→    const regressionSignal = computeRegressionSignal(beforeThisMatch, match.homeTeam, match.awayTeam);
   259→    const zScoreSignal = computeZScoreSignal(beforeThisMatch, match.homeTeam, match.awayTeam);
   260→    
   261→    const signalInput: SignalInput = { xgSignal, regressionSignal, zScoreSignal };
   262→    
   263→    const bttsChecklistInput: ChecklistInput = {
   264→      avgGoalsPerGame: leagueAvgs.avgTotalGoals,
   265→      over25Percent: leagueAvgs.over25Rate * 100,
   266→      bttsProb: pred.btts,
   267→      avgHomeGoals: leagueAvgs.avgHomeGoals,
   268→      avgAwayGoals: leagueAvgs.avgAwayGoals,
   269→      o25Prob: pred.over25,
   270→      o35Prob: o35Prob,
   271→      overallShotConversion: 10,
   272→      rollingHomeScored: homeRollingScored,
   273→      rollingAwayScored: awayRollingScored,
   274→      rollingCombinedScoring: rollingCombined,
   275→      o25ImpliedProb,
   276→    };
   277→    
   278→    // Run all detectors
   279→    const bttsScore = computeBttsChecklist(bttsChecklistInput, resolved);
   280→    const o35Score = computeOver35Checklist(bttsChecklistInput, resolved);
   281→    const strongBet = computeStrongBet(bttsChecklistInput, signalInput, resolved);
   282→    const greyResult = computeGreyResult(bttsChecklistInput, signalInput, resolved);
   283→    const goalFest = computeGoalFest(bttsChecklistInput, signalInput, resolved);
   284→    const drawProb = drawProbFromOdds(match.oddsAvgHome, match.oddsAvgDraw, match.oddsAvgAway);
   285→    const bttsBH = computeBTTSBothHalves({
   286→      o25Prob: pred.over25,
   287→      o35Prob,
   288→      bttsProb: pred.btts,
   289→      rollingCombinedScoring: rollingCombined,
   290→      o25ImpliedProb,
   291→      drawProb,
   292→      avgGoalsPerGame: leagueAvgs.avgTotalGoals,
   293→    });
   294→    
   295→    // Actual outcomes
   296→    const totalGoals = match.ftHomeGoals + match.ftAwayGoals;
   297→    const btts = match.ftHomeGoals > 0 && match.ftAwayGoals > 0;
   298→    const over25 = totalGoals > 2.5;
   299→    const over35 = totalGoals > 3.5;
   300→    // BTTS-BH: both teams score in EACH half
   301→    const shHomeGoals = match.ftHomeGoals - match.htHomeGoals;
   302→    const shAwayGoals = match.ftAwayGoals - match.htAwayGoals;
   303→    const bttsBH_actual = match.htHomeGoals > 0 && match.htAwayGoals > 0 && shHomeGoals > 0 && shAwayGoals > 0;
   304→
   305→    records.push({
   306→      league: leagueCode,
   307→      season: testSeason,
   308→      homeTeam: match.homeTeam,
   309→      awayTeam: match.awayTeam,
   310→      date: match.date,
   311→      totalGoals,
   312→      btts,
   313→      over25,
   314→      over35,
   315→      bttsChecklistScore: bttsScore,
   316→      over35ChecklistScore: o35Score,
   317→      isStrongBet: strongBet.isStrongBet,
   318→      strongBetPoints: strongBet.points,
   319→      isGreyResult: greyResult.isGreyResult,
   320→      greyResultScore: greyResult.score,
   321→      isGoalFest: goalFest.isGoalFest,
   322→      goalFestScore: goalFest.score,
   323→      isBttsBH: bttsBH.isBTTSBothHalves,
   324→      bttsBHScore: bttsBH.score,
   325→      bttsBH_actual,
   326→    });
   327→  }
   328→  
   329→  return records;
   330→}

   332→// --- Analysis ---
   333→interface DetectorMetrics {
   334→  detector: string;
   335→  outcome: string;
   336→  totalMatches: number;
   337→  fired: number;
   338→  fireRate: number;
   339→  hits: number;
   340→  hitRate: number;
   341→  baseRate: number;
   342→  lift: number;
   343→}
   344→
   345→function analyzeDetector(records: MatchRecord[], detector: string, outcome: 'btts' | 'over25' | 'over35' | 'o3goals' | 'grey4goals', filterFn: (r: MatchRecord) => boolean): DetectorMetrics {
   346→  const total = records.length;
   347→  const fired = records.filter(filterFn);
   348→  
   349→  let baseRate: number;
   350→  let hits: number;
   351→  
   352→  switch (outcome) {
   353→    case 'btts':
   354→      baseRate = records.filter(r => r.btts).length / total;
   355→      hits = fired.filter(r => r.btts).length;
   356→      break;
   357→    case 'over25':
   358→      baseRate = records.filter(r => r.over25).length / total;
   359→      hits = fired.filter(r => r.over25).length;
   360→      break;
   361→    case 'over35':
   362→      baseRate = records.filter(r => r.over35).length / total;
   363→      hits = fired.filter(r => r.over35).length;
   364→      break;
   365→    case 'o3goals':
   366→      baseRate = records.filter(r => r.totalGoals >= 3).length / total;
   367→      hits = fired.filter(r => r.totalGoals >= 3).length;
   368→      break;
   369→    case 'grey4goals':
   370→      baseRate = records.filter(r => r.totalGoals >= 4).length / total;
   371→      hits = fired.filter(r => r.totalGoals >= 4).length;
   372→      break;
   373→    case 'bttsBH':
   374→      baseRate = records.filter(r => r.bttsBH_actual).length / total;
   375→      hits = fired.filter(r => r.bttsBH_actual).length;
   376→      break;
   377→  }
   378→  
   379→  return {
   380→    detector,
   381→    outcome,
   382→    totalMatches: total,
   383→    fired: fired.length,
   384→    fireRate: fired.length / total,
   385→    hitRate: fired.length > 0 ? hits / fired.length : 0,
   386→    baseRate: baseRate!,
   387→    lift: fired.length > 0 ? (hits / fired.length) / baseRate! : 0,
   388→  };
   389→}

   396→function fmtLift(n: number): string {
   397→  const color = n >= 1.15 ? '✅' : n >= 1.05 ? '🟢' : n >= 0.95 ? '🟡' : '🔴';
   398→  return `${n.toFixed(2)}x ${color}`;

    31→    31→/**
    32→    32→ * Compute league baselines from results + analytics data.
    33→    33→ * Call this once per league/season, pass the result into all filter functions.
    34→    34→ */
    35→    35→export function computeLeagueBaselines(
    36→    36→  results: { ftHomeGoals: number; ftAwayGoals: number }[],
    37→    37→  analytics: {
    38→    38→    avgGoalsPerGame: number;
    39→    39→    over25Percent: number;
    40→    40→    avgHomeGoals: number;
    41→    41→    avgAwayGoals: number;
    42→    42→    overallShotConversion: string | number;
    43→    43→  }
    44→    44→): LeagueBaselines {
    45→    45→  const total = results.length || 1;
    46→    46→  const bttsCount = results.filter(r => r.ftHomeGoals > 0 && r.ftAwayGoals > 0).length;
    47→    47→  const over35Count = results.filter(r => r.ftHomeGoals + r.ftAwayGoals > 3.5).length;
    48→    48→
    49→    49→  return {
    50→    50→    avgGoalsPerGame: analytics.avgGoalsPerGame,
    51→    51→    over25Rate: analytics.over25Percent,
    52→    52→    bttsRate: (bttsCount / total) * 100,
    53→    53→    over35Rate: (over35Count / total) * 100,
    54→    54→    avgHomeGoals: analytics.avgHomeGoals,
    55→    55→    avgAwayGoals: analytics.avgAwayGoals,
    56→    56→    shotConversion: typeof analytics.overallShotConversion === 'string'
    57→    57→      ? parseFloat(analytics.overallShotConversion)
    58→    58→      : analytics.overallShotConversion,
    59→    59→  };
    60→    60→}

   110→   110→// ============================================================================
   111→   111→// Backtest-derived threshold registry
   112→   112→// ============================================================================
   113→   113→// This is the central store. Thresholds are added here when backtests run.
   114→   114→// Leagues without entries fall back to the hybrid formula.
   115→   115→// Supports localStorage persistence for client-side survival across page reloads.
   116→   116→// ============================================================================
   117→   117→
   118→   118→const STORAGE_KEY = 'techmari_backtest_thresholds';
   119→   119→
   120→   120→const backtestRegistry = new Map<string, LeagueBacktestThresholds>();
   121→   121→
   122→   122→let loadedFromStorage = false;
   123→   123→
   124→   124→/**
   125→   125→ * Load thresholds from localStorage into the in-memory registry.
   126→   126→ * Safe to call multiple times — only loads once per session.
   127→   127→ */
   128→   128→function loadFromLocalStorage(): void {
   129→   129→  if (loadedFromStorage || typeof window === 'undefined') return;
   130→   130→  loadedFromStorage = true;
   131→   131→  try {
   132→   132→    const raw = localStorage.getItem(STORAGE_KEY);
   133→   133→    if (!raw) return;
   134→   134→    const data = JSON.parse(raw);
   135→   135→    if (data.version !== 1) return;
   136→   136→    const entries: Record<string, LeagueBacktestThresholds> = data.thresholds;
   137→   137→    for (const [key, thresholds] of Object.entries(entries)) {
   138→   138→      backtestRegistry.set(key, thresholds);
   139→   139→    }
   140→   140→    const count = Object.keys(entries).length;
   141→   141→    if (count > 0) {
   142→   142→      console.log(`[BettingFilters] Loaded ${count} persisted threshold(s) from localStorage`);
   143→   143→    }
   144→   144→  } catch {
   145→   145→    // localStorage not available or corrupt data — silent fail
   146→   146→  }
   147→   147→}
   148→   148→
   149→   149→/**
   150→   150→ * Save all in-memory thresholds to localStorage.
   151→   151→ */
   152→   152→function saveToLocalStorage(): void {
   153→   153→  if (typeof window === 'undefined') return;
   154→   154→  try {
   155→   155→    const data: Record<string, LeagueBacktestThresholds> = {};
   156→   156→    for (const [key, thresholds] of backtestRegistry.entries()) {
   157→   157→      data[key] = thresholds;
   158→   158→    }
   159→   159→    localStorage.setItem(STORAGE_KEY, JSON.stringify({
   160→   160→      version: 1,
   161→   161→      updatedAt: new Date().toISOString(),
   162→   162→      thresholds: data,
   163→   163→    }));
   164→   164→  } catch {
   165→   165→    // localStorage not available or quota exceeded — silent fail
   166→   166→  }
   167→   167→}
   168→   168→
   169→   169→/**
   170→   170→ * Register backtest-derived thresholds for a league.
   171→   171→ * Also persists to localStorage automatically.
   172→   172→ */
   173→   173→export function registerBacktestThresholds(thresholds: LeagueBacktestThresholds): void {
   174→   174→  // Ensure we've loaded from localStorage first
   175→   175→  loadFromLocalStorage();
   176→   176→  backtestRegistry.set(thresholds.leagueName.toLowerCase(), thresholds);
   177→   177→  saveToLocalStorage();
   178→   178→}

   410→   410→/**
   411→   411→ * Resolve all thresholds for a given league.
   412→   412→ * Uses backtest-derived if available (sample >= minSampleSize), otherwise hybrid.
   413→   413→ */
   414→   414→export function resolveAllThresholds(
   415→   415→  leagueName: string | undefined,
   416→   416→  baselines: LeagueBaselines,
   417→   417→  minSampleSize = 150
   418→   418→) {
   419→   419→  const bt = leagueName ? getBacktestThresholds(leagueName) : undefined;
   420→   420→  const useBacktest = bt !== undefined && bt.sampleSize >= minSampleSize;
   421→   421→
   422→   422→  const src: 'backtest' | 'hybrid' = useBacktest ? 'backtest' : 'hybrid';
   423→   423→
   424→   424→  // BTTS match-level thresholds (lean 2-check)
   425→   425→  const btts: ResolvedThresholds = useBacktest
   426→   426→    ? { modelBttsProb: bt!.btts.modelBttsProb, modelO25Prob: bt!.btts.modelO25Prob, source: 'backtest' as const }
   427→   427→    : {
   428→   428→        modelBttsProb: hybridThreshold(BTTS_HYBRID_THRESHOLDS.modelBttsProb.floor, baselines.bttsRate, BTTS_HYBRID_THRESHOLDS.modelBttsProb.multiplier),
   429→   429→        modelO25Prob: hybridThreshold(BTTS_HYBRID_THRESHOLDS.modelO25Prob.floor, baselines.over25Rate, BTTS_HYBRID_THRESHOLDS.modelO25Prob.multiplier),
   430→   430→        source: 'hybrid' as const,
   431→   431→      };
   432→   432→
   433→   433→  // Over 3.5 match-level thresholds (lean 2-check)
   434→   434→  const over35: ResolvedOver35Thresholds = useBacktest
   435→   435→    ? { modelO35Prob: bt!.over35.modelO35Prob, bttsProb: bt!.over35.bttsProb, source: 'backtest' as const }
   436→   436→    : {
   437→   437→        modelO35Prob: hybridThreshold(OVER35_HYBRID_THRESHOLDS.modelO35Prob.floor, baselines.over35Rate, OVER35_HYBRID_THRESHOLDS.modelO35Prob.multiplier),
   438→   438→        bttsProb: hybridThreshold(OVER35_HYBRID_THRESHOLDS.bttsProb.floor, baselines.bttsRate, OVER35_HYBRID_THRESHOLDS.bttsProb.multiplier),
   439→   439→        source: 'hybrid' as const,
   440→   440→      };
   441→   441→
   442→   442→  // STRONG BET thresholds
   443→   443→  const strongBet: ResolvedStrongBetThresholds = useBacktest
   444→   444→    ? { ...bt!.strongBet, source: 'backtest' }
   445→   445→    : {
   446→   446→        o25Prob: hybridThreshold(STRONG_BET_HYBRID.o25Prob.floor, baselines.over25Rate, STRONG_BET_HYBRID.o25Prob.multiplier),
   447→   447→        o35Prob: hybridThreshold(STRONG_BET_HYBRID.o35Prob.floor, baselines.over35Rate, STRONG_BET_HYBRID.o35Prob.multiplier),
   448→   448→        bttsProb: hybridThreshold(STRONG_BET_HYBRID.bttsProb.floor, baselines.bttsRate, STRONG_BET_HYBRID.bttsProb.multiplier),
   449→   449→        bttsChecklistCount: STRONG_BET_HYBRID.bttsChecklistCount,

   451→   451→      };
   452→   452→
   453→   453→  // GREY RESULT thresholds

   460→   460→    : {
   461→   461→        bttsProb: GREY_RESULT_CONFIG.bttsProb.min,
   462→   462→        bttsProbMax: GREY_RESULT_CONFIG.bttsProb.max,

   464→   464→        o35Prob: hybridThreshold(GREY_RESULT_CONFIG.o35Prob.floor, baselines.over35Rate, GREY_RESULT_CONFIG.o35Prob.multiplier),

   466→   466→        over35ChecklistCount: GREY_RESULT_CONFIG.over35ChecklistCount,
   467→   467→        requiredChecks: GREY_RESULT_CONFIG.requiredChecks,
   468→   468→        source: 'hybrid' as const,
   469→   469→      };
   470→   470→
   471→   471→  // GOAL FEST thresholds (lean 6-check)
   472→   472→  const goalFest: ResolvedGoalFestThresholds = useBacktest
   473→   473→    ? {
   474→   474→        o25Prob: bt!.strongBet.o25Prob,
   475→   475→        bttsProb: bt!.strongBet.bttsProb,
   476→   476→        o35Prob: bt!.over35.modelO35Prob,
   477→   477→        requiredChecks: GOAL_FEST_CONFIG.requiredChecks,
   478→   478→        source: 'backtest' as const,
   479→   479→      }
   480→   480→    : {
   481→   481→        o25Prob: hybridThreshold(GOAL_FEST_CONFIG.o25Prob.floor, baselines.over25Rate, GOAL_FEST_CONFIG.o25Prob.multiplier),
   482→   482→        bttsProb: hybridThreshold(GOAL_FEST_CONFIG.bttsProb.floor, baselines.bttsRate, GOAL_FEST_CONFIG.bttsProb.multiplier),
   483→   483→        o35Prob: hybridThreshold(GOAL_FEST_CONFIG.o35Prob.floor, baselines.over35Rate, GOAL_FEST_CONFIG.o35Prob.multiplier),
   484→   484→        requiredChecks: GOAL_FEST_CONFIG.requiredChecks,
   485→   485→        source: 'hybrid' as const,
   486→   486→      };
   487→   487→
   488→   488→  return { btts, over35, strongBet, greyResult, goalFest, source: src };
   489→   489→}

   612→   612→    (xg === 'Over' || xg === 'Strong Over') && isRegressionUnder(reg);
   613→   613→
   614→   614→  const checks = [
   615→   615→    { check: 'O2.5 >=' + st.o25Prob.toFixed(0) + '%', points: p.o25, passed: checklistInput.o25Prob >= st.o25Prob },

   618→   618→    { check: 'Signal Divergence (xG vs Reg)', points: p.signalDivergence, passed: hasSignalDivergence(signals.xgSignal, signals.regressionSignal) },
   619→   619→    { check: 'O2.5 Implied >=' + O25_IMPLIED_THRESHOLDS.strongBet + '%', points: p.o25Implied, passed: checklistInput.o25ImpliedProb !== null && checklistInput.o25ImpliedProb >= O25_IMPLIED_THRESHOLDS.strongBet },

   621→   621→  ];

   750→   750→interface BacktestMatch {
   751→   751→  ftHomeGoals: number;
   752→   752→  ftAwayGoals: number;
   753→   753→  // Model predictions (raw or calibrated)
   754→   754→  predictedBtts?: number;
   755→   755→  predictedO25?: number;
   756→   756→  predictedO35?: number;
   757→   757→  // Match context
   758→   758→  avgHomeGoals?: number;
   759→   759→  avgAwayGoals?: number;
   760→   760→  shotConversion?: number;
   761→   761→}
   762→   762→
   763→   763→interface BacktestDerivationOptions {
   764→   764→  /** Minimum sample size to trust derived thresholds */
   765→   765→  minSampleSize?: number;
   766→   766→  /** For probability thresholds: target minimum precision (e.g. 0.65 means at least 65% accuracy) */
   767→   767→  minAccuracy?: number;
   768→   768→  /** For numeric thresholds (goals): target minimum precision */
   769→   769→  minGoalAccuracy?: number;
   770→   770→}
   771→   771→
   772→   772→/**
   773→   773→ * Derive optimal thresholds from backtest data using accuracy optimization.
   774→   774→ *
   775→   775→ * For each criterion, sweeps candidate threshold values and finds the one that
   776→   776→ * maximizes the correct prediction rate while meeting the minimum accuracy target.
   777→   777→ *
   778→   778→ * @param leagueName - League identifier for the registry
   779→   779→ * @param matches - Historical match data with predictions and outcomes
   780→   780→ * @param currentBaselines - Current league baselines (for fallback floors)
   781→   781→ * @param options - Tuning parameters
   782→   782→ * @returns Derived thresholds, or null if insufficient data
   783→   783→ */
   784→   784→export function deriveThresholdsFromBacktest(
   785→   785→  leagueName: string,
   786→   786→  matches: BacktestMatch[],
   787→   787→  currentBaselines: LeagueBaselines,
   788→   788→  options: BacktestDerivationOptions = {}
   789→   789→): LeagueBacktestThresholds | null {
   790→   790→  const {
   791→   791→    minSampleSize = 150,
   792→   792→    minAccuracy = 0.60,
   793→   793→    minGoalAccuracy = 0.55,
   794→   794→  } = options;
   795→   795→
   796→   796→  if (matches.length < minSampleSize) return null;
   797→   797→
   798→   798→  // --- Helper: compute BTTS hit rate at various thresholds ---
   799→   799→  const findOptimalProbThreshold = (
   800→   800→    getPredicted: (m: BacktestMatch) => number | undefined,
   801→   801→    getActual: (m: BacktestMatch) => boolean,
   802→   802→    searchRange: [number, number],
   803→   803→    step = 1,
   804→   804→    floor?: number
   805→   805→  ): number => {
   806→   806→    const [lo, hi] = searchRange;
   807→   807→    let bestThreshold = floor ?? lo;
   808→   808→    let bestAccuracy = 0;
   809→   809→    let bestSampleAbove = 0;
   810→   810→
   811→   811→    for (let t = lo; t <= hi; t += step) {
   812→   812→      const aboveThreshold = matches.filter(m => {
   813→   813→        const pred = getPredicted(m);
   814→   814→        return pred !== undefined && pred >= t;
   815→   815→      });
   816→   816→      if (aboveThreshold.length < 30) continue; // need meaningful sample above threshold
   817→   817→
   818→   818→      const correct = aboveThreshold.filter(m => getActual(m)).length;
   819→   819→      const accuracy = correct / aboveThreshold.length;
   820→   820→
   821→   821→      // Prefer: highest accuracy that meets minimum, with most samples
   822→   822→      if (accuracy >= minAccuracy && (accuracy > bestAccuracy || (accuracy === bestAccuracy && aboveThreshold.length > bestSampleAbove))) {
   823→   823→        bestThreshold = t;
   824→   824→        bestAccuracy = accuracy;
   825→   825→        bestSampleAbove = aboveThreshold.length;
   826→   826→      }
   827→   827→    }
   828→   828→
   829→   829→    // If no threshold met min accuracy, use the one with best accuracy above floor
   830→   830→    if (bestAccuracy < minAccuracy && floor !== undefined) {
   831→   831→      return floor;
   832→   832→    }
   833→   833→    return bestThreshold;
   834→   834→  };
   835→   835→
   836→   836→  const findOptimalGoalThreshold = (
   837→   837→    getPredicted: (m: BacktestMatch) => number | undefined,
   838→   838→    getActual: (m: BacktestMatch) => boolean,
   839→   839→    searchRange: [number, number],
   840→   840→    step = 0.05,
   841→   841→    floor?: number
   842→   842→  ): number => {
   843→   843→    const [lo, hi] = searchRange;
   844→   844→    let bestThreshold = floor ?? lo;
   845→   845→    let bestAccuracy = 0;
   846→   846→    let bestSampleAbove = 0;
   847→   847→
   848→   848→    for (let t = lo; t <= hi; t += step) {
   849→   849→      const aboveThreshold = matches.filter(m => {
   850→   850→        const pred = getPredicted(m);
   851→   851→        return pred !== undefined && pred >= t;
   852→   852→      });
   853→   853→      if (aboveThreshold.length < 30) continue;
   854→   854→
   855→   855→      const correct = aboveThreshold.filter(m => getActual(m)).length;
   856→   856→      const accuracy = correct / aboveThreshold.length;
   857→   857→
   858→   858→      if (accuracy >= minGoalAccuracy && (accuracy > bestAccuracy || (accuracy === bestAccuracy && aboveThreshold.length > bestSampleAbove))) {
   859→   859→        bestThreshold = Math.round(t * 100) / 100;
   860→   860→        bestAccuracy = accuracy;
   861→   861→        bestSampleAbove = aboveThreshold.length;
   862→   862→      }
   863→   863→    }
   864→   864→
   865→   865→    if (bestAccuracy < minGoalAccuracy && floor !== undefined) {
   866→   866→      return floor;
   867→   867→    }
   868→   868→    return bestThreshold;
   869→   869→  };
   870→   870→
   871→   871→  // --- Derive BTTS thresholds (lean 2-check) ---
   872→   872→  const bttsModelProb = findOptimalProbThreshold(
   873→   873→    m => m.predictedBtts,

   875→   875→    [40, 80],
   876→   876→    1,
   877→   877→    BTTS_HYBRID_THRESHOLDS.modelBttsProb.floor
   878→   878→  );
   879→   879→
   880→   880→  const bttsO25Prob = findOptimalProbThreshold(
   881→   881→    m => m.predictedO25,
   882→   882→    m => m.ftHomeGoals > 0 && m.ftAwayGoals > 0,
   883→   883→    [45, 85],
   884→   884→    1,
   885→   885→    BTTS_HYBRID_THRESHOLDS.modelO25Prob.floor
   886→   886→  );
   887→   887→
   888→   888→  // --- Derive Over 3.5 thresholds (lean 2-check) ---
   889→   889→  const o35ModelProb = findOptimalProbThreshold(
   890→   890→    m => m.predictedO35,
   891→   891→    m => (m.ftHomeGoals + m.ftAwayGoals) > 3.5,
   892→   892→    [20, 65],
   893→   893→    1,
   894→   894→    OVER35_HYBRID_THRESHOLDS.modelO35Prob.floor
   895→   895→  );
   896→   896→
   897→   897→  const o35BttsProb = findOptimalProbThreshold(

   899→   899→    m => (m.ftHomeGoals + m.ftAwayGoals) > 3.5,
   900→   900→    [35, 80],
   901→   901→    1,
   902→   902→    OVER35_HYBRID_THRESHOLDS.bttsProb.floor
   903→   903→  );
   904→   904→

   906→   906→  const sbO25Prob = findOptimalProbThreshold(

   909→   909→    [50, 85],
   910→   910→    1,
   911→   911→    STRONG_BET_HYBRID.o25Prob.floor

   918→   918→    1,

   989→   989→export function deriveSimpleThresholdsFromBacktest(

   993→   993→  options: BacktestDerivationOptions = {}
   994→   994→): LeagueBacktestThresholds | null {
   995→   995→  const { minSampleSize = 150 } = options;
   996→   996→
   997→   997→  if (matches.length < minSampleSize) return null;

  1000→  1000→  // separate BTTS from non-BTTS and O3.5 from non-O3.5 matches.

  1002→  1002→  // the Poisson BTTS probability is already a strong predictor.
  1003→  1003→

  1013→  1013→    const threshold = Math.max(BTTS_HYBRID_THRESHOLDS.modelBttsProb.floor, currentBaselines.bttsRate * mult);
  1014→  1014→    const aboveCount = bttsMatches.filter(() => currentBaselines.bttsRate >= threshold).length;

  1017→  1017→    const estimatedAccuracy = bttsMatches.length > 0
  1018→  1018→      ? bttsMatches.filter(() => currentBaselines.bttsRate >= threshold).length / Math.max(matches.length, 1)
  1019→  1019→      : 0;
  1020→  1020→    if (estimatedAccuracy > bestBttsAcc) {
  1021→  1021→      bestBttsAcc = estimatedAccuracy;
  1022→  1022→      bestBttsMult = mult;
  1023→  1023→    }
  1024→  1024→  }
  1025→  1025→

  1027→  1027→  // with the derived multiplier, falling back to defaults.

  1029→  1029→  const bttsRate = currentBaselines.bttsRate;

  1040→  1040→      awayAvgGoals: Math.max(BTTS_HYBRID_THRESHOLDS.awayAvgGoals.floor, currentBaselines.avgAwayGoals * BTTS_HYBRID_THRESHOLDS.awayAvgGoals.multiplier),

  1043→  1043→    },
  1044→  1044→    over35: {

  1046→  1046→      bttsProb: Math.max(OVER35_HYBRID_THRESHOLDS.bttsProb.floor, currentBaselines.bttsRate * OVER35_HYBRID_THRESHOLDS.bttsProb.multiplier),
  1047→  1047→      homeAvgGoals: Math.max(OVER35_HYBRID_THRESHOLDS.homeAvgGoals.floor, currentBaselines.avgHomeGoals * OVER35_HYBRID_THRESHOLDS.homeAvgGoals.multiplier),

  1071→  1071→// ============================================================================
  1072→  1072→// Bookie Odds Dampener — Heavy Favorite Adjustment
  1073→  1073→// ============================================================================
  1074→  1074→// Based on live results analysis: when bookie odds for one side are <=1.70,
  1075→  1075→// the model systematically overestimates O2.5 probability by 10-15%.
  1076→  1076→// Heavy favorites tend to win tight, controlling games (1-0, 2-0) rather
  1077→  1077→// than high-scoring shootouts. This dampener accounts for that pattern.
  1078→  1078→// ============================================================================
  1079→  1079→
  1080→  1080→/** Heavy favorite odds threshold (decimal odds) */
  1081→  1081→const HEAVY_FAVORITE_ODDS = 1.70;
  1082→  1082→
  1083→  1083→/** Maximum dampener amount (percentage points to subtract) */
  1084→  1084→const HEAVY_FAVORITE_DAMPENER = 10;
  1085→  1085→
  1086→  1086→/**
  1087→  1087→ * Apply bookie odds dampener to goal market probabilities.
  1088→  1088→ * When either team is a heavy favorite (avg odds <=1.70), reduce O2.5/O3.5/BTTS
  1089→  1089→ * probabilities to account for the "park the bus" effect.
  1090→  1090→ *
  1091→  1091→ * Only applies when valid odds are available (non-null, positive).
  1092→  1092→ *
  1093→  1093→ * @param over25Prob  - Model O2.5 probability (%)
  1094→  1094→ * @param over35Prob  - Model O3.5 probability (%)
  1095→  1095→ * @param bttsProb    - Model BTTS probability (%)
  1096→  1096→ * @param homeAvgOdds - Home team average odds (decimal)
  1097→  1097→ * @param awayAvgOdds - Away team average odds (decimal)
  1098→  1098→ * @returns Adjusted probabilities with dampener applied
  1099→  1099→ */
  1100→  1100→export function applyBookieOddsDampener(
  1101→  1101→  over25Prob: number,
  1102→  1102→  over35Prob: number,
  1103→  1103→  bttsProb: number,
  1104→  1104→  homeAvgOdds: number | null | undefined,
  1105→  1105→  awayAvgOdds: number | null | undefined,
  1106→  1106→  leagueAvgGoalsPerGame?: number
  1107→  1107→): { over25: number; over35: number; btts: number; dampened: boolean; reason: string } {
  1108→  1108→  if (!homeAvgOdds || !awayAvgOdds || homeAvgOdds <= 0 || awayAvgOdds <= 0) {
  1109→  1109→    return { over25: over25Prob, over35: over35Prob, btts: bttsProb, dampened: false, reason: '' };
  1110→  1110→  }
  1111→  1111→
  1112→  1112→  // League-adaptive: gentler dampener in high-variance leagues
  1113→  1113→  const variance = leagueAvgGoalsPerGame ? classifyLeagueVariance(leagueAvgGoalsPerGame) : 'low';
  1114→  1114→  const heavyThreshold = variance === 'high' ? 1.40 : variance === 'medium' ? 1.60 : HEAVY_FAVORITE_ODDS;
  1115→  1115→  const maxDampener = variance === 'high' ? 5 : variance === 'medium' ? 7 : HEAVY_FAVORITE_DAMPENER;
  1116→  1116→
  1117→  1117→  const homeIsHeavy = homeAvgOdds <= heavyThreshold;
  1118→  1118→  const awayIsHeavy = awayAvgOdds <= heavyThreshold;
  1119→  1119→
  1120→  1120→  if (!homeIsHeavy && !awayIsHeavy) {
  1121→  1121→    return { over25: over25Prob, over35: over35Prob, btts: bttsProb, dampened: false, reason: '' };
  1122→  1122→  }
  1123→  1123→
  1124→  1124→  const heavyOdds = Math.min(homeAvgOdds, awayAvgOdds);
  1125→  1125→  // Scale dampener: heavier favorite = bigger reduction
  1126→  1126→  // At 1.17 (PSG level): full dampener
  1127→  1127→  // At threshold: minimal 2% dampener
  1128→  1128→  const minDampener = variance === 'high' ? 2 : 3;
  1129→  1129→  const scale = Math.max(0, 1 - (heavyOdds - 1.17) / (heavyThreshold - 1.17));
  1130→  1130→  const dampener = Math.round(minDampener + (maxDampener - minDampener) * scale);
  1131→  1131→
  1132→  1132→  const leagueLabel = variance.charAt(0).toUpperCase() + variance.slice(1);
  1133→  1133→  const reason = homeIsHeavy
  1134→  1134→    ? `${leagueLabel}-variance: Heavy home favorite (${homeAvgOdds.toFixed(2)}): -${dampener}% dampener`
  1135→  1135→    : `${leagueLabel}-variance: Heavy away favorite (${awayAvgOdds.toFixed(2)}): -${dampener}% dampener`;
  1136→  1136→
  1137→  1137→  return {
  1138→  1138→    over25: Math.max(30, over25Prob - dampener),
  1139→  1139→    over35: Math.max(20, over35Prob - dampener),
  1140→  1140→    btts: Math.max(25, bttsProb - Math.round(dampener * 0.5)), // BTTS dampened less
  1141→  1141→    dampened: true,
  1142→  1142→    reason,
  1143→  1143→  };
  1144→  1144→}
  1145→  1145→
  1146→  1146→/**
  1147→  1147→ * Compute average bookie odds for a team from match results.
  1148→  1148→ * Averages odds across all matches for that team.
  1149→  1149→ *
  1150→  1150→ * @param matches - Match results with odds data
  1151→  1151→ * @param team    - Team name
  1152→  1152→ * @returns Average decimal odds, or null if insufficient data
  1153→  1153→ */
  1154→  1154→export function computeTeamAvgOdds(
  1155→  1155→  matches: { homeTeam: string; awayTeam: string; oddsAvgHome: number | null; oddsAvgAway: number | null }[],
  1156→  1156→  team: string
  1157→  1157→): number | null {
  1158→  1158→  const teamOdds: number[] = [];
  1159→  1159→
  1160→  1160→  for (const m of matches) {
  1161→  1161→    const odds = m.homeTeam === team ? m.oddsAvgHome : (m.awayTeam === team ? m.oddsAvgAway : null);
  1162→  1162→    if (odds && odds > 1 && odds < 20) {
  1163→  1163→      teamOdds.push(odds);
  1164→  1164→    }
  1165→  1165→  }
  1166→  1166→
  1167→  1167→  if (teamOdds.length < 3) return null;
  1168→  1168→
  1169→  1169→  return teamOdds.reduce((s, o) => s + o, 0) / teamOdds.length;
  1170→  1170→}

  1172→  1172→// ============================================================================
  1173→  1173→// League Variance Classification — Used by blowout/dampener adaptivity
  1174→  1174→// ============================================================================
  1175→  1175→// Classifies a league by its historical goal variance so that blowout and
  1176→  1176→// dampener thresholds can be calibrated to the league's natural scoring profile.
  1177→  1177→// High-variance leagues (2. Bundesliga, Eredivisie) see more goal explosions
  1178→  1178→// from heavy favorites, so suppression mechanisms should be gentler.
  1179→  1179→
  1180→  1180→export type LeagueVariance = 'high' | 'medium' | 'low';
  1181→  1181→
  1182→  1182→export function classifyLeagueVariance(avgGoalsPerGame: number): LeagueVariance {
  1183→  1183→  if (avgGoalsPerGame >= 3.0) return 'high';
  1184→  1184→  if (avgGoalsPerGame >= 2.5) return 'medium';
  1185→  1185→  return 'low';
  1186→  1186→}
  1187→  1187→
  1188→  1188→// ============================================================================
  1189→  1189→// Blowout Risk Qualifier — Override Under signals in massive quality gaps
  1190→  1190→// ============================================================================
  1191→  1191→// Based on live results analysis: when there's a huge quality gap (favorite's
  1192→  1192→// avg odds <= 1.50), the Regression "Under" signal is unreliable. Heavy favorites
  1193→  1193→// tend to dominate regardless of recent form — a "cooling off" period doesn't
  1194→  1194→// matter when you're 2-3 quality tiers above the opponent.
  1195→  1195→//
  1196→  1196→// This qualifier prevents the "Under" regression signal from canceling out
  1197→  1197→// legitimate "Over" signals (xG, Z-Score) in STRONG BET / GREY RESULT / GOAL FEST.
  1198→  1198→//
  1199→  1199→// Example: Man City (odds 1.20) vs Luton — Regression says "Under" (City only
  1200→  1200→// scored 1 goal last 3 games), but City will likely score 3-4 regardless.
  1201→  1201→// Without this qualifier, the Under signal would block 2 points in STRONG BET.
  1202→  1202→// ============================================================================
  1203→  1203→
  1204→  1204→/** Blowout threshold: favorite's average odds at or below this triggers override */
  1205→  1205→const BLOWOUT_ODDS_THRESHOLD = 1.50;
  1206→  1206→
  1207→  1207→/**
  1208→  1208→ * Apply blowout risk qualifier to regression signal.
  1209→  1209→ * When either team is a massive favorite (avg odds <= 1.50), override
  1210→  1210→ * Regression "Under"/"Strong Under" to "Neutral" because quality advantage
  1211→  1211→ * outweighs recent form in one-sided matchups.
  1212→  1212→ *
  1213→  1213→ * @param signals      - Current signal set (xgSignal, regressionSignal, zScoreSignal)
  1214→  1214→ * @param homeAvgOdds  - Home team average odds (decimal)
  1215→  1215→ * @param awayAvgOdds  - Away team average odds (decimal)
  1216→  1216→ * @returns Modified signals (regression may be overridden to Neutral)
  1217→  1217→ */
  1218→  1218→export function applyBlowoutQualifier(
  1219→  1219→  signals: SignalInput,
  1220→  1220→  homeAvgOdds: number | null | undefined,
  1221→  1221→  awayAvgOdds: number | null | undefined,
  1222→  1222→  leagueAvgGoalsPerGame?: number
  1223→  1223→): { signals: SignalInput; blowoutOverride: boolean; reason: string } {
  1224→  1224→  // Only applies when valid odds are available
  1225→  1225→  if (!homeAvgOdds || !awayAvgOdds || homeAvgOdds <= 0 || awayAvgOdds <= 0) {
  1226→  1226→    return { signals, blowoutOverride: false, reason: '' };
  1227→  1227→  }
  1228→  1228→
  1229→  1229→  // League-adaptive threshold: gentler in high-variance leagues
  1230→  1230→  const variance = leagueAvgGoalsPerGame ? classifyLeagueVariance(leagueAvgGoalsPerGame) : 'low';
  1231→  1231→  const blowoutThreshold = variance === 'high' ? 1.25 : variance === 'medium' ? 1.40 : BLOWOUT_ODDS_THRESHOLD;
  1232→  1232→
  1233→  1233→  const homeIsMassiveFavorite = homeAvgOdds <= blowoutThreshold;
  1234→  1234→  const awayIsMassiveFavorite = awayAvgOdds <= blowoutThreshold;
  1235→  1235→
  1236→  1236→  // Only trigger when one team is a massive favorite
  1237→  1237→  if (!homeIsMassiveFavorite && !awayIsMassiveFavorite) {
  1238→  1238→    return { signals, blowoutOverride: false, reason: '' };
  1239→  1239→  }
  1240→  1240→
  1241→  1241→  // In high-variance leagues, only override Strong Under (keep mild signals intact)
  1242→  1242→  // because hot favorites in goal-rich leagues often score freely
  1243→  1243→  const shouldOverride = variance === 'high'
  1244→  1244→    ? signals.regressionSignal === 'Strong Under'
  1245→  1245→    : signals.regressionSignal === 'Under' || signals.regressionSignal === 'Strong Under';
  1246→  1246→  if (!shouldOverride) {
  1247→  1247→    return { signals, blowoutOverride: false, reason: '' };
  1248→  1248→  }
  1249→  1249→
  1250→  1250→  const favoriteOdds = Math.min(homeAvgOdds, awayAvgOdds);
  1251→  1251→  const favorite = homeIsMassiveFavorite ? 'Home' : 'Away';
  1252→  1252→  const reason = `Blowout qualifier (${variance}-variance): ${favorite} favorite (${favoriteOdds.toFixed(2)} <= ${blowoutThreshold}) — Regression "${signals.regressionSignal}" overridden to Neutral`;
  1253→  1253→
  1254→  1254→  return {
  1255→  1255→    signals: {
  1256→  1256→      ...signals,
  1257→  1257→      regressionSignal: 'Neutral',
  1258→  1258→    },
  1259→  1259→    blowoutOverride: true,
  1260→  1260→    reason,
  1261→  1261→  };
  1262→  1262→}

  1268→  1268→// Under + Z-Score Neutral + xG Over) correctly identifies "goals are coming"
  1269→  1269→// but tends to surface ONE-SIDED regressions (one team due for a breakout).
  1270→  1270→// This system evaluates each team independently and produces a BTTS-specific
  1271→  1271→// conviction tier that is separate from the Over goal thesis.

  1285→  1285→  /** Away team xG overperformance diff */

  1311→  1311→/**
  1312→  1312→ * Classify a per-team regression deviation into a signal label.
  1313→  1313→ * Uses the same thresholds as REGRESSION_THRESHOLDS.perTeam.

  1315→  1315→export function classifyPerTeamRegressionSignal(combinedSignal: number): string {
  1316→  1316→  if (combinedSignal <= -0.8) return 'Strong Over';
  1317→  1317→  if (combinedSignal <= -0.3) return 'Over';
  1318→  1318→  if (combinedSignal >= 0.8) return 'Strong Under';
  1319→  1319→  if (combinedSignal >= 0.3) return 'Under';
  1320→  1320→  return 'Neutral';
  1321→  1321→}
  1322→  1322→
  1323→  1323→/**
  1324→  1324→ * Classify a per-team xG diff into a signal label.
  1325→  1325→ * Uses the same thresholds as XG_THRESHOLDS applied per-team.
  1326→  1326→ */
  1327→  1327→export function classifyPerTeamXgSignal(xgDiff: number): string {
  1328→  1328→  if (xgDiff <= -1.0) return 'Strong Over';
  1329→  1329→  if (xgDiff <= -0.5) return 'Over';
  1330→  1330→  if (xgDiff >= 1.0) return 'Strong Under';
  1331→  1331→  if (xgDiff >= 0.5) return 'Under';
  1332→  1332→  return 'Neutral';
  1333→  1333→}
  1334→  1334→
  1335→  1335→/**
  1336→  1336→ * Classify a per-team Z-Score into a signal label.
  1337→  1337→ * Negative Z = underperforming recently (goals expected to rise).
  1338→  1338→ */
  1339→  1339→export function classifyPerTeamZScoreSignal(zScore: number): string {
  1340→  1340→  if (zScore <= -1.5) return 'Strong Over';
  1341→  1341→  if (zScore <= -1.0) return 'Over';
  1342→  1342→  if (zScore >= 1.5) return 'Strong Under';
  1343→  1343→  if (zScore >= 1.0) return 'Under';
  1344→  1344→  return 'Neutral';
  1345→  1345→}

  1348→  1348→// Momentum Signal — Parallel track to Regression (form continuation)
  1349→  1349→// ============================================================================
  1350→  1350→// While Regression asks "is this team due to bounce back?", Momentum asks
  1351→  1351→// "is this team on a sustained hot streak that will continue?"
  1352→  1352→// Both tracks run simultaneously; a match can qualify through either pathway.
  1353→  1353→
  1354→  1354→export interface MomentumTeamInput {
  1355→  1355→  last5Goals: number[];    // total goals per match in last 5
  1356→  1356→  seasonAvg: number;       // season avg total goals per game
  1357→  1357→  xgDiff: number;          // actualGoals/matches - totalXg/matches (positive = overperforming)
  1358→  1358→  sotPerGame: number | null;
  1359→  1359→}
  1360→  1360→
  1361→  1361→export interface MomentumResult {
  1362→  1362→  homeSignal: string;      // 'Strong Momentum' | 'Momentum' | 'Neutral' | 'Cooling' | 'Cold'
  1363→  1363→  awaySignal: string;
  1364→  1364→  matchSignal: string;     // 'MOMENTUM OVER' | 'MOMENTUM LEAN OVER' | 'MOMENTUM NEUTRAL' | 'MOMENTUM LEAN UNDER' | 'MOMENTUM UNDER'
  1365→  1365→}
  1366→  1366→
  1367→  1367→function classifyTeamMomentum(team: MomentumTeamInput): string {
  1368→  1368→  if (team.last5Goals.length < 3) return 'Neutral';
  1369→  1369→
  1370→  1370→  const last5Avg = team.last5Goals.reduce((s, g) => s + g, 0) / team.last5Goals.length;
  1371→  1371→  const last3 = team.last5Goals.slice(0, Math.min(3, team.last5Goals.length));
  1372→  1372→  const last3Avg = last3.reduce((s, g) => s + g, 0) / last3.length;
  1373→  1373→
  1374→  1374→  // Consistency: coefficient of variation of last 5
  1375→  1375→  const mean = last5Avg;
  1376→  1376→  const variance = team.last5Goals.reduce((s, g) => s + (g - mean) ** 2, 0) / team.last5Goals.length;
  1377→  1377→  const stdDev = Math.sqrt(variance);
  1378→  1378→  const cv = mean > 0 ? stdDev / mean : 999;
  1379→  1379→
  1380→  1380→  // Sustained: 5+ game window above season average by meaningful margin
  1381→  1381→  const sustained = last5Avg > team.seasonAvg * 1.15;
  1382→  1382→
  1383→  1383→  // Accelerating: last 3 even hotter than last 5
  1384→  1384→  const accelerating = last3Avg > last5Avg * 1.05;
  1385→  1385→
  1386→  1386→  // xG sustainability: overperforming xG with shot volume to back it
  1387→  1387→  const xgSustainable = team.xgDiff > 0 && team.sotPerGame !== null && team.sotPerGame > 5.0;
  1388→  1388→
  1389→  1389→  // Strong Momentum: sustained + consistent + high output
  1390→  1390→  if (sustained && cv < 0.5 && last5Avg >= 2.0) return 'Strong Momentum';
  1391→  1391→  // Momentum: sustained with moderate variance, or recent surge with sustainable backing
  1392→  1392→  if (sustained && cv < 0.65) return 'Momentum';
  1393→  1393→  if (last5Avg > team.seasonAvg * 1.15 && accelerating && (xgSustainable || team.xgDiff <= 0)) return 'Momentum';
  1394→  1394→  // Cold: declining output
  1395→  1395→  if (last5Avg < team.seasonAvg * 0.75) {
  1396→  1396→    return last3Avg < last5Avg * 0.8 ? 'Cold' : 'Cooling';
  1397→  1397→  }
  1398→  1398→  return 'Neutral';
  1399→  1399→}

  1401→  1401→export function computeMomentumSignal(home: MomentumTeamInput, away: MomentumTeamInput): MomentumResult {
  1402→  1402→  const homeSignal = classifyTeamMomentum(home);
  1403→  1403→  const awaySignal = classifyTeamMomentum(away);
  1404→  1404→
  1405→  1405→  const isOver = (s: string) => s === 'Strong Momentum' || s === 'Momentum';
  1406→  1406→  const isUnder = (s: string) => s === 'Cold' || s === 'Cooling';
  1407→  1407→
  1408→  1408→  let matchSignal: string;
  1409→  1409→  if (isOver(homeSignal) && isOver(awaySignal)) matchSignal = 'MOMENTUM OVER';
  1410→  1410→  else if ((isOver(homeSignal) || isOver(awaySignal)) && !isUnder(homeSignal) && !isUnder(awaySignal)) matchSignal = 'MOMENTUM LEAN OVER';
  1411→  1411→  else if (isUnder(homeSignal) && isUnder(awaySignal)) matchSignal = 'MOMENTUM UNDER';
  1412→  1412→  else if (isUnder(homeSignal) || isUnder(awaySignal)) matchSignal = 'MOMENTUM LEAN UNDER';
  1413→  1413→  else matchSignal = 'MOMENTUM NEUTRAL';
  1414→  1414→
  1415→  1415→  return { homeSignal, awaySignal, matchSignal };
  1416→  1416→}

  1418→  1418→// ============================================================================
  1419→  1419→// BTTS Dual-Team Qualification System
  1420→  1420→// ============================================================================
  1421→  1421→// Evaluates each team independently to determine whether BOTH teams are likely
  1422→  1422→// to score. Returns a tier that caps BTTS probability independently from the
  1423→  1423→// Over goal thesis.
  1424→  1424→//
  1425→  1425→// Now supports TWO qualification pathways:
  1426→  1426→//   1. Regression Under: both teams underperforming → due to bounce back
  1427→  1427→//   2. Momentum Over: both teams on hot streaks → likely to keep scoring
  1428→  1428→//
  1429→  1429→// Tier mapping:
  1430→  1430→//   5+ → BTTS STRONG   (cap 70%) — both teams individually qualify
  1431→  1431→//   3–4 → BTTS QUALIFIED (cap 55%) — mild BTTS confidence
  1432→  1432→//   1–2 → BTTS WEAK     (cap 40%) — one-sided regression likely
  1433→  1433→//   ≤0  → BTTS AVOID    (cap 25%) — do not bet BTTS
  1434→  1434→// ============================================================================
  1435→  1435→export function computeBTTSQualification(input: BttsQualificationInput): BttsQualificationResult {
  1436→  1436→  let score = 0;
  1437→  1437→  const checks: { check: string; points: number; passed: boolean }[] = [];
  1438→  1438→
  1439→  1439→  const isRegressionUnder = (sig: string) => sig === 'Under' || sig === 'Strong Under';
  1440→  1440→  const isXgOver = (diff: number) => diff <= -0.5; // underperforming xG = goals due
  1441→  1441→  const isZCold = (z: number) => z >= 1.0; // overperforming Z = cold streak coming
  1442→  1442→  const isZNeutralOrBetter = (z: number) => z < 1.0; // not overheated
  1443→  1443→
  1444→  1444→  // Momentum helpers
  1445→  1445→  const isMomentum = (s: string | undefined) => s === 'Strong Momentum' || s === 'Momentum';
  1446→  1446→  const homeMom = isMomentum(input.homeMomentumSignal);
  1447→  1447→  const awayMom = isMomentum(input.awayMomentumSignal);
  1448→  1448→  const bothMom = homeMom && awayMom;
  1449→  1449→  const oneMom = homeMom || awayMom;
  1450→  1450→
  1451→  1451→  // Check 1: Regression Under OR Momentum Over — dual qualification pathway
  1452→  1452→  const bothRegUnder = isRegressionUnder(input.homeRegressionSignal) && isRegressionUnder(input.awayRegressionSignal);
  1453→  1453→  const oneRegUnder = isRegressionUnder(input.homeRegressionSignal) || isRegressionUnder(input.awayRegressionSignal);
  1454→  1454→
  1455→  1455→  if (bothRegUnder && bothMom) {
  1456→  1456→    // Supercharged: both signals agree from both sides
  1457→  1457→    score += 5;
  1458→  1458→    checks.push({ check: 'Both Regression Under AND Momentum Over (supercharged)', points: 5, passed: true });
  1459→  1459→  } else if (bothRegUnder || bothMom) {
  1460→  1460→    score += 3;
  1461→  1461→    checks.push({ check: bothRegUnder ? 'Both teams Regression Under/StrongUnder' : 'Both teams Momentum Over (hot streak)', points: 3, passed: true });
  1462→  1462→  } else if ((oneRegUnder && oneMom)) {
  1463→  1463→    score += 2; // Mixed but both signal types point to goals
  1464→  1464→    checks.push({ check: 'One Regression Under + One Momentum Over', points: 2, passed: true });
  1465→  1465→  } else if (oneRegUnder || oneMom) {
  1466→  1466→    score += 1;
  1467→  1467→    checks.push({ check: oneRegUnder ? 'One team Regression Under' : 'One team Momentum Over', points: 1, passed: true });
  1468→  1468→  } else {
  1469→  1469→    checks.push({ check: 'No Regression Under or Momentum Over', points: 3, passed: false });
  1470→  1470→  }
  1471→  1471→
  1472→  1472→  // Check 2: BOTH teams xG Over (underperforming xG = creating chances, finishing due)
  1473→  1473→  const bothXgOver = isXgOver(input.homeXgDiff) && isXgOver(input.awayXgDiff);
  1474→  1474→  if (bothXgOver) {
  1475→  1475→    score += 2;
  1476→  1476→    checks.push({ check: 'Both teams xG Over (underperforming)', points: 2, passed: true });
  1477→  1477→  } else {
  1478→  1478→    checks.push({ check: 'Both teams xG Over (underperforming)', points: 2, passed: false });
  1479→  1479→  }
  1480→  1480→
  1481→  1481→  // Check 3: BOTH teams Z-Score not overheated
  1482→  1482→  const bothZOk = isZNeutralOrBetter(input.homeZScore) && isZNeutralOrBetter(input.awayZScore);
  1483→  1483→  if (bothZOk) {
  1484→  1484→    score += 1;
  1485→  1485→    checks.push({ check: 'Both teams Z-Score Neutral/better', points: 1, passed: true });
  1486→  1486→  } else {
  1487→  1487→    checks.push({ check: 'Both teams Z-Score Neutral/better', points: 1, passed: false });
  1488→  1488→  }
  1489→  1489→
  1490→  1490→  // Check 4: Competitive matchup (favorite odds 1.80–2.50)
  1491→  1491→  if (input.favoriteOdds && input.favoriteOdds >= 1.80 && input.favoriteOdds <= 2.50) {
  1492→  1492→    score += 1;
  1493→  1493→    checks.push({ check: 'Competitive matchup (odds 1.80-2.50)', points: 1, passed: true });
  1494→  1494→  } else {
  1495→  1495→    checks.push({ check: 'Competitive matchup (odds 1.80-2.50)', points: 1, passed: false });
  1496→  1496→  }
  1497→  1497→
  1498→  1498→  // Penalty 5: Massive favorite (≤1.50) — one side dominates, BTTS unlikely
  1499→  1499→  if (input.favoriteOdds && input.favoriteOdds <= 1.50) {
  1500→  1500→    score -= 2;
  1501→  1501→    checks.push({ check: 'Massive favorite (odds ≤1.50)', points: -2, passed: true });
  1502→  1502→  } else {
  1503→  1503→    checks.push({ check: 'Massive favorite (odds ≤1.50)', points: -2, passed: false });
  1504→  1504→  }
  1505→  1505→
  1506→  1506→  // Penalty 6: Either team can't finish (SOT conversion <25%)
  1507→  1507→  const homeCantFinish = input.homeSotConversion !== null && input.homeSotConversion < 25;
  1508→  1508→  const awayCantFinish = input.awaySotConversion !== null && input.awaySotConversion < 25;
  1509→  1509→  if (homeCantFinish) score -= 1;
  1510→  1510→  if (awayCantFinish) score -= 1;
  1511→  1511→  checks.push({
  1512→  1512→    check: `Home SOT conversion ${homeCantFinish ? '< 25%' : '≥ 25%'}${input.homeSotConversion !== null ? ` (${input.homeSotConversion.toFixed(0)}%)` : ' (N/A)'}`,
  1513→  1513→    points: -1,
  1514→  1514→    passed: !homeCantFinish,
  1515→  1515→  });
  1516→  1516→  checks.push({
  1517→  1517→    check: `Away SOT conversion ${awayCantFinish ? '< 25%' : '≥ 25%'}${input.awaySotConversion !== null ? ` (${input.awaySotConversion.toFixed(0)}%)` : ' (N/A)'}`,
  1518→  1518→    points: -1,
  1519→  1519→    passed: !awayCantFinish,
  1520→  1520→  });
  1521→  1521→
  1522→  1522→  // Penalty 7: Either team in a strong cold Z-Score streak
  1523→  1523→  const homeCold = isZCold(input.homeZScore);
  1524→  1524→  const awayCold = isZCold(input.awayZScore);
  1525→  1525→  if (homeCold) score -= 2;
  1526→  1526→  if (awayCold) score -= 2;
  1527→  1527→  checks.push({
  1528→  1528→    check: `Home Z-Score ${homeCold ? 'cold (≥1.0)' : 'ok'}`,
  1529→  1529→    points: -2,
  1530→  1530→    passed: !homeCold,
  1531→  1531→  });
  1532→  1532→  checks.push({
  1533→  1533→    check: `Away Z-Score ${awayCold ? 'cold (≥1.0)' : 'ok'}`,
  1534→  1534→    points: -2,
  1535→  1535→    passed: !awayCold,
  1536→  1536→  });
  1537→  1537→
  1538→  1538→  // Determine tier
  1539→  1539→  let tier: BttsQualificationTier;
  1540→  1540→  let bttsCap: number;
  1541→  1541→  if (score >= 5) {
  1542→  1542→    tier = 'BTTS STRONG';
  1543→  1543→    bttsCap = 70;
  1544→  1544→  } else if (score >= 3) {
  1545→  1545→    tier = 'BTTS QUALIFIED';
  1546→  1546→    bttsCap = 55;
  1547→  1547→  } else if (score >= 1) {
  1548→  1548→    tier = 'BTTS WEAK';
  1549→  1549→    bttsCap = 40;
  1550→  1550→  } else {
  1551→  1551→    tier = 'BTTS AVOID';
  1552→  1552→    bttsCap = 25;
  1553→  1553→  }
  1554→  1554→
  1555→  1555→  return { tier, score, breakdown: checks, bttsCap };
  1556→  1556→}

  1625→  1625→    score += 1;

  1633→  1633→  if (bothRegUnder) {
  1634→  1634→    score += 2;
  1635→  1635→    checks.push({ check: 'Both teams Regression Under', points: 2, passed: true });
  1636→  1636→  } else {
  1637→  1637→    checks.push({ check: 'Both teams Regression Under', points: 2, passed: false });
  1638→  1638→  }

  1641→  1641→  const bothXgOver = isXgOver(input.homeXgSignal) && isXgOver(input.awayXgSignal);
  1642→  1642→  if (bothXgOver) {
  1643→  1643→    score += 2;

  1664→  1664→  if (xgBalanced) {

  1673→  1673→    score += 1;

  1675→  1675→  } else {
  1676→  1676→    checks.push({ check: 'Not a competitive matchup', points: 1, passed: false });
  1677→  1677→  }
  1678→  1678→

  1680→  1680→  if (!xgBalanced) {
  1681→  1681→    score -= 1;
  1682→  1682→    // Already counted in check 5 as non-passed, no duplicate
  1683→  1683→  }
  1684→  1684→

  1700→  1700→    points: -1,

  1704→  1704→    check: `Away SOT conversion ${awayCantFinish ? '< 25%' : '≥ 25%'}${input.awaySotConversion !== null ? ` (${input.awaySotConversion.toFixed(0)}%)` : ' (N/A)'}`,
  1705→  1705→    points: -1,
  1706→  1706→    passed: !awayCantFinish,
  1707→  1707→  });
  1708→  1708→
  1709→  1709→  // Determine tier
  1710→  1710→  let tier: GoalTier;
  1711→  1711→  let recommendation: string;
  1712→  1712→  if (score >= 6) {

  1714→  1714→    recommendation = 'Full confidence on Over 2.5, consider Over 3.5';
  1715→  1715→  } else if (score >= 4) {
  1716→  1716→    tier = 'GOAL LIKELY';
  1717→  1717→    recommendation = 'Solid Over 2.5 bet';
  1718→  1718→  } else if (score >= 2) {
  1719→  1719→    tier = 'GOAL BORDERLINE';
  1720→  1720→    recommendation = 'Over 1.5 strong, Over 2.5 cautious';
  1721→  1721→  } else if (score >= 0) {
  1722→  1722→    tier = 'GOAL THIN';
  1723→  1723→    recommendation = 'Over 1.5 only, skip Over 2.5';
  1724→  1724→  } else {
  1725→  1725→    tier = 'GOAL STALL';
  1726→  1726→    recommendation = 'Likely 1-2 goals, avoid Over markets above 1.5';
  1727→  1727→  }
  1728→  1728→
  1729→  1729→  return { tier, score, breakdown: checks, recommendation };
  1730→  1730→}

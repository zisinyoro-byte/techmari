    24→    24→/**
    25→    25→ * Compute league baselines from results + analytics data.
    26→    26→ * Call this once per league/season, pass the result into all filter functions.
    27→    27→ */
    28→    28→export function computeLeagueBaselines(
    29→    29→  results: { ftHomeGoals: number; ftAwayGoals: number }[],
    30→    30→  analytics: {
    31→    31→    avgGoalsPerGame: number;
    32→    32→    over25Percent: number;
    33→    33→    avgHomeGoals: number;
    34→    34→    avgAwayGoals: number;
    35→    35→    overallShotConversion: string | number;
    36→    36→  }
    37→    37→): LeagueBaselines {
    38→    38→  const total = results.length || 1;
    39→    39→  const bttsCount = results.filter(r => r.ftHomeGoals > 0 && r.ftAwayGoals > 0).length;
    40→    40→  const over35Count = results.filter(r => r.ftHomeGoals + r.ftAwayGoals > 3.5).length;
    41→    41→
    42→    42→  return {
    43→    43→    avgGoalsPerGame: analytics.avgGoalsPerGame,
    44→    44→    over25Rate: analytics.over25Percent,
    45→    45→    bttsRate: (bttsCount / total) * 100,
    46→    46→    over35Rate: (over35Count / total) * 100,
    47→    47→    avgHomeGoals: analytics.avgHomeGoals,
    48→    48→    avgAwayGoals: analytics.avgAwayGoals,
    49→    49→    shotConversion: typeof analytics.overallShotConversion === 'string'
    50→    50→      ? parseFloat(analytics.overallShotConversion)
    51→    51→      : analytics.overallShotConversion,
    52→    52→  };
    53→    53→}

   114→   114→// ============================================================================
   115→   115→// Backtest-derived threshold registry
   116→   116→// ============================================================================
   117→   117→// This is the central store. Thresholds are added here when backtests run.
   118→   118→// Leagues without entries fall back to the hybrid formula.
   119→   119→// Supports localStorage persistence for client-side survival across page reloads.
   120→   120→// ============================================================================
   121→   121→
   122→   122→const STORAGE_KEY = 'techmari_backtest_thresholds';
   123→   123→
   124→   124→const backtestRegistry = new Map<string, LeagueBacktestThresholds>();
   125→   125→
   126→   126→let loadedFromStorage = false;
   127→   127→
   128→   128→/**
   129→   129→ * Load thresholds from localStorage into the in-memory registry.
   130→   130→ * Safe to call multiple times — only loads once per session.
   131→   131→ */
   132→   132→function loadFromLocalStorage(): void {
   133→   133→  if (loadedFromStorage || typeof window === 'undefined') return;
   134→   134→  loadedFromStorage = true;
   135→   135→  try {
   136→   136→    const raw = localStorage.getItem(STORAGE_KEY);
   137→   137→    if (!raw) return;
   138→   138→    const data = JSON.parse(raw);
   139→   139→    if (data.version !== 1) return;
   140→   140→    const entries: Record<string, LeagueBacktestThresholds> = data.thresholds;
   141→   141→    for (const [key, thresholds] of Object.entries(entries)) {
   142→   142→      backtestRegistry.set(key, thresholds);
   143→   143→    }
   144→   144→    const count = Object.keys(entries).length;
   145→   145→    if (count > 0) {
   146→   146→      console.log(`[BettingFilters] Loaded ${count} persisted threshold(s) from localStorage`);
   147→   147→    }
   148→   148→  } catch {
   149→   149→    // localStorage not available or corrupt data — silent fail
   150→   150→  }
   151→   151→}
   152→   152→
   153→   153→/**
   154→   154→ * Save all in-memory thresholds to localStorage.
   155→   155→ */
   156→   156→function saveToLocalStorage(): void {
   157→   157→  if (typeof window === 'undefined') return;
   158→   158→  try {
   159→   159→    const data: Record<string, LeagueBacktestThresholds> = {};
   160→   160→    for (const [key, thresholds] of backtestRegistry.entries()) {
   161→   161→      data[key] = thresholds;
   162→   162→    }
   163→   163→    localStorage.setItem(STORAGE_KEY, JSON.stringify({
   164→   164→      version: 1,
   165→   165→      updatedAt: new Date().toISOString(),
   166→   166→      thresholds: data,
   167→   167→    }));
   168→   168→  } catch {
   169→   169→    // localStorage not available or quota exceeded — silent fail
   170→   170→  }
   171→   171→}
   172→   172→
   173→   173→/**
   174→   174→ * Register backtest-derived thresholds for a league.
   175→   175→ * Also persists to localStorage automatically.
   176→   176→ */
   177→   177→export function registerBacktestThresholds(thresholds: LeagueBacktestThresholds): void {
   178→   178→  // Ensure we've loaded from localStorage first
   179→   179→  loadFromLocalStorage();
   180→   180→  backtestRegistry.set(thresholds.leagueName.toLowerCase(), thresholds);
   181→   181→  saveToLocalStorage();
   182→   182→}

   430→   430→/**
   431→   431→ * Resolve all thresholds for a given league.
   432→   432→ * Uses backtest-derived if available (sample >= minSampleSize), otherwise hybrid.
   433→   433→ */
   434→   434→export function resolveAllThresholds(
   435→   435→  leagueName: string | undefined,
   436→   436→  baselines: LeagueBaselines,
   437→   437→  minSampleSize = 150
   438→   438→) {
   439→   439→  const bt = leagueName ? getBacktestThresholds(leagueName) : undefined;
   440→   440→  const useBacktest = bt !== undefined && bt.sampleSize >= minSampleSize;
   441→   441→
   442→   442→  // League-level thresholds (always absolute)
   443→   443→  const league: ResolvedLeagueThresholds = useBacktest
   444→   444→    ? {
   445→   445→        bttsLeagueAvgGoals: bt!.bttsLeagueAvgGoals,
   446→   446→        bttsLeagueO25Rate: bt!.bttsLeagueO25Rate,
   447→   447→        o35LeagueAvgGoals: bt!.o35LeagueAvgGoals,
   448→   448→        o35LeagueO25Rate: bt!.o35LeagueO25Rate,
   449→   449→      }
   450→   450→    : {
   451→   451→        bttsLeagueAvgGoals: BTTS_LEAGUE_THRESHOLDS.leagueAvgGoals,
   452→   452→        bttsLeagueO25Rate: BTTS_LEAGUE_THRESHOLDS.leagueO25Rate,
   453→   453→        o35LeagueAvgGoals: OVER35_LEAGUE_THRESHOLDS.leagueAvgGoals,
   454→   454→        o35LeagueO25Rate: OVER35_LEAGUE_THRESHOLDS.leagueO25Rate,
   455→   455→      };
   456→   456→
   457→   457→  const src: 'backtest' | 'hybrid' = useBacktest ? 'backtest' : 'hybrid';
   458→   458→
   459→   459→  // BTTS match-level thresholds
   460→   460→  const btts: ResolvedThresholds = useBacktest
   461→   461→    ? { ...bt!.btts, source: 'backtest' }
   462→   462→    : {

   565→   565→ * Expanded from 7 to 9 checks: added O2.5 implied probability + rolling combined scoring.
   566→   566→ * These two features were the strongest predictors in 10K-match analysis.
   567→   567→ */
   568→   568→export function computeBttsChecklist(
   569→   569→  input: ChecklistInput,
   570→   570→  resolved: ReturnType<typeof resolveAllThresholds>
   571→   571→): number {
   572→   572→  const lt = resolved.league;
   573→   573→  const rt = resolved.btts;
   574→   574→  let count = 0;
   575→   575→  if (input.avgGoalsPerGame >= lt.bttsLeagueAvgGoals) count++;
   576→   576→  if (input.over25Percent >= lt.bttsLeagueO25Rate) count++;
   577→   577→  if (input.bttsProb >= rt.modelBttsProb) count++;
   578→   578→  if (input.avgHomeGoals >= rt.homeAvgGoals) count++;
   579→   579→  if (input.avgAwayGoals >= rt.awayAvgGoals) count++;
   580→   580→  if (input.o25Prob >= rt.modelO25Prob) count++;
   581→   581→  if (input.overallShotConversion >= rt.shotConversion) count++;
   582→   582→  // NEW: O2.5 implied probability (strongest single predictor — corr 0.192)
   583→   583→  if (input.o25ImpliedProb !== null && input.o25ImpliedProb >= O25_IMPLIED_THRESHOLDS.btts) count++;
   584→   584→  // NEW: Rolling combined scoring (adds ~20pp discrimination over season averages)
   585→   585→  if (input.rollingCombinedScoring >= ROLLING_SCORING_THRESHOLDS.btts) count++;
   586→   586→  return count;
   587→   587→}

   658→   658→/**
   659→   659→ * Compute STRONG BET using points-based system with resolved thresholds.
   660→   660→ * Signal checks updated based on 7,110-match backtest analysis:
   661→   661→ *   xG = Over (mild underperformance, finishing due to click)
   662→   662→ *   Regression = Under (recently hot, goals keep flowing)
   663→   663→ *   Z-Score = Neutral (no anomaly, natural attacking football)
   664→   664→ *   Signal Divergence (xG + Regression point opposite = high-variance games)
   665→   665→ * Returns { isStrongBet, points, maxPoints, breakdown }
   666→   666→ */
   667→   667→export function computeStrongBet(
   668→   668→  checklistInput: ChecklistInput,
   669→   669→  signals: SignalInput,
   670→   670→  resolved: ReturnType<typeof resolveAllThresholds>,
   671→   671→  options?: { momentumSignal?: string; leagueBttsRate?: number }
   672→   672→): {
   673→   673→  isStrongBet: boolean;
   674→   674→  points: number;
   675→   675→  maxPoints: number;
   676→   676→  breakdown: { check: string; points: number; passed: boolean }[];
   677→   677→} {
   678→   678→  const st = resolved.strongBet;
   679→   679→  const p = STRONG_BET_POINTS;
   680→   680→
   681→   681→  const bttsCount = computeBttsChecklist(checklistInput, resolved);
   682→   682→
   683→   683→  // Signal helpers — backtest-optimized conditions
   684→   684→  const isXgMild = (sig: string) => sig === 'Over' || sig === 'Under';
   685→   685→  const isRegressionUnder = (sig: string) => sig === 'Under' || sig === 'Strong Under';
   686→   686→  const isZScoreNeutral = (sig: string) => sig === 'Neutral';
   687→   687→  // Divergence: xG points one way (Over=goals due), Regression points opposite (Under=hot)
   688→   688→  const hasSignalDivergence = (xg: string, reg: string) =>
   689→   689→    (xg === 'Over' || xg === 'Strong Over') && isRegressionUnder(reg);
   690→   690→
   691→   691→  // League-adaptive BTTS weights: reduce in BTTS-poor leagues, boost Over weights
   692→   692→  const bttsRate = options?.leagueBttsRate;
   693→   693→  const adaptiveBttsProb = bttsRate !== undefined && bttsRate < 55 ? 0 : p.btts;
   694→   694→  const adaptiveBttsChecklist = bttsRate !== undefined && bttsRate < 45 ? 1 : p.bttsChecklist;
   695→   695→  const freedPts = (p.btts - adaptiveBttsProb) + (p.bttsChecklist - adaptiveBttsChecklist);
   696→   696→  const adaptiveO25 = p.o25 + (freedPts > 1 ? 1 : 0);
   697→   697→  const adaptiveO35 = p.o35 + (freedPts > 2 ? 1 : 0);
   698→   698→
   699→   699→  // Momentum alternative: Regression Under OR Momentum Over qualifies
   700→   700→  const isMomentumOver = options?.momentumSignal === 'MOMENTUM OVER' || options?.momentumSignal === 'MOMENTUM LEAN OVER';
   701→   701→  const regressionOrMomentum = isRegressionUnder(signals.regressionSignal) || isMomentumOver;
   702→   702→
   703→   703→  const checks = [
   704→   704→    { check: `O2.5 >=${st.o25Prob.toFixed(0)}%`, points: adaptiveO25, passed: checklistInput.o25Prob >= st.o25Prob },
   705→   705→    { check: `O3.5 >=${st.o35Prob.toFixed(0)}%`, points: adaptiveO35, passed: checklistInput.o35Prob >= st.o35Prob },
   706→   706→    { check: `BTTS >=${st.bttsProb.toFixed(0)}%`, points: adaptiveBttsProb, passed: adaptiveBttsProb > 0 && checklistInput.bttsProb >= st.bttsProb },
   707→   707→    { check: `BTTS Checklist >=${st.bttsChecklistCount}/9`, points: adaptiveBttsChecklist, passed: bttsCount >= st.bttsChecklistCount },
   708→   708→    { check: 'xG = Over/Under', points: p.xgSignal, passed: isXgMild(signals.xgSignal) },
   709→   709→    { check: isMomentumOver ? 'Momentum Over (alt)' : 'Regression = Under', points: p.regressionSignal, passed: regressionOrMomentum },
   710→   710→    { check: 'Z-Score = Neutral', points: p.zScoreSignal, passed: isZScoreNeutral(signals.zScoreSignal) },
   711→   711→    { check: 'Signal Divergence', points: p.signalDivergence, passed: hasSignalDivergence(signals.xgSignal, signals.regressionSignal) },
   712→   712→    // NEW: O2.5 implied probability — strongest single predictor (corr 0.192)
   713→   713→    { check: `O2.5 Implied >=${O25_IMPLIED_THRESHOLDS.strongBet}%`, points: 1, passed: checklistInput.o25ImpliedProb !== null && checklistInput.o25ImpliedProb >= O25_IMPLIED_THRESHOLDS.strongBet },
   714→   714→    // NEW: Rolling combined scoring — adds ~20pp discrimination
   715→   715→    { check: `Rolling Scoring >=${ROLLING_SCORING_THRESHOLDS.strongBet}`, points: 1, passed: checklistInput.rollingCombinedScoring >= ROLLING_SCORING_THRESHOLDS.strongBet },
   716→   716→  ];
   717→   717→
   718→   718→  const totalPoints = checks.reduce((sum, c) => sum + (c.passed ? c.points : 0), 0);
   719→   719→  const maxPoints = checks.reduce((sum, c) => sum + c.points, 0);
   720→   720→
   721→   721→  return {
   722→   722→    isStrongBet: totalPoints >= p.threshold,
   723→   723→    points: totalPoints,
   724→   724→    maxPoints,
   725→   725→    breakdown: checks,
   726→   726→  };
   727→   727→}
   728→   728→
   729→   729→/**
   730→   730→ * Compute GREY RESULT using resolved thresholds.
   731→   731→ * Signal checks updated based on 7,110-match backtest analysis:
   732→   732→ *   Regression = Neutral or Under (normal variance or recently hot)
   733→   733→ *   Z-Score = Neutral (universal goal-fest sweet spot, Strong Over nearly impossible)
   734→   734→ *   xG = Over or Under (both mild states beat extremes)
   735→   735→ * Returns { isGreyResult, score, totalChecks, breakdown }
   736→   736→ */
   737→   737→export function computeGreyResult(
   738→   738→  checklistInput: ChecklistInput,
   739→   739→  signals: SignalInput,
   740→   740→  resolved: ReturnType<typeof resolveAllThresholds>
   741→   741→): {
   742→   742→  isGreyResult: boolean;
   743→   743→  score: number;
   744→   744→  totalChecks: number;
   745→   745→  breakdown: { check: string; passed: boolean }[];
   746→   746→} {
   747→   747→  const gt = resolved.greyResult;
   748→   748→  const bttsCount = computeBttsChecklist(checklistInput, resolved);
   749→   749→  const o35Count = computeOver35Checklist(checklistInput, resolved);
   750→   750→
   751→   751→  // Backtest-optimized signal helpers
   752→   752→  // Regression: Neutral or Under = normal variance or recently hot (75% of matches)
   753→   753→  // Z-Score: Neutral or Under = team near/below their own average (not anomalously high)
   754→   754→  // xG: Over or Under = mild deviation from xG (excludes extremes)
   755→   755→  const isRegNeutralOrUnder = (sig: string) => sig === 'Neutral' || sig === 'Under' || sig === 'Strong Under';
   756→   756→  const isZsNeutralOrUnder = (sig: string) => sig === 'Neutral' || sig === 'Under';
   757→   757→  const isXgMild = (sig: string) => sig === 'Over' || sig === 'Under';
   758→   758→
   759→   759→  const checks = [
   760→   760→    { check: 'Regression = Neutral/Under', passed: isRegNeutralOrUnder(signals.regressionSignal) },
   761→   761→    { check: 'Z-Score = Neutral/Under', passed: isZsNeutralOrUnder(signals.zScoreSignal) },
   762→   762→    { check: 'xG = Over/Under', passed: isXgMild(signals.xgSignal) },
   763→   763→    { check: `BTTS Checklist >=${gt.bttsChecklistCount}/9`, passed: bttsCount >= gt.bttsChecklistCount },
   764→   764→    { check: `BTTS ${gt.bttsProb.toFixed(0)}-${gt.bttsProbMax.toFixed(0)}%`, passed: checklistInput.bttsProb >= gt.bttsProb && checklistInput.bttsProb <= gt.bttsProbMax },
   765→   765→    { check: `O2.5 >=${gt.o25Prob.toFixed(0)}%`, passed: checklistInput.o25Prob >= gt.o25Prob },
   766→   766→    { check: `O3.5 Checklist >=${gt.over35ChecklistCount}/9`, passed: o35Count >= gt.over35ChecklistCount },
   767→   767→    { check: `O3.5 >=${gt.o35Prob.toFixed(0)}%`, passed: checklistInput.o35Prob >= gt.o35Prob },
   768→   768→    // NEW: Rolling combined scoring confirms recent goal form
   769→   769→    { check: `Rolling Scoring >=${ROLLING_SCORING_THRESHOLDS.greyResult}`, passed: checklistInput.rollingCombinedScoring >= ROLLING_SCORING_THRESHOLDS.greyResult },
   770→   770→    // NEW: O2.5 implied supports the goal thesis
   771→   771→    { check: `O2.5 Implied >=${O25_IMPLIED_THRESHOLDS.btts}%`, passed: checklistInput.o25ImpliedProb !== null && checklistInput.o25ImpliedProb >= O25_IMPLIED_THRESHOLDS.btts },
   772→   772→  ];
   773→   773→
   774→   774→  const score = checks.filter(c => c.passed).length;
   775→   775→

   777→   777→    isGreyResult: score >= gt.requiredChecks,

   783→   783→
   784→   784→/**
   785→   785→ * Compute GOAL FEST indicator v2 — backtest-optimized combo detector.
   786→   786→ * Updated from 7,230 + 7,110-match analysis across 4 leagues x 5 seasons.
   787→   787→ *
   788→   788→ * Captures two distinct goal-fest patterns:
   789→   789→ *   Pattern A (Divergence): xG=Over + Reg=Under/StrongUnder + ZS=Neutral
   790→   790→ *     → Teams underperforming xG but on hot streaks = finishing is about to explode
   791→   791→ *   Pattern B (Overperformance): xG=Under/StrongUnder + Reg=Neutral + ZS=Neutral
   792→   792→ *     → Teams scoring above xG, normal regression, no anomaly = natural high-scoring
   793→   793→ *
   794→   794→ * Both patterns share Z-Score = Neutral as the common thread.
   795→   795→ *
   796→   796→ * 6 checks, need 5 of 6 to qualify:
   797→   797→ *   1. xG = Over/Under/Strong Under (expanded — Strong Under = clinical finishing goldmine)
   798→   798→ *   2. Signal Divergence OR Regression = Under/StrongUnder (broadened combo check)
   799→   799→ *   3. Z-Score = Neutral (the universal goal-fest sweet spot)
   800→   800→ *   4. O2.5 >= threshold (model agrees on goals)
   801→   801→ *   5. BTTS >= threshold (model agrees both teams score)
   802→   802→ *   6. NEW: O3.5 >= threshold (more direct predictor for 4+ goals)
   803→   803→ *
   804→   804→ * Expected: ~300-400 matches per 7,110, ~33-35% O3.5 hit rate
   805→   805→ * Returns { isGoalFest, score, totalChecks, breakdown }
   806→   806→ */
   807→   807→export function computeGoalFest(
   808→   808→  checklistInput: ChecklistInput,
   809→   809→  signals: SignalInput,
   810→   810→  resolved: ReturnType<typeof resolveAllThresholds>
   811→   811→): {
   812→   812→  isGoalFest: boolean;
   813→   813→  score: number;
   814→   814→  totalChecks: number;
   815→   815→  breakdown: { check: string; passed: boolean }[];
   816→   816→} {
   817→   817→  const gf = resolved.goalFest;
   818→   818→
   819→   819→  // Signal conditions (v2)
   820→   820→  // Check 1: xG mild — Over, Under, or Strong Under
   821→   821→  //   Strong Under = teams massively outperforming xG = clinical finishing goldmine
   822→   822→  const isXgMild = (sig: string) => sig === 'Over' || sig === 'Under' || sig === 'Strong Under';
   823→   823→
   824→   824→  // Check 2: Signal Divergence (xG and Regression point opposite directions)
   825→   825→  //   This is the KEY selectivity filter: xG says goals due (Over/Under) but
   826→   826→  //   regression says they're already flowing (Under = hot streak).
   827→   827→  //   The divergence = high-variance game where finishing is about to explode.
   828→   828→  //   NOTE: Do NOT add `|| isRegressionBearish(reg)` — that makes the check
   829→   829→  //   trivially redundant and lets in non-divergent matches, hurting O3.5 hit rate.
   830→   830→  const isRegressionBearish = (sig: string) => sig === 'Under' || sig === 'Strong Under';
   831→   831→  const hasSignalDivergence = (xg: string, reg: string) =>
   832→   832→    (xg === 'Over' || xg === 'Under' || xg === 'Strong Under') && isRegressionBearish(reg);
   833→   833→
   834→   834→  const isZScoreNeutral = (sig: string) => sig === 'Neutral';
   835→   835→
   836→   836→  const checks = [
   837→   837→    { check: 'xG = Over/Under/StrongUnder', passed: isXgMild(signals.xgSignal) },
   838→   838→    { check: 'Signal Divergence (xG vs Reg)', passed: hasSignalDivergence(signals.xgSignal, signals.regressionSignal) },
   839→   839→    { check: 'Z-Score = Neutral', passed: isZScoreNeutral(signals.zScoreSignal) },
   840→   840→    { check: `O2.5 >=${gf.o25Prob.toFixed(0)}%`, passed: checklistInput.o25Prob >= gf.o25Prob },
   841→   841→    { check: `BTTS >=${gf.bttsProb.toFixed(0)}%`, passed: checklistInput.bttsProb >= gf.bttsProb },
   842→   842→    { check: `O3.5 >=${gf.o35Prob.toFixed(0)}%`, passed: checklistInput.o35Prob >= gf.o35Prob },
   843→   843→    // NEW: O2.5 implied probability confirms market expects goals
   844→   844→    { check: `O2.5 Implied >=${O25_IMPLIED_THRESHOLDS.goalFest}%`, passed: checklistInput.o25ImpliedProb !== null && checklistInput.o25ImpliedProb >= O25_IMPLIED_THRESHOLDS.goalFest },
   845→   845→    // NEW: Rolling combined scoring confirms recent hot form
   846→   846→    { check: `Rolling Scoring >=${ROLLING_SCORING_THRESHOLDS.goalFest}`, passed: checklistInput.rollingCombinedScoring >= ROLLING_SCORING_THRESHOLDS.goalFest },
   847→   847→  ];
   848→   848→
   849→   849→  const score = checks.filter(c => c.passed).length;
   850→   850→
   851→   851→  // Need 6 of 8 checks to qualify (was 5 of 6 — expanded with 2 new checks)
   852→   852→  return {
   853→   853→    isGoalFest: score >= 6,
   854→   854→    score,
   855→   855→    totalChecks: checks.length,
   856→   856→    breakdown: checks,
   857→   857→  };
   858→   858→}

   922→   922→/**
   923→   923→ * Derive optimal thresholds from backtest data using accuracy optimization.
   924→   924→ *
   925→   925→ * For each criterion, sweeps candidate threshold values and finds the one that
   926→   926→ * maximizes the correct prediction rate while meeting the minimum accuracy target.
   927→   927→ *
   928→   928→ * @param leagueName - League identifier for the registry
   929→   929→ * @param matches - Historical match data with predictions and outcomes
   930→   930→ * @param currentBaselines - Current league baselines (for fallback floors)
   931→   931→ * @param options - Tuning parameters
   932→   932→ * @returns Derived thresholds, or null if insufficient data
   933→   933→ */
   934→   934→export function deriveThresholdsFromBacktest(
   935→   935→  leagueName: string,
   936→   936→  matches: BacktestMatch[],
   937→   937→  currentBaselines: LeagueBaselines,
   938→   938→  options: BacktestDerivationOptions = {}
   939→   939→): LeagueBacktestThresholds | null {
   940→   940→  const {
   941→   941→    minSampleSize = 150,
   942→   942→    minAccuracy = 0.60,
   943→   943→    minGoalAccuracy = 0.55,
   944→   944→  } = options;
   945→   945→
   946→   946→  if (matches.length < minSampleSize) return null;
   947→   947→
   948→   948→  // --- Helper: compute BTTS hit rate at various thresholds ---
   949→   949→  const findOptimalProbThreshold = (
   950→   950→    getPredicted: (m: BacktestMatch) => number | undefined,
   951→   951→    getActual: (m: BacktestMatch) => boolean,
   952→   952→    searchRange: [number, number],
   953→   953→    step = 1,
   954→   954→    floor?: number
   955→   955→  ): number => {
   956→   956→    const [lo, hi] = searchRange;
   957→   957→    let bestThreshold = floor ?? lo;
   958→   958→    let bestAccuracy = 0;
   959→   959→    let bestSampleAbove = 0;
   960→   960→
   961→   961→    for (let t = lo; t <= hi; t += step) {
   962→   962→      const aboveThreshold = matches.filter(m => {
   963→   963→        const pred = getPredicted(m);
   964→   964→        return pred !== undefined && pred >= t;
   965→   965→      });
   966→   966→      if (aboveThreshold.length < 30) continue; // need meaningful sample above threshold
   967→   967→
   968→   968→      const correct = aboveThreshold.filter(m => getActual(m)).length;
   969→   969→      const accuracy = correct / aboveThreshold.length;
   970→   970→
   971→   971→      // Prefer: highest accuracy that meets minimum, with most samples
   972→   972→      if (accuracy >= minAccuracy && (accuracy > bestAccuracy || (accuracy === bestAccuracy && aboveThreshold.length > bestSampleAbove))) {
   973→   973→        bestThreshold = t;
   974→   974→        bestAccuracy = accuracy;
   975→   975→        bestSampleAbove = aboveThreshold.length;
   976→   976→      }
   977→   977→    }
   978→   978→
   979→   979→    // If no threshold met min accuracy, use the one with best accuracy above floor
   980→   980→    if (bestAccuracy < minAccuracy && floor !== undefined) {
   981→   981→      return floor;
   982→   982→    }
   983→   983→    return bestThreshold;
   984→   984→  };

   986→   986→  const findOptimalGoalThreshold = (
   987→   987→    getPredicted: (m: BacktestMatch) => number | undefined,
   988→   988→    getActual: (m: BacktestMatch) => boolean,

  1012→  1012→      }
  1013→  1013→    }

  1015→  1015→    if (bestAccuracy < minGoalAccuracy && floor !== undefined) {

  1207→  1207→  options: BacktestDerivationOptions = {}

  1209→  1209→  const { minSampleSize = 150 } = options;
  1210→  1210→
  1211→  1211→  if (matches.length < minSampleSize) return null;
  1212→  1212→

  1214→  1214→  // separate BTTS from non-BTTS and O3.5 from non-O3.5 matches.

  1216→  1216→  // the Poisson BTTS probability is already a strong predictor.

  1225→  1225→  let bestBttsAcc = 0;

  1227→  1227→    const threshold = Math.max(BTTS_HYBRID_THRESHOLDS.modelBttsProb.floor, currentBaselines.bttsRate * mult);

  1230→  1230→    // For simple derivation, use the rate-based threshold directly
  1231→  1231→    const estimatedAccuracy = bttsMatches.length > 0
  1232→  1232→      ? bttsMatches.filter(() => currentBaselines.bttsRate >= threshold).length / Math.max(matches.length, 1)
  1233→  1233→      : 0;

  1235→  1235→      bestBttsAcc = estimatedAccuracy;
  1236→  1236→      bestBttsMult = mult;
  1237→  1237→    }
  1238→  1238→  }
  1239→  1239→

  1241→  1241→  // with the derived multiplier, falling back to defaults.

  1285→  1285→// ============================================================================
  1286→  1286→// Bookie Odds Dampener — Heavy Favorite Adjustment
  1287→  1287→// ============================================================================
  1288→  1288→// Based on live results analysis: when bookie odds for one side are <=1.70,
  1289→  1289→// the model systematically overestimates O2.5 probability by 10-15%.
  1290→  1290→// Heavy favorites tend to win tight, controlling games (1-0, 2-0) rather
  1291→  1291→// than high-scoring shootouts. This dampener accounts for that pattern.
  1292→  1292→// ============================================================================
  1293→  1293→
  1294→  1294→/** Heavy favorite odds threshold (decimal odds) */
  1295→  1295→const HEAVY_FAVORITE_ODDS = 1.70;
  1296→  1296→
  1297→  1297→/** Maximum dampener amount (percentage points to subtract) */
  1298→  1298→const HEAVY_FAVORITE_DAMPENER = 10;
  1299→  1299→
  1300→  1300→/**
  1301→  1301→ * Apply bookie odds dampener to goal market probabilities.
  1302→  1302→ * When either team is a heavy favorite (avg odds <=1.70), reduce O2.5/O3.5/BTTS
  1303→  1303→ * probabilities to account for the "park the bus" effect.
  1304→  1304→ *
  1305→  1305→ * Only applies when valid odds are available (non-null, positive).
  1306→  1306→ *
  1307→  1307→ * @param over25Prob  - Model O2.5 probability (%)
  1308→  1308→ * @param over35Prob  - Model O3.5 probability (%)
  1309→  1309→ * @param bttsProb    - Model BTTS probability (%)
  1310→  1310→ * @param homeAvgOdds - Home team average odds (decimal)
  1311→  1311→ * @param awayAvgOdds - Away team average odds (decimal)
  1312→  1312→ * @returns Adjusted probabilities with dampener applied
  1313→  1313→ */
  1314→  1314→export function applyBookieOddsDampener(
  1315→  1315→  over25Prob: number,
  1316→  1316→  over35Prob: number,
  1317→  1317→  bttsProb: number,
  1318→  1318→  homeAvgOdds: number | null | undefined,
  1319→  1319→  awayAvgOdds: number | null | undefined,
  1320→  1320→  leagueAvgGoalsPerGame?: number
  1321→  1321→): { over25: number; over35: number; btts: number; dampened: boolean; reason: string } {
  1322→  1322→  if (!homeAvgOdds || !awayAvgOdds || homeAvgOdds <= 0 || awayAvgOdds <= 0) {
  1323→  1323→    return { over25: over25Prob, over35: over35Prob, btts: bttsProb, dampened: false, reason: '' };
  1324→  1324→  }
  1325→  1325→
  1326→  1326→  // League-adaptive: gentler dampener in high-variance leagues
  1327→  1327→  const variance = leagueAvgGoalsPerGame ? classifyLeagueVariance(leagueAvgGoalsPerGame) : 'low';
  1328→  1328→  const heavyThreshold = variance === 'high' ? 1.40 : variance === 'medium' ? 1.60 : HEAVY_FAVORITE_ODDS;
  1329→  1329→  const maxDampener = variance === 'high' ? 5 : variance === 'medium' ? 7 : HEAVY_FAVORITE_DAMPENER;
  1330→  1330→
  1331→  1331→  const homeIsHeavy = homeAvgOdds <= heavyThreshold;
  1332→  1332→  const awayIsHeavy = awayAvgOdds <= heavyThreshold;
  1333→  1333→
  1334→  1334→  if (!homeIsHeavy && !awayIsHeavy) {
  1335→  1335→    return { over25: over25Prob, over35: over35Prob, btts: bttsProb, dampened: false, reason: '' };
  1336→  1336→  }
  1337→  1337→
  1338→  1338→  const heavyOdds = Math.min(homeAvgOdds, awayAvgOdds);
  1339→  1339→  // Scale dampener: heavier favorite = bigger reduction
  1340→  1340→  // At 1.17 (PSG level): full dampener
  1341→  1341→  // At threshold: minimal 2% dampener
  1342→  1342→  const minDampener = variance === 'high' ? 2 : 3;
  1343→  1343→  const scale = Math.max(0, 1 - (heavyOdds - 1.17) / (heavyThreshold - 1.17));
  1344→  1344→  const dampener = Math.round(minDampener + (maxDampener - minDampener) * scale);
  1345→  1345→
  1346→  1346→  const leagueLabel = variance.charAt(0).toUpperCase() + variance.slice(1);
  1347→  1347→  const reason = homeIsHeavy
  1348→  1348→    ? `${leagueLabel}-variance: Heavy home favorite (${homeAvgOdds.toFixed(2)}): -${dampener}% dampener`
  1349→  1349→    : `${leagueLabel}-variance: Heavy away favorite (${awayAvgOdds.toFixed(2)}): -${dampener}% dampener`;
  1350→  1350→
  1351→  1351→  return {
  1352→  1352→    over25: Math.max(30, over25Prob - dampener),
  1353→  1353→    over35: Math.max(20, over35Prob - dampener),
  1354→  1354→    btts: Math.max(25, bttsProb - Math.round(dampener * 0.5)), // BTTS dampened less
  1355→  1355→    dampened: true,
  1356→  1356→    reason,
  1357→  1357→  };
  1358→  1358→}
  1359→  1359→
  1360→  1360→/**
  1361→  1361→ * Compute average bookie odds for a team from match results.
  1362→  1362→ * Averages odds across all matches for that team.
  1363→  1363→ *
  1364→  1364→ * @param matches - Match results with odds data
  1365→  1365→ * @param team    - Team name
  1366→  1366→ * @returns Average decimal odds, or null if insufficient data
  1367→  1367→ */
  1368→  1368→export function computeTeamAvgOdds(
  1369→  1369→  matches: { homeTeam: string; awayTeam: string; oddsAvgHome: number | null; oddsAvgAway: number | null }[],
  1370→  1370→  team: string
  1371→  1371→): number | null {
  1372→  1372→  const teamOdds: number[] = [];
  1373→  1373→
  1374→  1374→  for (const m of matches) {
  1375→  1375→    const odds = m.homeTeam === team ? m.oddsAvgHome : (m.awayTeam === team ? m.oddsAvgAway : null);
  1376→  1376→    if (odds && odds > 1 && odds < 20) {
  1377→  1377→      teamOdds.push(odds);
  1378→  1378→    }
  1379→  1379→  }
  1380→  1380→
  1381→  1381→  if (teamOdds.length < 3) return null;
  1382→  1382→
  1383→  1383→  return teamOdds.reduce((s, o) => s + o, 0) / teamOdds.length;
  1384→  1384→}

  1387→  1387→// League Variance Classification — Used by blowout/dampener adaptivity
  1388→  1388→// ============================================================================
  1389→  1389→// Classifies a league by its historical goal variance so that blowout and
  1390→  1390→// dampener thresholds can be calibrated to the league's natural scoring profile.
  1391→  1391→// High-variance leagues (2. Bundesliga, Eredivisie) see more goal explosions
  1392→  1392→// from heavy favorites, so suppression mechanisms should be gentler.
  1393→  1393→

  1395→  1395→
  1396→  1396→export function classifyLeagueVariance(avgGoalsPerGame: number): LeagueVariance {
  1397→  1397→  if (avgGoalsPerGame >= 3.0) return 'high';
  1398→  1398→  if (avgGoalsPerGame >= 2.5) return 'medium';
  1399→  1399→  return 'low';

  1402→  1402→// ============================================================================
  1403→  1403→// Blowout Risk Qualifier — Override Under signals in massive quality gaps
  1404→  1404→// ============================================================================
  1405→  1405→// Based on live results analysis: when there's a huge quality gap (favorite's
  1406→  1406→// avg odds <= 1.50), the Regression "Under" signal is unreliable. Heavy favorites
  1407→  1407→// tend to dominate regardless of recent form — a "cooling off" period doesn't
  1408→  1408→// matter when you're 2-3 quality tiers above the opponent.
  1409→  1409→//
  1410→  1410→// This qualifier prevents the "Under" regression signal from canceling out
  1411→  1411→// legitimate "Over" signals (xG, Z-Score) in STRONG BET / GREY RESULT / GOAL FEST.
  1412→  1412→//
  1413→  1413→// Example: Man City (odds 1.20) vs Luton — Regression says "Under" (City only
  1414→  1414→// scored 1 goal last 3 games), but City will likely score 3-4 regardless.
  1415→  1415→// Without this qualifier, the Under signal would block 2 points in STRONG BET.
  1416→  1416→// ============================================================================
  1417→  1417→
  1418→  1418→/** Blowout threshold: favorite's average odds at or below this triggers override */
  1419→  1419→const BLOWOUT_ODDS_THRESHOLD = 1.50;
  1420→  1420→
  1421→  1421→/**
  1422→  1422→ * Apply blowout risk qualifier to regression signal.
  1423→  1423→ * When either team is a massive favorite (avg odds <= 1.50), override
  1424→  1424→ * Regression "Under"/"Strong Under" to "Neutral" because quality advantage
  1425→  1425→ * outweighs recent form in one-sided matchups.
  1426→  1426→ *
  1427→  1427→ * @param signals      - Current signal set (xgSignal, regressionSignal, zScoreSignal)
  1428→  1428→ * @param homeAvgOdds  - Home team average odds (decimal)
  1429→  1429→ * @param awayAvgOdds  - Away team average odds (decimal)
  1430→  1430→ * @returns Modified signals (regression may be overridden to Neutral)
  1431→  1431→ */
  1432→  1432→export function applyBlowoutQualifier(
  1433→  1433→  signals: SignalInput,
  1434→  1434→  homeAvgOdds: number | null | undefined,
  1435→  1435→  awayAvgOdds: number | null | undefined,
  1436→  1436→  leagueAvgGoalsPerGame?: number
  1437→  1437→): { signals: SignalInput; blowoutOverride: boolean; reason: string } {
  1438→  1438→  // Only applies when valid odds are available
  1439→  1439→  if (!homeAvgOdds || !awayAvgOdds || homeAvgOdds <= 0 || awayAvgOdds <= 0) {
  1440→  1440→    return { signals, blowoutOverride: false, reason: '' };
  1441→  1441→  }
  1442→  1442→
  1443→  1443→  // League-adaptive threshold: gentler in high-variance leagues
  1444→  1444→  const variance = leagueAvgGoalsPerGame ? classifyLeagueVariance(leagueAvgGoalsPerGame) : 'low';
  1445→  1445→  const blowoutThreshold = variance === 'high' ? 1.25 : variance === 'medium' ? 1.40 : BLOWOUT_ODDS_THRESHOLD;
  1446→  1446→
  1447→  1447→  const homeIsMassiveFavorite = homeAvgOdds <= blowoutThreshold;
  1448→  1448→  const awayIsMassiveFavorite = awayAvgOdds <= blowoutThreshold;
  1449→  1449→
  1450→  1450→  // Only trigger when one team is a massive favorite
  1451→  1451→  if (!homeIsMassiveFavorite && !awayIsMassiveFavorite) {
  1452→  1452→    return { signals, blowoutOverride: false, reason: '' };
  1453→  1453→  }
  1454→  1454→
  1455→  1455→  // In high-variance leagues, only override Strong Under (keep mild signals intact)
  1456→  1456→  // because hot favorites in goal-rich leagues often score freely
  1457→  1457→  const shouldOverride = variance === 'high'
  1458→  1458→    ? signals.regressionSignal === 'Strong Under'
  1459→  1459→    : signals.regressionSignal === 'Under' || signals.regressionSignal === 'Strong Under';
  1460→  1460→  if (!shouldOverride) {
  1461→  1461→    return { signals, blowoutOverride: false, reason: '' };
  1462→  1462→  }
  1463→  1463→
  1464→  1464→  const favoriteOdds = Math.min(homeAvgOdds, awayAvgOdds);
  1465→  1465→  const favorite = homeIsMassiveFavorite ? 'Home' : 'Away';
  1466→  1466→  const reason = `Blowout qualifier (${variance}-variance): ${favorite} favorite (${favoriteOdds.toFixed(2)} <= ${blowoutThreshold}) — Regression "${signals.regressionSignal}" overridden to Neutral`;
  1467→  1467→
  1468→  1468→  return {
  1469→  1469→    signals: {
  1470→  1470→      ...signals,
  1471→  1471→      regressionSignal: 'Neutral',
  1472→  1472→    },
  1473→  1473→    blowoutOverride: true,
  1474→  1474→    reason,
  1475→  1475→  };
  1476→  1476→}

  1581→  1581→function classifyTeamMomentum(team: MomentumTeamInput): string {
  1582→  1582→  if (team.last5Goals.length < 3) return 'Neutral';
  1583→  1583→
  1584→  1584→  const last5Avg = team.last5Goals.reduce((s, g) => s + g, 0) / team.last5Goals.length;
  1585→  1585→  const last3 = team.last5Goals.slice(0, Math.min(3, team.last5Goals.length));
  1586→  1586→  const last3Avg = last3.reduce((s, g) => s + g, 0) / last3.length;
  1587→  1587→
  1588→  1588→  // Consistency: coefficient of variation of last 5
  1589→  1589→  const mean = last5Avg;
  1590→  1590→  const variance = team.last5Goals.reduce((s, g) => s + (g - mean) ** 2, 0) / team.last5Goals.length;
  1591→  1591→  const stdDev = Math.sqrt(variance);
  1592→  1592→  const cv = mean > 0 ? stdDev / mean : 999;
  1593→  1593→
  1594→  1594→  // Sustained: 5+ game window above season average by meaningful margin
  1595→  1595→  const sustained = last5Avg > team.seasonAvg * 1.15;
  1596→  1596→
  1597→  1597→  // Accelerating: last 3 even hotter than last 5
  1598→  1598→  const accelerating = last3Avg > last5Avg * 1.05;
  1599→  1599→
  1600→  1600→  // xG sustainability: overperforming xG with shot volume to back it
  1601→  1601→  const xgSustainable = team.xgDiff > 0 && team.sotPerGame !== null && team.sotPerGame > 5.0;
  1602→  1602→
  1603→  1603→  // Strong Momentum: sustained + consistent + high output
  1604→  1604→  if (sustained && cv < 0.5 && last5Avg >= 2.0) return 'Strong Momentum';
  1605→  1605→  // Momentum: sustained with moderate variance, or recent surge with sustainable backing
  1606→  1606→  if (sustained && cv < 0.65) return 'Momentum';
  1607→  1607→  if (last5Avg > team.seasonAvg * 1.15 && accelerating && (xgSustainable || team.xgDiff <= 0)) return 'Momentum';
  1608→  1608→  // Cold: declining output
  1609→  1609→  if (last5Avg < team.seasonAvg * 0.75) {
  1610→  1610→    return last3Avg < last5Avg * 0.8 ? 'Cold' : 'Cooling';
  1611→  1611→  }
  1612→  1612→  return 'Neutral';
  1613→  1613→}

  1615→  1615→export function computeMomentumSignal(home: MomentumTeamInput, away: MomentumTeamInput): MomentumResult {

  1618→  1618→
  1619→  1619→  const isOver = (s: string) => s === 'Strong Momentum' || s === 'Momentum';
  1620→  1620→  const isUnder = (s: string) => s === 'Cold' || s === 'Cooling';
  1621→  1621→
  1622→  1622→  let matchSignal: string;
  1623→  1623→  if (isOver(homeSignal) && isOver(awaySignal)) matchSignal = 'MOMENTUM OVER';
  1624→  1624→  else if ((isOver(homeSignal) || isOver(awaySignal)) && !isUnder(homeSignal) && !isUnder(awaySignal)) matchSignal = 'MOMENTUM LEAN OVER';
  1625→  1625→  else if (isUnder(homeSignal) && isUnder(awaySignal)) matchSignal = 'MOMENTUM UNDER';
  1626→  1626→  else if (isUnder(homeSignal) || isUnder(awaySignal)) matchSignal = 'MOMENTUM LEAN UNDER';
  1627→  1627→  else matchSignal = 'MOMENTUM NEUTRAL';
  1628→  1628→
  1629→  1629→  return { homeSignal, awaySignal, matchSignal };

  1643→  1643→// Tier mapping:
  1644→  1644→//   5+ → BTTS STRONG   (cap 70%) — both teams individually qualify
  1645→  1645→//   3–4 → BTTS QUALIFIED (cap 55%) — mild BTTS confidence
  1646→  1646→//   1–2 → BTTS WEAK     (cap 40%) — one-sided regression likely

  1649→  1649→export function computeBTTSQualification(input: BttsQualificationInput): BttsQualificationResult {
  1650→  1650→  let score = 0;
  1651→  1651→  const checks: { check: string; points: number; passed: boolean }[] = [];
  1652→  1652→
  1653→  1653→  const isRegressionUnder = (sig: string) => sig === 'Under' || sig === 'Strong Under';
  1654→  1654→  const isXgOver = (diff: number) => diff <= -0.5; // underperforming xG = goals due
  1655→  1655→  const isZCold = (z: number) => z >= 1.0; // overperforming Z = cold streak coming
  1656→  1656→  const isZNeutralOrBetter = (z: number) => z < 1.0; // not overheated

  1661→  1661→  const awayMom = isMomentum(input.awayMomentumSignal);
  1662→  1662→  const bothMom = homeMom && awayMom;
  1663→  1663→  const oneMom = homeMom || awayMom;
  1664→  1664→
  1665→  1665→  // Check 1: Regression Under OR Momentum Over — dual qualification pathway
  1666→  1666→  const bothRegUnder = isRegressionUnder(input.homeRegressionSignal) && isRegressionUnder(input.awayRegressionSignal);
  1667→  1667→  const oneRegUnder = isRegressionUnder(input.homeRegressionSignal) || isRegressionUnder(input.awayRegressionSignal);
  1668→  1668→
  1669→  1669→  if (bothRegUnder && bothMom) {
  1670→  1670→    // Supercharged: both signals agree from both sides
  1671→  1671→    score += 5;

  1673→  1673→  } else if (bothRegUnder || bothMom) {
  1674→  1674→    score += 3;

  1676→  1676→  } else if ((oneRegUnder && oneMom)) {
  1677→  1677→    score += 2; // Mixed but both signal types point to goals

  1679→  1679→  } else if (oneRegUnder || oneMom) {

  1682→  1682→  } else {
  1683→  1683→    checks.push({ check: 'No Regression Under or Momentum Over', points: 3, passed: false });

  1688→  1688→  if (bothXgOver) {
  1689→  1689→    score += 2;
  1690→  1690→    checks.push({ check: 'Both teams xG Over (underperforming)', points: 2, passed: true });
  1691→  1691→  } else {
  1692→  1692→    checks.push({ check: 'Both teams xG Over (underperforming)', points: 2, passed: false });
  1693→  1693→  }

  1697→  1697→  if (bothZOk) {
  1698→  1698→    score += 1;
  1699→  1699→    checks.push({ check: 'Both teams Z-Score Neutral/better', points: 1, passed: true });
  1700→  1700→  } else {
  1701→  1701→    checks.push({ check: 'Both teams Z-Score Neutral/better', points: 1, passed: false });
  1702→  1702→  }
  1703→  1703→
  1704→  1704→  // Check 4: Competitive matchup (favorite odds 1.80–2.50)
  1705→  1705→  if (input.favoriteOdds && input.favoriteOdds >= 1.80 && input.favoriteOdds <= 2.50) {
  1706→  1706→    score += 1;
  1707→  1707→    checks.push({ check: 'Competitive matchup (odds 1.80-2.50)', points: 1, passed: true });
  1708→  1708→  } else {

  1738→  1738→  const awayCold = isZCold(input.awayZScore);

  1742→  1742→    check: `Home Z-Score ${homeCold ? 'cold (≥1.0)' : 'ok'}`,

  1764→  1764→  } else {

  1769→  1769→  return { tier, score, breakdown: checks, bttsCap };

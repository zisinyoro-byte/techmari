   998→                            {(() => {
   999→                              // Estimate rho from low-score frequencies
  1000→                              const total = analytics.totalMatches
  1001→                              const under25 = analytics.under25Count

  1053→                          if (x === 0 && y === 0) return 1 - lambda1 * lambda2 * rho
  1054→                          if (x === 0 && y === 1) return 1 + lambda1 * rho

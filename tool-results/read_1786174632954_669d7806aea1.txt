  1049→/**
  1050→ * Derive thresholds from backtest data using a simpler approach when model
  1051→ * predictions are not available — uses league averages as the predictor.
  1052→ */
  1053→export function deriveSimpleThresholdsFromBacktest(
  1054→  leagueName: string,
  1055→  matches: { ftHomeGoals: number; ftAwayGoals: number }[],
  1056→  currentBaselines: LeagueBaselines,
  1057→  options: BacktestDerivationOptions = {}
  1058→): LeagueBacktestThresholds | null {
  1059→  const { minSampleSize = 150 } = options;
  1060→
  1061→  if (matches.length < minSampleSize) return null;
  1062→
  1063→  // Simple derivation: find the league-average-based thresholds that best
  1064→  // separate BTTS from non-BTTS and O3.5 from non-O3.5 matches.
  1065→  // This uses the Dixon-Coles probability approach: given league averages,
  1066→  // the Poisson BTTS probability is already a strong predictor.
  1067→
  1068→  const bttsMatches = matches.filter(m => m.ftHomeGoals > 0 && m.ftAwayGoals > 0);
  1069→  const over35Matches = matches.filter(m => m.ftHomeGoals + m.ftAwayGoals > 3.5);
  1070→
  1071→  // Derive BTTS probability threshold from league BTTS rate
  1072→  // A match is "above average" if its BTTS prob > league BTTS rate
  1073→  // Find the multiplier that gives best accuracy
  1074→  let bestBttsMult = BTTS_HYBRID_THRESHOLDS.modelBttsProb.multiplier;
  1075→  let bestBttsAcc = 0;
  1076→  for (let mult = 1.0; mult <= 1.3; mult += 0.01) {
  1077→    const threshold = Math.max(BTTS_HYBRID_THRESHOLDS.modelBttsProb.floor, currentBaselines.bttsRate * mult);
  1078→    const aboveCount = bttsMatches.filter(() => currentBaselines.bttsRate >= threshold).length;
  1079→    // All BTTS matches have BTTS prob ~= league rate, so check hit rate
  1080→    // For simple derivation, use the rate-based threshold directly
  1081→    const estimatedAccuracy = bttsMatches.length > 0
  1082→      ? bttsMatches.filter(() => currentBaselines.bttsRate >= threshold).length / Math.max(matches.length, 1)
  1083→      : 0;
  1084→    if (estimatedAccuracy > bestBttsAcc) {
  1085→      bestBttsAcc = estimatedAccuracy;
  1086→      bestBttsMult = mult;
  1087→    }
  1088→  }
  1089→
  1090→  // For simple derivation without per-match predictions, use the hybrid formula
  1091→  // with the derived multiplier, falling back to defaults.
  1092→  const bttsFloor = BTTS_HYBRID_THRESHOLDS.modelBttsProb.floor;
  1093→  const bttsRate = currentBaselines.bttsRate;
  1094→
  1095→  return {
  1096→    leagueName,
  1097→    bttsLeagueAvgGoals: BTTS_LEAGUE_THRESHOLDS.leagueAvgGoals,
  1098→    bttsLeagueO25Rate: BTTS_LEAGUE_THRESHOLDS.leagueO25Rate,
  1099→    o35LeagueAvgGoals: OVER35_LEAGUE_THRESHOLDS.leagueAvgGoals,
  1100→    o35LeagueO25Rate: OVER35_LEAGUE_THRESHOLDS.leagueO25Rate,
  1101→    btts: {
  1102→      modelBttsProb: Math.max(bttsFloor, bttsRate * bestBttsMult),
  1103→      homeAvgGoals: Math.max(BTTS_HYBRID_THRESHOLDS.homeAvgGoals.floor, currentBaselines.avgHomeGoals * BTTS_HYBRID_THRESHOLDS.homeAvgGoals.multiplier),
  1104→      awayAvgGoals: Math.max(BTTS_HYBRID_THRESHOLDS.awayAvgGoals.floor, currentBaselines.avgAwayGoals * BTTS_HYBRID_THRESHOLDS.awayAvgGoals.multiplier),
  1105→      modelO25Prob: Math.max(BTTS_HYBRID_THRESHOLDS.modelO25Prob.floor, currentBaselines.over25Rate * BTTS_HYBRID_THRESHOLDS.modelO25Prob.multiplier),
  1106→      shotConversion: Math.max(BTTS_HYBRID_THRESHOLDS.shotConversion.floor, currentBaselines.shotConversion * BTTS_HYBRID_THRESHOLDS.shotConversion.multiplier),
  1107→    },
  1108→    over35: {
  1109→      modelO35Prob: Math.max(OVER35_HYBRID_THRESHOLDS.modelO35Prob.floor, currentBaselines.over35Rate * OVER35_HYBRID_THRESHOLDS.modelO35Prob.multiplier),
  1110→      bttsProb: Math.max(OVER35_HYBRID_THRESHOLDS.bttsProb.floor, currentBaselines.bttsRate * OVER35_HYBRID_THRESHOLDS.bttsProb.multiplier),
  1111→      homeAvgGoals: Math.max(OVER35_HYBRID_THRESHOLDS.homeAvgGoals.floor, currentBaselines.avgHomeGoals * OVER35_HYBRID_THRESHOLDS.homeAvgGoals.multiplier),
  1112→      awayAvgGoals: Math.max(OVER35_HYBRID_THRESHOLDS.awayAvgGoals.floor, currentBaselines.avgAwayGoals * OVER35_HYBRID_THRESHOLDS.awayAvgGoals.multiplier),
  1113→      shotConversion: Math.max(OVER35_HYBRID_THRESHOLDS.shotConversion.floor, currentBaselines.shotConversion * OVER35_HYBRID_THRESHOLDS.shotConversion.multiplier),
  1114→    },
  1115→    strongBet: {
  1116→      o25Prob: Math.max(STRONG_BET_HYBRID.o25Prob.floor, currentBaselines.over25Rate * STRONG_BET_HYBRID.o25Prob.multiplier),
  1117→      o35Prob: Math.max(STRONG_BET_HYBRID.o35Prob.floor, currentBaselines.over35Rate * STRONG_BET_HYBRID.o35Prob.multiplier),
  1118→      bttsProb: Math.max(STRONG_BET_HYBRID.bttsProb.floor, currentBaselines.bttsRate * STRONG_BET_HYBRID.bttsProb.multiplier),
  1119→      bttsChecklistCount: STRONG_BET_HYBRID.bttsChecklistCount,
  1120→    },
  1121→    greyResult: {
  1122→      bttsProb: GREY_RESULT_CONFIG.bttsProb.min,
  1123→      bttsProbMax: GREY_RESULT_CONFIG.bttsProb.max,
  1124→      o25Prob: Math.max(GREY_RESULT_CONFIG.o25Prob.floor, currentBaselines.over25Rate * GREY_RESULT_CONFIG.o25Prob.multiplier),
  1125→      o35Prob: Math.max(GREY_RESULT_CONFIG.o35Prob.floor, currentBaselines.over35Rate * GREY_RESULT_CONFIG.o35Prob.multiplier),
  1126→      bttsChecklistCount: GREY_RESULT_CONFIG.bttsChecklistCount,
  1127→      over35ChecklistCount: GREY_RESULT_CONFIG.over35ChecklistCount,
  1128→      requiredChecks: GREY_RESULT_CONFIG.requiredChecks,
  1129→    },
  1130→    sampleSize: matches.length,
  1131→    derivedAt: new Date().toISOString(),
  1132→  };
  1133→}
  1134→
  1135→// ============================================================================
  1136→// Bookie Odds Dampener — Heavy Favorite Adjustment
  1137→// ============================================================================
  1138→// Based on live results analysis: when bookie odds for one side are <=1.70,
  1139→// the model systematically overestimates O2.5 probability by 10-15%.
  1140→// Heavy favorites tend to win tight, controlling games (1-0, 2-0) rather
  1141→// than high-scoring shootouts. This dampener accounts for that pattern.
  1142→// ============================================================================
  1143→
  1144→/** Heavy favorite odds threshold (decimal odds) */
  1145→const HEAVY_FAVORITE_ODDS = 1.70;
  1146→
  1147→/** Maximum dampener amount (percentage points to subtract) */
  1148→const HEAVY_FAVORITE_DAMPENER = 10;
  1149→
  1150→/**
  1151→ * Apply bookie odds dampener to goal market probabilities.
  1152→ * When either team is a heavy favorite (avg odds <=1.70), reduce O2.5/O3.5/BTTS
  1153→ * probabilities to account for the "park the bus" effect.
  1154→ *
  1155→ * Only applies when valid odds are available (non-null, positive).
  1156→ *
  1157→ * @param over25Prob  - Model O2.5 probability (%)
  1158→ * @param over35Prob  - Model O3.5 probability (%)
  1159→ * @param bttsProb    - Model BTTS probability (%)
  1160→ * @param homeAvgOdds - Home team average odds (decimal)
  1161→ * @param awayAvgOdds - Away team average odds (decimal)
  1162→ * @returns Adjusted probabilities with dampener applied
  1163→ */
  1164→export function applyBookieOddsDampener(
  1165→  over25Prob: number,
  1166→  over35Prob: number,
  1167→  bttsProb: number,
  1168→  homeAvgOdds: number | null | undefined,
  1169→  awayAvgOdds: number | null | undefined,
  1170→  leagueAvgGoalsPerGame?: number
  1171→): { over25: number; over35: number; btts: number; dampened: boolean; reason: string } {
  1172→  if (!homeAvgOdds || !awayAvgOdds || homeAvgOdds <= 0 || awayAvgOdds <= 0) {
  1173→    return { over25: over25Prob, over35: over35Prob, btts: bttsProb, dampened: false, reason: '' };
  1174→  }
  1175→
  1176→  // League-adaptive: gentler dampener in high-variance leagues
  1177→  const variance = leagueAvgGoalsPerGame ? classifyLeagueVariance(leagueAvgGoalsPerGame) : 'low';
  1178→  const heavyThreshold = variance === 'high' ? 1.40 : variance === 'medium' ? 1.60 : HEAVY_FAVORITE_ODDS;
  1179→  const maxDampener = variance === 'high' ? 5 : variance === 'medium' ? 7 : HEAVY_FAVORITE_DAMPENER;
  1180→
  1181→  const homeIsHeavy = homeAvgOdds <= heavyThreshold;
  1182→  const awayIsHeavy = awayAvgOdds <= heavyThreshold;
  1183→
  1184→  if (!homeIsHeavy && !awayIsHeavy) {
  1185→    return { over25: over25Prob, over35: over35Prob, btts: bttsProb, dampened: false, reason: '' };
  1186→  }
  1187→
  1188→  const heavyOdds = Math.min(homeAvgOdds, awayAvgOdds);
  1189→  // Scale dampener: heavier favorite = bigger reduction
  1190→  // At 1.17 (PSG level): full dampener
  1191→  // At threshold: minimal 2% dampener
  1192→  const minDampener = variance === 'high' ? 2 : 3;
  1193→  const scale = Math.max(0, 1 - (heavyOdds - 1.17) / (heavyThreshold - 1.17));
  1194→  const dampener = Math.round(minDampener + (maxDampener - minDampener) * scale);
  1195→
  1196→  const leagueLabel = variance.charAt(0).toUpperCase() + variance.slice(1);
  1197→  const reason = homeIsHeavy
  1198→    ? `${leagueLabel}-variance: Heavy home favorite (${homeAvgOdds.toFixed(2)}): -${dampener}% dampener`
  1199→    : `${leagueLabel}-variance: Heavy away favorite (${awayAvgOdds.toFixed(2)}): -${dampener}% dampener`;
  1200→
  1201→  return {
  1202→    over25: Math.max(30, over25Prob - dampener),
  1203→    over35: Math.max(20, over35Prob - dampener),
  1204→    btts: Math.max(25, bttsProb - Math.round(dampener * 0.5)), // BTTS dampened less
  1205→    dampened: true,
  1206→    reason,
  1207→  };
  1208→}

  1210→/**
  1211→ * Compute average bookie odds for a team from match results.
  1212→ * Averages odds across all matches for that team.
  1213→ *
  1214→ * @param matches - Match results with odds data
  1215→ * @param team    - Team name
  1216→ * @returns Average decimal odds, or null if insufficient data
  1217→ */
  1218→export function computeTeamAvgOdds(
  1219→  matches: { homeTeam: string; awayTeam: string; oddsAvgHome: number | null; oddsAvgAway: number | null }[],
  1220→  team: string
  1221→): number | null {
  1222→  const teamOdds: number[] = [];
  1223→
  1224→  for (const m of matches) {
  1225→    const odds = m.homeTeam === team ? m.oddsAvgHome : (m.awayTeam === team ? m.oddsAvgAway : null);
  1226→    if (odds && odds > 1 && odds < 20) {
  1227→      teamOdds.push(odds);
  1228→    }
  1229→  }
  1230→
  1231→  if (teamOdds.length < 3) return null;
  1232→
  1233→  return teamOdds.reduce((s, o) => s + o, 0) / teamOdds.length;
  1234→}

  1236→// ============================================================================
  1237→// League Variance Classification — Used by blowout/dampener adaptivity
  1238→// ============================================================================
  1239→// Classifies a league by its historical goal variance so that blowout and
  1240→// dampener thresholds can be calibrated to the league's natural scoring profile.
  1241→// High-variance leagues (2. Bundesliga, Eredivisie) see more goal explosions
  1242→// from heavy favorites, so suppression mechanisms should be gentler.
  1243→
  1244→export type LeagueVariance = 'high' | 'medium' | 'low';
  1245→
  1246→export function classifyLeagueVariance(avgGoalsPerGame: number): LeagueVariance {
  1247→  if (avgGoalsPerGame >= 3.0) return 'high';
  1248→  if (avgGoalsPerGame >= 2.5) return 'medium';
  1249→  return 'low';
  1250→}
  1251→
  1252→// ============================================================================
  1253→// Blowout Risk Qualifier — Override Under signals in massive quality gaps
  1254→// ============================================================================
  1255→// Based on live results analysis: when there's a huge quality gap (favorite's
  1256→// avg odds <= 1.50), the Regression "Under" signal is unreliable. Heavy favorites
  1257→// tend to dominate regardless of recent form — a "cooling off" period doesn't
  1258→// matter when you're 2-3 quality tiers above the opponent.
  1259→//
  1260→// This qualifier prevents the "Under" regression signal from canceling out
  1261→// legitimate "Over" signals (xG, Z-Score) in STRONG BET / GREY RESULT / GOAL FEST.
  1262→//
  1263→// Example: Man City (odds 1.20) vs Luton — Regression says "Under" (City only
  1264→// scored 1 goal last 3 games), but City will likely score 3-4 regardless.
  1265→// Without this qualifier, the Under signal would block 2 points in STRONG BET.
  1266→// ============================================================================
  1267→
  1268→/** Blowout threshold: favorite's average odds at or below this triggers override */
  1269→const BLOWOUT_ODDS_THRESHOLD = 1.50;
  1270→
  1271→/**
  1272→ * Apply blowout risk qualifier to regression signal.
  1273→ * When either team is a massive favorite (avg odds <= 1.50), override
  1274→ * Regression "Under"/"Strong Under" to "Neutral" because quality advantage
  1275→ * outweighs recent form in one-sided matchups.
  1276→ *
  1277→ * @param signals      - Current signal set (xgSignal, regressionSignal, zScoreSignal)
  1278→ * @param homeAvgOdds  - Home team average odds (decimal)
  1279→ * @param awayAvgOdds  - Away team average odds (decimal)
  1280→ * @returns Modified signals (regression may be overridden to Neutral)
  1281→ */
  1282→export function applyBlowoutQualifier(
  1283→  signals: SignalInput,
  1284→  homeAvgOdds: number | null | undefined,
  1285→  awayAvgOdds: number | null | undefined,
  1286→  leagueAvgGoalsPerGame?: number
  1287→): { signals: SignalInput; blowoutOverride: boolean; reason: string } {
  1288→  // Only applies when valid odds are available
  1289→  if (!homeAvgOdds || !awayAvgOdds || homeAvgOdds <= 0 || awayAvgOdds <= 0) {
  1290→    return { signals, blowoutOverride: false, reason: '' };
  1291→  }
  1292→
  1293→  // League-adaptive threshold: gentler in high-variance leagues
  1294→  const variance = leagueAvgGoalsPerGame ? classifyLeagueVariance(leagueAvgGoalsPerGame) : 'low';
  1295→  const blowoutThreshold = variance === 'high' ? 1.25 : variance === 'medium' ? 1.40 : BLOWOUT_ODDS_THRESHOLD;
  1296→
  1297→  const homeIsMassiveFavorite = homeAvgOdds <= blowoutThreshold;
  1298→  const awayIsMassiveFavorite = awayAvgOdds <= blowoutThreshold;
  1299→
  1300→  // Only trigger when one team is a massive favorite
  1301→  if (!homeIsMassiveFavorite && !awayIsMassiveFavorite) {
  1302→    return { signals, blowoutOverride: false, reason: '' };
  1303→  }
  1304→
  1305→  // In high-variance leagues, only override Strong Under (keep mild signals intact)
  1306→  // because hot favorites in goal-rich leagues often score freely
  1307→  const shouldOverride = variance === 'high'
  1308→    ? signals.regressionSignal === 'Strong Under'
  1309→    : signals.regressionSignal === 'Under' || signals.regressionSignal === 'Strong Under';
  1310→  if (!shouldOverride) {
  1311→    return { signals, blowoutOverride: false, reason: '' };
  1312→  }
  1313→
  1314→  const favoriteOdds = Math.min(homeAvgOdds, awayAvgOdds);
  1315→  const favorite = homeIsMassiveFavorite ? 'Home' : 'Away';
  1316→  const reason = `Blowout qualifier (${variance}-variance): ${favorite} favorite (${favoriteOdds.toFixed(2)} <= ${blowoutThreshold}) — Regression "${signals.regressionSignal}" overridden to Neutral`;
  1317→
  1318→  return {
  1319→    signals: {
  1320→      ...signals,
  1321→      regressionSignal: 'Neutral',
  1322→    },
  1323→    blowoutOverride: true,
  1324→    reason,
  1325→  };
  1326→}
  1327→
  1328→// ============================================================================
  1329→// BTTS Dual-Team Qualification System
  1330→// ============================================================================
  1331→// BTTS requires BOTH teams to score — the match-level signal combo (Regression
  1332→// Under + Z-Score Neutral + xG Over) correctly identifies "goals are coming"
  1333→// but tends to surface ONE-SIDED regressions (one team due for a breakout).
  1334→// This system evaluates each team independently and produces a BTTS-specific
  1335→// conviction tier that is separate from the Over goal thesis.

  1375→/**
  1376→ * Classify a per-team regression deviation into a signal label.
  1377→ * Uses the same thresholds as REGRESSION_THRESHOLDS.perTeam.
  1378→ */
  1379→export function classifyPerTeamRegressionSignal(combinedSignal: number): string {
  1380→  if (combinedSignal <= -0.8) return 'Strong Over';
  1381→  if (combinedSignal <= -0.3) return 'Over';
  1382→  if (combinedSignal >= 0.8) return 'Strong Under';
  1383→  if (combinedSignal >= 0.3) return 'Under';
  1384→  return 'Neutral';
  1385→}
  1386→
  1387→/**
  1388→ * Classify a per-team xG diff into a signal label.
  1389→ * Uses the same thresholds as XG_THRESHOLDS applied per-team.
  1390→ */
  1391→export function classifyPerTeamXgSignal(xgDiff: number): string {
  1392→  if (xgDiff <= -1.0) return 'Strong Over';
  1393→  if (xgDiff <= -0.5) return 'Over';
  1394→  if (xgDiff >= 1.0) return 'Strong Under';
  1395→  if (xgDiff >= 0.5) return 'Under';
  1396→  return 'Neutral';
  1397→}
  1398→
  1399→/**
  1400→ * Classify a per-team Z-Score into a signal label.
  1401→ * Negative Z = underperforming recently (goals expected to rise).
  1402→ */
  1403→export function classifyPerTeamZScoreSignal(zScore: number): string {
  1404→  if (zScore <= -1.5) return 'Strong Over';
  1405→  if (zScore <= -1.0) return 'Over';
  1406→  if (zScore >= 1.5) return 'Strong Under';
  1407→  if (zScore >= 1.0) return 'Under';
  1408→  return 'Neutral';
  1409→}

  1411→// ============================================================================
  1412→// Momentum Signal — Parallel track to Regression (form continuation)
  1413→// ============================================================================
  1414→// While Regression asks "is this team due to bounce back?", Momentum asks
  1415→// "is this team on a sustained hot streak that will continue?"
  1416→// Both tracks run simultaneously; a match can qualify through either pathway.
  1417→
  1418→export interface MomentumTeamInput {
  1419→  last5Goals: number[];    // total goals per match in last 5
  1420→  seasonAvg: number;       // season avg total goals per game
  1421→  xgDiff: number;          // actualGoals/matches - totalXg/matches (positive = overperforming)
  1422→  sotPerGame: number | null;
  1423→}
  1424→
  1425→export interface MomentumResult {
  1426→  homeSignal: string;      // 'Strong Momentum' | 'Momentum' | 'Neutral' | 'Cooling' | 'Cold'
  1427→  awaySignal: string;
  1428→  matchSignal: string;     // 'MOMENTUM OVER' | 'MOMENTUM LEAN OVER' | 'MOMENTUM NEUTRAL' | 'MOMENTUM LEAN UNDER' | 'MOMENTUM UNDER'
  1429→}
  1430→
  1431→function classifyTeamMomentum(team: MomentumTeamInput): string {
  1432→  if (team.last5Goals.length < 3) return 'Neutral';
  1433→
  1434→  const last5Avg = team.last5Goals.reduce((s, g) => s + g, 0) / team.last5Goals.length;
  1435→  const last3 = team.last5Goals.slice(0, Math.min(3, team.last5Goals.length));
  1436→  const last3Avg = last3.reduce((s, g) => s + g, 0) / last3.length;
  1437→
  1438→  // Consistency: coefficient of variation of last 5
  1439→  const mean = last5Avg;
  1440→  const variance = team.last5Goals.reduce((s, g) => s + (g - mean) ** 2, 0) / team.last5Goals.length;
  1441→  const stdDev = Math.sqrt(variance);
  1442→  const cv = mean > 0 ? stdDev / mean : 999;
  1443→
  1444→  // Sustained: 5+ game window above season average by meaningful margin
  1445→  const sustained = last5Avg > team.seasonAvg * 1.15;
  1446→
  1447→  // Accelerating: last 3 even hotter than last 5
  1448→  const accelerating = last3Avg > last5Avg * 1.05;
  1449→
  1450→  // xG sustainability: overperforming xG with shot volume to back it
  1451→  const xgSustainable = team.xgDiff > 0 && team.sotPerGame !== null && team.sotPerGame > 5.0;
  1452→
  1453→  // Strong Momentum: sustained + consistent + high output
  1454→  if (sustained && cv < 0.5 && last5Avg >= 2.0) return 'Strong Momentum';
  1455→  // Momentum: sustained with moderate variance, or recent surge with sustainable backing
  1456→  if (sustained && cv < 0.65) return 'Momentum';
  1457→  if (last5Avg > team.seasonAvg * 1.15 && accelerating && (xgSustainable || team.xgDiff <= 0)) return 'Momentum';
  1458→  // Cold: declining output
  1459→  if (last5Avg < team.seasonAvg * 0.75) {
  1460→    return last3Avg < last5Avg * 0.8 ? 'Cold' : 'Cooling';
  1461→  }
  1462→  return 'Neutral';
  1463→}
  1464→
  1465→export function computeMomentumSignal(home: MomentumTeamInput, away: MomentumTeamInput): MomentumResult {
  1466→  const homeSignal = classifyTeamMomentum(home);
  1467→  const awaySignal = classifyTeamMomentum(away);
  1468→
  1469→  const isOver = (s: string) => s === 'Strong Momentum' || s === 'Momentum';
  1470→  const isUnder = (s: string) => s === 'Cold' || s === 'Cooling';
  1471→
  1472→  let matchSignal: string;
  1473→  if (isOver(homeSignal) && isOver(awaySignal)) matchSignal = 'MOMENTUM OVER';
  1474→  else if ((isOver(homeSignal) || isOver(awaySignal)) && !isUnder(homeSignal) && !isUnder(awaySignal)) matchSignal = 'MOMENTUM LEAN OVER';
  1475→  else if (isUnder(homeSignal) && isUnder(awaySignal)) matchSignal = 'MOMENTUM UNDER';
  1476→  else if (isUnder(homeSignal) || isUnder(awaySignal)) matchSignal = 'MOMENTUM LEAN UNDER';
  1477→  else matchSignal = 'MOMENTUM NEUTRAL';
  1478→
  1479→  return { homeSignal, awaySignal, matchSignal };
  1480→}
  1481→
  1482→// ============================================================================
  1483→// BTTS Dual-Team Qualification System
  1484→// ============================================================================
  1485→// Evaluates each team independently to determine whether BOTH teams are likely
  1486→// to score. Returns a tier that caps BTTS probability independently from the
  1487→// Over goal thesis.
  1488→//
  1489→// Now supports TWO qualification pathways:
  1490→//   1. Regression Under: both teams underperforming → due to bounce back
  1491→//   2. Momentum Over: both teams on hot streaks → likely to keep scoring
  1492→//
  1493→// Tier mapping:
  1494→//   5+ → BTTS STRONG   (cap 70%) — both teams individually qualify
  1495→//   3–4 → BTTS QUALIFIED (cap 55%) — mild BTTS confidence
  1496→//   1–2 → BTTS WEAK     (cap 40%) — one-sided regression likely
  1497→//   ≤0  → BTTS AVOID    (cap 25%) — do not bet BTTS
  1498→// ============================================================================
  1499→export function computeBTTSQualification(input: BttsQualificationInput): BttsQualificationResult {
  1500→  let score = 0;
  1501→  const checks: { check: string; points: number; passed: boolean }[] = [];
  1502→
  1503→  const isRegressionUnder = (sig: string) => sig === 'Under' || sig === 'Strong Under';
  1504→  const isXgOver = (diff: number) => diff <= -0.5; // underperforming xG = goals due
  1505→  const isZCold = (z: number) => z >= 1.0; // overperforming Z = cold streak coming
  1506→  const isZNeutralOrBetter = (z: number) => z < 1.0; // not overheated
  1507→
  1508→  // Momentum helpers
  1509→  const isMomentum = (s: string | undefined) => s === 'Strong Momentum' || s === 'Momentum';
  1510→  const homeMom = isMomentum(input.homeMomentumSignal);
  1511→  const awayMom = isMomentum(input.awayMomentumSignal);
  1512→  const bothMom = homeMom && awayMom;
  1513→  const oneMom = homeMom || awayMom;
  1514→
  1515→  // Check 1: Regression Under OR Momentum Over — dual qualification pathway
  1516→  const bothRegUnder = isRegressionUnder(input.homeRegressionSignal) && isRegressionUnder(input.awayRegressionSignal);
  1517→  const oneRegUnder = isRegressionUnder(input.homeRegressionSignal) || isRegressionUnder(input.awayRegressionSignal);
  1518→
  1519→  if (bothRegUnder && bothMom) {
  1520→    // Supercharged: both signals agree from both sides
  1521→    score += 5;
  1522→    checks.push({ check: 'Both Regression Under AND Momentum Over (supercharged)', points: 5, passed: true });
  1523→  } else if (bothRegUnder || bothMom) {
  1524→    score += 3;
  1525→    checks.push({ check: bothRegUnder ? 'Both teams Regression Under/StrongUnder' : 'Both teams Momentum Over (hot streak)', points: 3, passed: true });
  1526→  } else if ((oneRegUnder && oneMom)) {
  1527→    score += 2; // Mixed but both signal types point to goals
  1528→    checks.push({ check: 'One Regression Under + One Momentum Over', points: 2, passed: true });
  1529→  } else if (oneRegUnder || oneMom) {
  1530→    score += 1;
  1531→    checks.push({ check: oneRegUnder ? 'One team Regression Under' : 'One team Momentum Over', points: 1, passed: true });
  1532→  } else {
  1533→    checks.push({ check: 'No Regression Under or Momentum Over', points: 3, passed: false });
  1534→  }
  1535→
  1536→  // Check 2: BOTH teams xG Over (underperforming xG = creating chances, finishing due)
  1537→  const bothXgOver = isXgOver(input.homeXgDiff) && isXgOver(input.awayXgDiff);
  1538→  if (bothXgOver) {
  1539→    score += 2;
  1540→    checks.push({ check: 'Both teams xG Over (underperforming)', points: 2, passed: true });
  1541→  } else {
  1542→    checks.push({ check: 'Both teams xG Over (underperforming)', points: 2, passed: false });
  1543→  }
  1544→
  1545→  // Check 3: BOTH teams Z-Score not overheated
  1546→  const bothZOk = isZNeutralOrBetter(input.homeZScore) && isZNeutralOrBetter(input.awayZScore);
  1547→  if (bothZOk) {
  1548→    score += 1;
  1549→    checks.push({ check: 'Both teams Z-Score Neutral/better', points: 1, passed: true });
  1550→  } else {
  1551→    checks.push({ check: 'Both teams Z-Score Neutral/better', points: 1, passed: false });
  1552→  }
  1553→
  1554→  // Check 4: Competitive matchup (favorite odds 1.80–2.50)
  1555→  if (input.favoriteOdds && input.favoriteOdds >= 1.80 && input.favoriteOdds <= 2.50) {
  1556→    score += 1;
  1557→    checks.push({ check: 'Competitive matchup (odds 1.80-2.50)', points: 1, passed: true });
  1558→  } else {
  1559→    checks.push({ check: 'Competitive matchup (odds 1.80-2.50)', points: 1, passed: false });
  1560→  }
  1561→
  1562→  // Penalty 5: Massive favorite (≤1.50) — one side dominates, BTTS unlikely
  1563→  if (input.favoriteOdds && input.favoriteOdds <= 1.50) {
  1564→    score -= 2;
  1565→    checks.push({ check: 'Massive favorite (odds ≤1.50)', points: -2, passed: true });
  1566→  } else {
  1567→    checks.push({ check: 'Massive favorite (odds ≤1.50)', points: -2, passed: false });
  1568→  }
  1569→
  1570→  // Penalty 6: Either team can't finish (SOT conversion <25%)
  1571→  const homeCantFinish = input.homeSotConversion !== null && input.homeSotConversion < 25;
  1572→  const awayCantFinish = input.awaySotConversion !== null && input.awaySotConversion < 25;
  1573→  if (homeCantFinish) score -= 1;
  1574→  if (awayCantFinish) score -= 1;
  1575→  checks.push({
  1576→    check: `Home SOT conversion ${homeCantFinish ? '< 25%' : '≥ 25%'}${input.homeSotConversion !== null ? ` (${input.homeSotConversion.toFixed(0)}%)` : ' (N/A)'}`,
  1577→    points: -1,
  1578→    passed: !homeCantFinish,
  1579→  });
  1580→  checks.push({
  1581→    check: `Away SOT conversion ${awayCantFinish ? '< 25%' : '≥ 25%'}${input.awaySotConversion !== null ? ` (${input.awaySotConversion.toFixed(0)}%)` : ' (N/A)'}`,
  1582→    points: -1,
  1583→    passed: !awayCantFinish,
  1584→  });
  1585→
  1586→  // Penalty 7: Either team in a strong cold Z-Score streak
  1587→  const homeCold = isZCold(input.homeZScore);
  1588→  const awayCold = isZCold(input.awayZScore);
  1589→  if (homeCold) score -= 2;
  1590→  if (awayCold) score -= 2;
  1591→  checks.push({
  1592→    check: `Home Z-Score ${homeCold ? 'cold (≥1.0)' : 'ok'}`,
  1593→    points: -2,
  1594→    passed: !homeCold,
  1595→  });
  1596→  checks.push({
  1597→    check: `Away Z-Score ${awayCold ? 'cold (≥1.0)' : 'ok'}`,
  1598→    points: -2,
  1599→    passed: !awayCold,
  1600→  });
  1601→
  1602→  // Determine tier
  1603→  let tier: BttsQualificationTier;
  1604→  let bttsCap: number;
  1605→  if (score >= 5) {
  1606→    tier = 'BTTS STRONG';
  1607→    bttsCap = 70;
  1608→  } else if (score >= 3) {
  1609→    tier = 'BTTS QUALIFIED';
  1610→    bttsCap = 55;
  1611→  } else if (score >= 1) {
  1612→    tier = 'BTTS WEAK';
  1613→    bttsCap = 40;
  1614→  } else {
  1615→    tier = 'BTTS AVOID';
  1616→    bttsCap = 25;
  1617→  }
  1618→
  1619→  return { tier, score, breakdown: checks, bttsCap };
  1620→}

  1666→/**
  1667→ * Compute Third Goal qualifier — distinguishes "2-goal games" from "3+ goal games"
  1668→ * within pre-filtered selections.
  1669→ *
  1670→ * Tier mapping:
  1671→ *   6+  → GOAL RICH      — Full confidence on Over 2.5, consider Over 3.5
  1672→ *   4–5 → GOAL LIKELY    — Solid Over 2.5 bet
  1673→ *   2–3 → GOAL BORDERLINE — Over 1.5 strong, Over 2.5 cautious
  1674→ *   0–1 → GOAL THIN      — Over 1.5 only, skip Over 2.5
  1675→ *   <0  → GOAL STALL     — Likely 1-2 goals, avoid Over markets above 1.5
  1676→ */
  1677→export function computeThirdGoalQualifier(input: ThirdGoalQualifierInput): ThirdGoalQualifierResult {
  1678→  let score = 0;
  1679→  const checks: { check: string; points: number; passed: boolean }[] = [];
  1680→
  1681→  const isRegressionUnder = (sig: string) => sig === 'Under' || sig === 'Strong Under';
  1682→  const isXgOver = (sig: string) => sig === 'Over' || sig === 'Strong Over';
  1683→
  1684→  // Check 1: Combined xG > 3.0 (very high chance creation)
  1685→  if (input.combinedXgTotal > 3.0) {
  1686→    score += 3;
  1687→    checks.push({ check: `Combined xG ${input.combinedXgTotal.toFixed(2)} > 3.0`, points: 3, passed: true });
  1688→  } else if (input.combinedXgTotal >= 2.5) {
  1689→    score += 1;
  1690→    checks.push({ check: `Combined xG ${input.combinedXgTotal.toFixed(2)} in 2.5-3.0`, points: 1, passed: true });
  1691→  } else {
  1692→    checks.push({ check: `Combined xG ${input.combinedXgTotal.toFixed(2)} < 2.5`, points: 3, passed: false });
  1693→  }
  1694→
  1695→  // Check 2: BOTH teams Regression Under (double regression)
  1696→  const bothRegUnder = isRegressionUnder(input.homeRegressionSignal) && isRegressionUnder(input.awayRegressionSignal);
  1697→  if (bothRegUnder) {
  1698→    score += 2;
  1699→    checks.push({ check: 'Both teams Regression Under', points: 2, passed: true });
  1700→  } else {
  1701→    checks.push({ check: 'Both teams Regression Under', points: 2, passed: false });
  1702→  }
  1703→
  1704→  // Check 3: BOTH teams xG Over (both creating chances)
  1705→  const bothXgOver = isXgOver(input.homeXgSignal) && isXgOver(input.awayXgSignal);
  1706→  if (bothXgOver) {
  1707→    score += 2;
  1708→    checks.push({ check: 'Both teams xG Over', points: 2, passed: true });
  1709→  } else {
  1710→    checks.push({ check: 'Both teams xG Over', points: 2, passed: false });
  1711→  }
  1712→
  1713→  // Check 4: BOTH teams average SOT > 5.5 (sustained pressure)
  1714→  const homeHighSot = input.homeAvgSot !== null && input.homeAvgSot > 5.5;
  1715→  const awayHighSot = input.awayAvgSot !== null && input.awayAvgSot > 5.5;
  1716→  if (homeHighSot && awayHighSot) {
  1717→    score += 2;
  1718→    checks.push({ check: `Both teams SOT > 5.5 (${input.homeAvgSot?.toFixed(1)} / ${input.awayAvgSot?.toFixed(1)})`, points: 2, passed: true });
  1719→  } else if (homeHighSot || awayHighSot) {
  1720→    checks.push({ check: `One team SOT > 5.5 (${input.homeAvgSot?.toFixed(1) ?? 'N/A'} / ${input.awayAvgSot?.toFixed(1) ?? 'N/A'})`, points: 2, passed: false });
  1721→  } else {
  1722→    checks.push({ check: `Neither team SOT > 5.5 (${input.homeAvgSot?.toFixed(1) ?? 'N/A'} / ${input.awayAvgSot?.toFixed(1) ?? 'N/A'})`, points: 2, passed: false });
  1723→  }
  1724→
  1725→  // Check 5: Balanced xG (within 0.8 of each other) — competitive game
  1726→  // We check if xG signals are similar (not one-sided)
  1727→  const xgBalanced = isXgOver(input.homeXgSignal) === isXgOver(input.awayXgSignal);
  1728→  if (xgBalanced) {
  1729→    score += 1;
  1730→    checks.push({ check: 'xG signals balanced (both teams similar)', points: 1, passed: true });
  1731→  } else {
  1732→    checks.push({ check: 'xG signals lopsided (one-sided)', points: 1, passed: false });
  1733→  }
  1734→
  1735→  // Check 6: Favorite odds 1.80–2.50 (open competitive game)
  1736→  if (input.favoriteOdds && input.favoriteOdds >= 1.80 && input.favoriteOdds <= 2.50) {
  1737→    score += 1;
  1738→    checks.push({ check: 'Competitive matchup (odds 1.80-2.50)', points: 1, passed: true });
  1739→  } else {
  1740→    checks.push({ check: 'Not a competitive matchup', points: 1, passed: false });
  1741→  }
  1742→
  1743→  // Penalty 7: Lopsided xG (one team xG Over, other not)
  1744→  if (!xgBalanced) {
  1745→    score -= 1;
  1746→    // Already counted in check 5 as non-passed, no duplicate
  1747→  }
  1748→
  1749→  // Penalty 8: Massive favorite (≤1.50) — might be controlled 2-0
  1750→  if (input.favoriteOdds && input.favoriteOdds <= 1.50) {
  1751→    score -= 1;
  1752→    checks.push({ check: 'Massive favorite (odds ≤1.50) — controlled game risk', points: -1, passed: true });
  1753→  } else {
  1754→    checks.push({ check: 'Massive favorite risk', points: -1, passed: false });
  1755→  }
  1756→
  1757→  // Penalty 9: Either team can't finish (SOT conversion <25%)
  1758→  const homeCantFinish = input.homeSotConversion !== null && input.homeSotConversion < 25;
  1759→  const awayCantFinish = input.awaySotConversion !== null && input.awaySotConversion < 25;
  1760→  if (homeCantFinish) score -= 1;
  1761→  if (awayCantFinish) score -= 1;
  1762→  checks.push({
  1763→    check: `Home SOT conversion ${homeCantFinish ? '< 25%' : '≥ 25%'}${input.homeSotConversion !== null ? ` (${input.homeSotConversion.toFixed(0)}%)` : ' (N/A)'}`,
  1764→    points: -1,
  1765→    passed: !homeCantFinish,
  1766→  });
  1767→  checks.push({
  1768→    check: `Away SOT conversion ${awayCantFinish ? '< 25%' : '≥ 25%'}${input.awaySotConversion !== null ? ` (${input.awaySotConversion.toFixed(0)}%)` : ' (N/A)'}`,
  1769→    points: -1,
  1770→    passed: !awayCantFinish,
  1771→  });
  1772→
  1773→  // Determine tier
  1774→  let tier: GoalTier;
  1775→  let recommendation: string;
  1776→  if (score >= 6) {
  1777→    tier = 'GOAL RICH';
  1778→    recommendation = 'Full confidence on Over 2.5, consider Over 3.5';
  1779→  } else if (score >= 4) {
  1780→    tier = 'GOAL LIKELY';
  1781→    recommendation = 'Solid Over 2.5 bet';
  1782→  } else if (score >= 2) {
  1783→    tier = 'GOAL BORDERLINE';
  1784→    recommendation = 'Over 1.5 strong, Over 2.5 cautious';
  1785→  } else if (score >= 0) {
  1786→    tier = 'GOAL THIN';
  1787→    recommendation = 'Over 1.5 only, skip Over 2.5';
  1788→  } else {
  1789→    tier = 'GOAL STALL';
  1790→    recommendation = 'Likely 1-2 goals, avoid Over markets above 1.5';
  1791→  }
  1792→
  1793→  return { tier, score, breakdown: checks, recommendation };
  1794→}

  1797→// Dominant Team Qualifier — Flags one-sided demolition games
  1798→// ============================================================================
  1799→// Catches games where one team is overwhelmingly stronger and likely to score
  1800→// 3+ goals regardless of opponent contribution. This is the anti-BTTS card —
  1801→// it identifies Elversberg 3-0, Furth 3-0, AZ 3-3 type profiles where Over
  1802→// markets hit but BTTS does NOT.
  1803→
  1804→export interface DominantTeamInput {
  1805→  favoriteOdds: number;
  1806→  favoriteAvgGoalsPerGame: number;
  1807→  underdogAvgConcededPerGame: number;
  1808→  favoriteXgPerGame: number;
  1809→  underdogXgPerGame: number;
  1810→  favoriteLast5Avg: number;
  1811→  favoriteSotPerGame: number | null;
  1812→  leagueAvgGoalsPerGame: number;
  1813→  underdogAwayBttsRate: number | null;
  1814→  favoriteLast3AllLow: boolean;  // true if last 3 games all < 1.5 total goals
  1815→}
  1816→
  1817→export type DominantTier = 'DOMINANT EXPECTED' | 'DOMINANT LIKELY' | 'DOMINANT POSSIBLE' | 'NOT DOMINANT';
  1818→
  1819→export interface DominantTeamResult {
  1820→  tier: DominantTier;
  1821→  score: number;
  1822→  breakdown: { check: string; points: number; passed: boolean }[];
  1823→  over25Rec: string;
  1824→  bttsRec: string;
  1825→}

  1827→export function computeDominantTeamQualifier(input: DominantTeamInput): DominantTeamResult {
  1828→  let score = 0;
  1829→  const checks: { check: string; points: number; passed: boolean }[] = [];
  1830→
  1831→  // Positive checks
  1832→  const favOddsLow = input.favoriteOdds <= 1.40;
  1833→  if (favOddsLow) { score += 2; }
  1834→  checks.push({ check: `Favorite odds ${input.favoriteOdds.toFixed(2)} ${favOddsLow ? '<= 1.40' : '> 1.40'}`, points: 2, passed: favOddsLow });
  1835→
  1836→  const favHighAvg = input.favoriteAvgGoalsPerGame > 2.0;
  1837→  if (favHighAvg) { score += 2; }
  1838→  checks.push({ check: `Favorite avg ${input.favoriteAvgGoalsPerGame.toFixed(1)} goals/game ${favHighAvg ? '> 2.0' : '<= 2.0'}`, points: 2, passed: favHighAvg });
  1839→
  1840→  const underdogLeaks = input.underdogAvgConcededPerGame > 1.5;
  1841→  if (underdogLeaks) { score += 1; }
  1842→  checks.push({ check: `Underdog concedes ${input.underdogAvgConcededPerGame.toFixed(1)}/game ${underdogLeaks ? '> 1.5' : '<= 1.5'}`, points: 1, passed: underdogLeaks });
  1843→
  1844→  const xgGap = input.favoriteXgPerGame - input.underdogXgPerGame;
  1845→  const hugeXgGap = xgGap > 1.0;
  1846→  if (hugeXgGap) { score += 2; }
  1847→  checks.push({ check: `xG gap ${xgGap.toFixed(1)} ${hugeXgGap ? '> 1.0' : '<= 1.0'}`, points: 2, passed: hugeXgGap });
  1848→
  1849→  const favLast5Hot = input.favoriteLast5Avg > 2.0;
  1850→  if (favLast5Hot) { score += 1; }
  1851→  checks.push({ check: `Favorite last 5 avg ${input.favoriteLast5Avg.toFixed(1)} ${favLast5Hot ? '> 2.0' : '<= 2.0'}`, points: 1, passed: favLast5Hot });
  1852→
  1853→  const favHighSot = input.favoriteSotPerGame !== null && input.favoriteSotPerGame > 6.0;
  1854→  if (favHighSot) { score += 1; }
  1855→  checks.push({ check: `Favorite SOT ${input.favoriteSotPerGame?.toFixed(1) ?? 'N/A'}/game ${favHighSot ? '> 6.0' : '<= 6.0'}`, points: 1, passed: favHighSot });
  1856→
  1857→  const variance = classifyLeagueVariance(input.leagueAvgGoalsPerGame);
  1858→  const isHighVar = variance === 'high';
  1859→  if (isHighVar) { score += 1; }
  1860→  checks.push({ check: `High-variance league (${input.leagueAvgGoalsPerGame.toFixed(1)} avg)`, points: 1, passed: isHighVar });
  1861→
  1862→  const underdogBttsPoor = input.underdogAwayBttsRate !== null && input.underdogAwayBttsRate < 40;
  1863→  if (underdogBttsPoor) { score += 1; }
  1864→  checks.push({ check: `Underdog away BTTS ${input.underdogAwayBttsRate !== null ? `${input.underdogAwayBttsRate.toFixed(0)}%` : 'N/A'} ${underdogBttsPoor ? '< 40%' : '>= 40%'}`, points: 1, passed: underdogBttsPoor });
  1865→
  1866→  // Penalties
  1867→  if (input.favoriteLast3AllLow) { score -= 2; }
  1868→  checks.push({ check: 'Favorite cooling off (last 3 < 1.5 goals)', points: -2, passed: input.favoriteLast3AllLow });
  1869→
  1870→  // Determine tier
  1871→  let tier: DominantTier;
  1872→  let over25Rec: string;
  1873→  let bttsRec: string;
  1874→  if (score >= 6) {
  1875→    tier = 'DOMINANT EXPECTED'; over25Rec = 'Strong Over 2.5'; bttsRec = 'BTTS NO (one-sided)';
  1876→  } else if (score >= 4) {
  1877→    tier = 'DOMINANT LIKELY'; over25Rec = 'Over 2.5'; bttsRec = 'BTTS NO';
  1878→  } else if (score >= 2) {
  1879→    tier = 'DOMINANT POSSIBLE'; over25Rec = 'Lean Over 2.5'; bttsRec = 'Depends on opponent';
  1880→  } else {
  1881→    tier = 'NOT DOMINANT'; over25Rec = 'N/A'; bttsRec = 'N/A';
  1882→  }
  1883→
  1884→  return { tier, score, breakdown: checks, over25Rec, bttsRec };
  1885→}

  1887→// ============================================================================
  1888→// BTTS BOTH HALVES DETECTOR
  1889→// ============================================================================
  1890→// Dedicated detector for "Both Teams Score in Both Halves" (BTTS-BH).
  1891→// This is the rarest and most profitable goal pattern — ~5.2% base rate.
  1892→//
  1893→// Design: Lean 3-check system, calibrated on 24,057 matches across 7 leagues.
  1894→// Only signals with proven discriminative power survive:
  1895→//   1. O2.5 implied probability (bookmaker odds) — 1.38x lift, PRIMARY
  1896→//   2. Draw probability < 25% — 1.21x lift, SECONDARY
  1897→//   3. Rolling combined scoring >= 3.0 — 1.10x lift, TERTIARY
  1898→//
  1899→// All other model-derived signals (BTTS rate, O3.5 rate, O2.5 model rate,
  1900→// league avg goals, elite O2.5) were tested and found flat (0.99-1.06x)
  1901→// across 24K games. They were removed to reduce noise.
  1902→//
  1903→// Need 2 of 3 checks to qualify.
  1904→// ============================================================================
  1905→
  1906→export type BTTSBothHalvesTier =
  1907→  | 'BTTS-BH STRONG'
  1908→  | 'BTTS-BH QUALIFIED'
  1909→  | 'BTTS-BH BORDERLINE'
  1910→  | 'BTTS-BH UNLIKELY';
  1911→
  1912→export interface BTTSBothHalvesInput {
  1913→  o25Prob: number;              // O2.5 model/rolling probability (unused — kept for interface compat)
  1914→  o35Prob: number;              // O3.5 rolling rate (unused — kept for interface compat)
  1915→  bttsProb: number;             // BTTS rolling rate (unused — kept for interface compat)
  1916→  rollingCombinedScoring: number;
  1917→  o25ImpliedProb: number | null; // from bookmaker O2.5 odds
  1918→  drawProb: number;             // draw probability from odds
  1919→  avgGoalsPerGame: number;      // league average goals (unused — kept for interface compat)

  1922→export interface BTTSBothHalvesResult {
  1923→  isBTTSBothHalves: boolean;
  1924→  tier: BTTSBothHalvesTier;
  1925→  score: number;
  1926→  totalChecks: number;
  1927→  breakdown: { check: string; passed: boolean; weight: string }[];
  1928→  reasoning: string[];

  1930→
  1931→/**
  1932→ * Compute BTTS Both Halves indicator.
  1933→ *
  1934→ * Calibrated on 24,057 matches across 7 European leagues (10 seasons each).
  1935→ * 1,249 BTTS-BH games (5.19% base rate).
  1936→ *
  1937→ * 3-check system — only signals with proven lift survive:
  1938→ *   - O2.5 implied >= 65%: 1.38x lift (strongest single predictor)
  1939→ *   - Draw prob < 25%:      1.21x lift (low draw = open, balanced play)
  1940→ *   - Rolling scoring >= 3.0: 1.10x lift (recent form, independent)
  1941→ *
  1942→ * Tier system:
  1943→ *   STRONG (3/3):    All signals aligned — rare, highest confidence
  1944→ *   QUALIFIED (2/3): Core checks pass — actionable signal
  1945→ *   BORDERLINE (1/3): One indicator only — watch list
  1946→ *   UNLIKELY (0/3):  Insufficient evidence
  1947→ */
  1948→export function computeBTTSBothHalves(input: BTTSBothHalvesInput): BTTSBothHalvesResult {
  1949→  const cfg = BTTS_BOTH_HALVES_CONFIG;
  1950→  const reasoning: string[] = [];
  1951→
  1952→  const checks: { check: string; passed: boolean; weight: string }[] = [
  1953→    // CHECK 1: O2.5 implied probability — THE strongest predictor (1.38x lift at 65%)
  1954→    {
  1955→      check: 'O2.5 Implied >= ' + cfg.o25Implied + '%',
  1956→      passed: input.o25ImpliedProb !== null && input.o25ImpliedProb >= cfg.o25Implied,
  1957→      weight: 'CRITICAL',
  1958→    },
  1959→
  1960→    // CHECK 2: Draw probability INVERTED — lower draw = better for BTTS-BH
  1961→    // Multi-league calibration: draw < 25% gives 1.21x lift
  1962→    {
  1963→      check: 'Draw Prob < ' + cfg.drawProbMax + '%',
  1964→      passed: input.drawProb < cfg.drawProbMax,
  1965→      weight: 'HIGH',
  1966→    },
  1967→
  1968→    // CHECK 3: Rolling combined scoring — both teams scoring recently
  1969→    // 1.10x lift at 3.0, marginal but independent of bookmaker signals
  1970→    {
  1971→      check: 'Rolling Scoring >= ' + cfg.rollingScoring,
  1972→      passed: input.rollingCombinedScoring >= cfg.rollingScoring,
  1973→      weight: 'MEDIUM',
  1974→    },
  1975→  ];
  1976→
  1977→  const score = checks.filter(c => c.passed).length;
  1978→
  1979→  // Build reasoning
  1980→  if (input.o25ImpliedProb !== null && input.o25ImpliedProb >= cfg.o25Implied) {
  1981→    reasoning.push('Bookmakers imply high goal expectation (' + input.o25ImpliedProb.toFixed(0) + '% O2.5). This is the strongest BTTS-BH predictor (1.38x lift across 24K games).');
  1982→  }
  1983→  if (input.drawProb < cfg.drawProbMax) {
  1984→    reasoning.push('Low draw probability (' + input.drawProb.toFixed(0) + '%) — balanced matchup favors open play over defensive draws (1.21x lift).');
  1985→  }
  1986→  if (input.rollingCombinedScoring >= cfg.rollingScoring) {
  1987→    reasoning.push('Both teams showing recent scoring form (' + input.rollingCombinedScoring.toFixed(1) + ' combined goals last 5).');
  1988→  }
  1989→  if (reasoning.length === 0) {
  1990→    reasoning.push('Insufficient signal alignment for BTTS in both halves. O2.5 implied probability is the primary gate.');
  1991→  }
  1992→
  1993→  // Determine tier
  1994→  let tier: BTTSBothHalvesTier;
  1995→  if (score >= 3) {
  1996→    tier = 'BTTS-BH STRONG';
  1997→  } else if (score >= 2) {
  1998→    tier = 'BTTS-BH QUALIFIED';
  1999→  } else if (score >= 1) {
  2000→    tier = 'BTTS-BH BORDERLINE';
  2001→  } else {
  2002→    tier = 'BTTS-BH UNLIKELY';
  2003→  }
  2004→
  2005→  return {
  2006→    isBTTSBothHalves: score >= cfg.requiredChecks,
  2007→    tier,
  2008→    score,
  2009→    totalChecks: checks.length,
  2010→    breakdown: checks,
  2011→    reasoning,
  2012→  };
  2013→}

  1001→  1001→

  1036→  1036→// ============================================================================
  1037→  1037→// Bookie Odds Dampener — Heavy Favorite Adjustment
  1038→  1038→// ============================================================================
  1039→  1039→// Based on live results analysis: when bookie odds for one side are <=1.70,
  1040→  1040→// the model systematically overestimates O2.5 probability by 10-15%.
  1041→  1041→// Heavy favorites tend to win tight, controlling games (1-0, 2-0) rather
  1042→  1042→// than high-scoring shootouts. This dampener accounts for that pattern.
  1043→  1043→// ============================================================================
  1044→  1044→
  1045→  1045→/** Heavy favorite odds threshold (decimal odds) */
  1046→  1046→const HEAVY_FAVORITE_ODDS = 1.70;
  1047→  1047→
  1048→  1048→/** Maximum dampener amount (percentage points to subtract) */
  1049→  1049→const HEAVY_FAVORITE_DAMPENER = 10;
  1050→  1050→
  1051→  1051→/**
  1052→  1052→ * Apply bookie odds dampener to goal market probabilities.
  1053→  1053→ * When either team is a heavy favorite (avg odds <=1.70), reduce O2.5/O3.5/BTTS
  1054→  1054→ * probabilities to account for the "park the bus" effect.
  1055→  1055→ *
  1056→  1056→ * Only applies when valid odds are available (non-null, positive).
  1057→  1057→ *
  1058→  1058→ * @param over25Prob  - Model O2.5 probability (%)
  1059→  1059→ * @param over35Prob  - Model O3.5 probability (%)
  1060→  1060→ * @param bttsProb    - Model BTTS probability (%)
  1061→  1061→ * @param homeAvgOdds - Home team average odds (decimal)
  1062→  1062→ * @param awayAvgOdds - Away team average odds (decimal)
  1063→  1063→ * @returns Adjusted probabilities with dampener applied
  1064→  1064→ */
  1065→  1065→export function applyBookieOddsDampener(
  1066→  1066→  over25Prob: number,
  1067→  1067→  over35Prob: number,
  1068→  1068→  bttsProb: number,
  1069→  1069→  homeAvgOdds: number | null | undefined,
  1070→  1070→  awayAvgOdds: number | null | undefined,
  1071→  1071→  leagueAvgGoalsPerGame?: number
  1072→  1072→): { over25: number; over35: number; btts: number; dampened: boolean; reason: string } {
  1073→  1073→  if (!homeAvgOdds || !awayAvgOdds || homeAvgOdds <= 0 || awayAvgOdds <= 0) {
  1074→  1074→    return { over25: over25Prob, over35: over35Prob, btts: bttsProb, dampened: false, reason: '' };
  1075→  1075→  }
  1076→  1076→
  1077→  1077→  // League-adaptive: gentler dampener in high-variance leagues
  1078→  1078→  const variance = leagueAvgGoalsPerGame ? classifyLeagueVariance(leagueAvgGoalsPerGame) : 'low';
  1079→  1079→  const heavyThreshold = variance === 'high' ? 1.40 : variance === 'medium' ? 1.60 : HEAVY_FAVORITE_ODDS;
  1080→  1080→  const maxDampener = variance === 'high' ? 5 : variance === 'medium' ? 7 : HEAVY_FAVORITE_DAMPENER;
  1081→  1081→
  1082→  1082→  const homeIsHeavy = homeAvgOdds <= heavyThreshold;
  1083→  1083→  const awayIsHeavy = awayAvgOdds <= heavyThreshold;
  1084→  1084→
  1085→  1085→  if (!homeIsHeavy && !awayIsHeavy) {
  1086→  1086→    return { over25: over25Prob, over35: over35Prob, btts: bttsProb, dampened: false, reason: '' };
  1087→  1087→  }
  1088→  1088→
  1089→  1089→  const heavyOdds = Math.min(homeAvgOdds, awayAvgOdds);
  1090→  1090→  // Scale dampener: heavier favorite = bigger reduction
  1091→  1091→  // At 1.17 (PSG level): full dampener
  1092→  1092→  // At threshold: minimal 2% dampener
  1093→  1093→  const minDampener = variance === 'high' ? 2 : 3;
  1094→  1094→  const scale = Math.max(0, 1 - (heavyOdds - 1.17) / (heavyThreshold - 1.17));
  1095→  1095→  const dampener = Math.round(minDampener + (maxDampener - minDampener) * scale);
  1096→  1096→
  1097→  1097→  const leagueLabel = variance.charAt(0).toUpperCase() + variance.slice(1);
  1098→  1098→  const reason = homeIsHeavy
  1099→  1099→    ? `${leagueLabel}-variance: Heavy home favorite (${homeAvgOdds.toFixed(2)}): -${dampener}% dampener`
  1100→  1100→    : `${leagueLabel}-variance: Heavy away favorite (${awayAvgOdds.toFixed(2)}): -${dampener}% dampener`;
  1101→  1101→
  1102→  1102→  return {
  1103→  1103→    over25: Math.max(30, over25Prob - dampener),
  1104→  1104→    over35: Math.max(20, over35Prob - dampener),
  1105→  1105→    btts: Math.max(25, bttsProb - Math.round(dampener * 0.5)), // BTTS dampened less
  1106→  1106→    dampened: true,
  1107→  1107→    reason,
  1108→  1108→  };
  1109→  1109→}

  1111→  1111→/**
  1112→  1112→ * Compute average bookie odds for a team from match results.
  1113→  1113→ * Averages odds across all matches for that team.
  1114→  1114→ *
  1115→  1115→ * @param matches - Match results with odds data
  1116→  1116→ * @param team    - Team name
  1117→  1117→ * @returns Average decimal odds, or null if insufficient data
  1118→  1118→ */
  1119→  1119→export function computeTeamAvgOdds(
  1120→  1120→  matches: { homeTeam: string; awayTeam: string; oddsAvgHome: number | null; oddsAvgAway: number | null }[],
  1121→  1121→  team: string
  1122→  1122→): number | null {
  1123→  1123→  const teamOdds: number[] = [];
  1124→  1124→
  1125→  1125→  for (const m of matches) {
  1126→  1126→    const odds = m.homeTeam === team ? m.oddsAvgHome : (m.awayTeam === team ? m.oddsAvgAway : null);
  1127→  1127→    if (odds && odds > 1 && odds < 20) {
  1128→  1128→      teamOdds.push(odds);
  1129→  1129→    }
  1130→  1130→  }
  1131→  1131→
  1132→  1132→  if (teamOdds.length < 3) return null;
  1133→  1133→
  1134→  1134→  return teamOdds.reduce((s, o) => s + o, 0) / teamOdds.length;
  1135→  1135→}

  1138→  1138→// League Variance Classification — Used by blowout/dampener adaptivity
  1139→  1139→// ============================================================================
  1140→  1140→// Classifies a league by its historical goal variance so that blowout and
  1141→  1141→// dampener thresholds can be calibrated to the league's natural scoring profile.
  1142→  1142→// High-variance leagues (2. Bundesliga, Eredivisie) see more goal explosions
  1143→  1143→// from heavy favorites, so suppression mechanisms should be gentler.
  1144→  1144→

  1146→  1146→
  1147→  1147→export function classifyLeagueVariance(avgGoalsPerGame: number): LeagueVariance {
  1148→  1148→  if (avgGoalsPerGame >= 3.0) return 'high';
  1149→  1149→  if (avgGoalsPerGame >= 2.5) return 'medium';
  1150→  1150→  return 'low';
  1151→  1151→}

  1153→  1153→// ============================================================================
  1154→  1154→// Blowout Risk Qualifier — Override Under signals in massive quality gaps
  1155→  1155→// ============================================================================
  1156→  1156→// Based on live results analysis: when there's a huge quality gap (favorite's
  1157→  1157→// avg odds <= 1.50), the Regression "Under" signal is unreliable. Heavy favorites
  1158→  1158→// tend to dominate regardless of recent form — a "cooling off" period doesn't
  1159→  1159→// matter when you're 2-3 quality tiers above the opponent.
  1160→  1160→//
  1161→  1161→// This qualifier prevents the "Under" regression signal from canceling out
  1162→  1162→// legitimate "Over" signals (xG, Z-Score) in STRONG BET / GREY RESULT / GOAL FEST.
  1163→  1163→//
  1164→  1164→// Example: Man City (odds 1.20) vs Luton — Regression says "Under" (City only
  1165→  1165→// scored 1 goal last 3 games), but City will likely score 3-4 regardless.
  1166→  1166→// Without this qualifier, the Under signal would block 2 points in STRONG BET.
  1167→  1167→// ============================================================================
  1168→  1168→
  1169→  1169→/** Blowout threshold: favorite's average odds at or below this triggers override */
  1170→  1170→const BLOWOUT_ODDS_THRESHOLD = 1.50;
  1171→  1171→
  1172→  1172→/**
  1173→  1173→ * Apply blowout risk qualifier to regression signal.
  1174→  1174→ * When either team is a massive favorite (avg odds <= 1.50), override
  1175→  1175→ * Regression "Under"/"Strong Under" to "Neutral" because quality advantage
  1176→  1176→ * outweighs recent form in one-sided matchups.
  1177→  1177→ *
  1178→  1178→ * @param signals      - Current signal set (xgSignal, regressionSignal, zScoreSignal)
  1179→  1179→ * @param homeAvgOdds  - Home team average odds (decimal)
  1180→  1180→ * @param awayAvgOdds  - Away team average odds (decimal)
  1181→  1181→ * @returns Modified signals (regression may be overridden to Neutral)
  1182→  1182→ */
  1183→  1183→export function applyBlowoutQualifier(
  1184→  1184→  signals: SignalInput,
  1185→  1185→  homeAvgOdds: number | null | undefined,
  1186→  1186→  awayAvgOdds: number | null | undefined,
  1187→  1187→  leagueAvgGoalsPerGame?: number
  1188→  1188→): { signals: SignalInput; blowoutOverride: boolean; reason: string } {
  1189→  1189→  // Only applies when valid odds are available
  1190→  1190→  if (!homeAvgOdds || !awayAvgOdds || homeAvgOdds <= 0 || awayAvgOdds <= 0) {
  1191→  1191→    return { signals, blowoutOverride: false, reason: '' };
  1192→  1192→  }
  1193→  1193→
  1194→  1194→  // League-adaptive threshold: gentler in high-variance leagues
  1195→  1195→  const variance = leagueAvgGoalsPerGame ? classifyLeagueVariance(leagueAvgGoalsPerGame) : 'low';
  1196→  1196→  const blowoutThreshold = variance === 'high' ? 1.25 : variance === 'medium' ? 1.40 : BLOWOUT_ODDS_THRESHOLD;
  1197→  1197→
  1198→  1198→  const homeIsMassiveFavorite = homeAvgOdds <= blowoutThreshold;
  1199→  1199→  const awayIsMassiveFavorite = awayAvgOdds <= blowoutThreshold;
  1200→  1200→
  1201→  1201→  // Only trigger when one team is a massive favorite
  1202→  1202→  if (!homeIsMassiveFavorite && !awayIsMassiveFavorite) {
  1203→  1203→    return { signals, blowoutOverride: false, reason: '' };
  1204→  1204→  }
  1205→  1205→
  1206→  1206→  // In high-variance leagues, only override Strong Under (keep mild signals intact)
  1207→  1207→  // because hot favorites in goal-rich leagues often score freely
  1208→  1208→  const shouldOverride = variance === 'high'
  1209→  1209→    ? signals.regressionSignal === 'Strong Under'
  1210→  1210→    : signals.regressionSignal === 'Under' || signals.regressionSignal === 'Strong Under';
  1211→  1211→  if (!shouldOverride) {
  1212→  1212→    return { signals, blowoutOverride: false, reason: '' };
  1213→  1213→  }
  1214→  1214→
  1215→  1215→  const favoriteOdds = Math.min(homeAvgOdds, awayAvgOdds);
  1216→  1216→  const favorite = homeIsMassiveFavorite ? 'Home' : 'Away';
  1217→  1217→  const reason = `Blowout qualifier (${variance}-variance): ${favorite} favorite (${favoriteOdds.toFixed(2)} <= ${blowoutThreshold}) — Regression "${signals.regressionSignal}" overridden to Neutral`;
  1218→  1218→
  1219→  1219→  return {
  1220→  1220→    signals: {
  1221→  1221→      ...signals,
  1222→  1222→      regressionSignal: 'Neutral',
  1223→  1223→    },
  1224→  1224→    blowoutOverride: true,
  1225→  1225→    reason,
  1226→  1226→  };
  1227→  1227→}

  1329→  1329→  matchSignal: string;     // 'MOMENTUM OVER' | 'MOMENTUM LEAN OVER' | 'MOMENTUM NEUTRAL' | 'MOMENTUM LEAN UNDER' | 'MOMENTUM UNDER'

  1331→  1331→
  1332→  1332→function classifyTeamMomentum(team: MomentumTeamInput): string {
  1333→  1333→  if (team.last5Goals.length < 3) return 'Neutral';
  1334→  1334→
  1335→  1335→  const last5Avg = team.last5Goals.reduce((s, g) => s + g, 0) / team.last5Goals.length;
  1336→  1336→  const last3 = team.last5Goals.slice(0, Math.min(3, team.last5Goals.length));
  1337→  1337→  const last3Avg = last3.reduce((s, g) => s + g, 0) / last3.length;
  1338→  1338→
  1339→  1339→  // Consistency: coefficient of variation of last 5
  1340→  1340→  const mean = last5Avg;
  1341→  1341→  const variance = team.last5Goals.reduce((s, g) => s + (g - mean) ** 2, 0) / team.last5Goals.length;
  1342→  1342→  const stdDev = Math.sqrt(variance);
  1343→  1343→  const cv = mean > 0 ? stdDev / mean : 999;
  1344→  1344→
  1345→  1345→  // Sustained: 5+ game window above season average by meaningful margin
  1346→  1346→  const sustained = last5Avg > team.seasonAvg * 1.15;
  1347→  1347→
  1348→  1348→  // Accelerating: last 3 even hotter than last 5
  1349→  1349→  const accelerating = last3Avg > last5Avg * 1.05;
  1350→  1350→
  1351→  1351→  // xG sustainability: overperforming xG with shot volume to back it
  1352→  1352→  const xgSustainable = team.xgDiff > 0 && team.sotPerGame !== null && team.sotPerGame > 5.0;
  1353→  1353→
  1354→  1354→  // Strong Momentum: sustained + consistent + high output
  1355→  1355→  if (sustained && cv < 0.5 && last5Avg >= 2.0) return 'Strong Momentum';
  1356→  1356→  // Momentum: sustained with moderate variance, or recent surge with sustainable backing
  1357→  1357→  if (sustained && cv < 0.65) return 'Momentum';
  1358→  1358→  if (last5Avg > team.seasonAvg * 1.15 && accelerating && (xgSustainable || team.xgDiff <= 0)) return 'Momentum';
